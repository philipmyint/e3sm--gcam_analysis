   52  def process_inputs(inputs):
   53      """
   54      Processes a dictionary of inputs (keys are options, values are choices for those options) for creating spatial plots from data in NetCDF files.
   55
   56      Parameters:
   57          inputs: Dictionary containing the user plotting choice inputs for different options. This dictionary may be incomplete or have invalid values.
   58
   59      Returns:
   60          List of dictionaries, each of which specifies completed plotting options for a single variable. If the user did not select a plotting option
   61          for a particular variable, the default choice for that plotting option will be selected.
   62      """
   63      # If the NetCDF files are specified as a list of lists (for ensemble plots), check if they need to be transposed.
   64      # Users can now specify output files in two formats:
   65      #   Format A (organized by ensemble member - original format):
   66      #       [["Control", "Full feedback"], ["Control_2", "Full feedback_2"], ...]
   67      #   Format B (organized by file set - new user-friendly format):
   68      #       [["Control", "Control_2", ...], ["Full feedback", "Full feedback_2", ...]]
   69      # The plotting functions expect Format A internally, so Format B will be automatically transposed.
   70      if check_is_list_of_lists(inputs['netcdf_files']):
   71          inputs['netcdf_files'], was_transposed = transpose_scenarios_if_needed(inputs['netcdf_files'])
   72          notify_netcdf_files_transposed = inputs.get('notify_netcdf_files_transposed', False)
   73          if was_transposed and notify_netcdf_files_transposed:
   74              print(f"Note: NetCDF files were automatically transposed from 'organized by file set' format to 'organized by ensemble member' format.")
   75
   76      # Turn the list of NetCDF files and their corresponding labels into a list of lists, if they are not already in that form.
   77      if not check_is_list_of_lists(inputs['netcdf_files']):
   78          if isinstance(inputs['netcdf_files'], str):
   79              inputs['netcdf_files'] = [[inputs['netcdf_files']]]
   80          elif isinstance(inputs['netcdf_files'], list):
   81              inputs['netcdf_files'] = [inputs['netcdf_files']]
   82
   83      # Collect all NetCDF file paths from the (potentially nested) list and check that each one exists.
   84      netcdf_files_flat = []
   85      raw_files = inputs['netcdf_files']
   86      if check_is_list_of_lists(raw_files):
   87          for sublist in raw_files:
   88              netcdf_files_flat.extend(sublist)
   89      elif isinstance(raw_files, list):
   90          netcdf_files_flat.extend(raw_files)
   91      else:
   92          netcdf_files_flat.append(raw_files)
   93      missing_files = [f for f in netcdf_files_flat if not os.path.exists(f)]
   94      if missing_files:
   95          raise FileNotFoundError(f"Error: the following NetCDF files were not found: {missing_files}")
   96
   97      # Read one of the NetCDF files into an xarray Dataset so that we can later get the variables contained in them and the units of these variables.
   98      ds = xr.open_dataset(inputs['netcdf_files'][0][0] if check_is_list_of_lists(inputs['netcdf_files']) else
   99                           inputs['netcdf_files'][0] if isinstance(inputs['netcdf_files'], list) else

  108          inputs['variables'] = variables
  109      # If the user entered a string to indicate a single variable, put that string in a list.
  110      if isinstance(variables, str):
  111          variables = [variables]
  112      variables = [variable for variable in variables if not variable.upper().startswith(('AREA', 'LANDFRAC'))]
  113      inputs['variables'] = variables
  114      if not variables:
  115          ds.close()
  116          raise ValueError('Error: no plottable variables remain after excluding AREA and LANDFRAC fields.')
  117
  118      # Check that all requested variables exist in the first NetCDF file.
  119      missing_vars = [v for v in variables if v not in ds]
  120      if missing_vars:
  121          print(f"Warning: the following variables were not found in '{netcdf_files_flat[0]}' and will be skipped: {missing_vars}")
  122          variables = [v for v in variables if v not in missing_vars]
  123          inputs['variables'] = variables
  124          if not variables:
  125              print(f"Warning: no valid variables remain. Skipping '{inputs.get('plot_name', 'plot')}'.")
  126              ds.close()
  127              return []
  128
  129      # For the plotting options that have not been specified in the inputs dictionary, add keys for them if necessary and use the default values.
  130      for key in default_inputs.keys():
  131          if key not in inputs:
  132              inputs[key] = default_inputs[key]
  133
  134      # If the user specified anything other than a dictionary (e.g., a single value [string, integer, float] or a list) for the other plotting options,
  135      # assume that they want to use that value/list for all the variables. Enable this by creating dictionaries with the keys given by the variables.
  136      # This also covers the case above, where default values were added for all plotting options that are not specified in the inputs dictionary.
  137      for key, value in inputs.items():
  138          if key != 'variables':
  139              if not isinstance(value, dict):
  140                  inputs[key] = dict.fromkeys(variables, value)
  141
  142      # For each variable, if a plotting option is missing (has not been specified), fill in with the default corresponding to that plotting option.
  143      if 'plot_name' not in inputs:
  144          inputs['plot_name'] = {}
  145      if 'title' not in inputs:
  146          inputs['title'] = {}
  147      for variable in variables:
  148          # Use the default for the plot directory and create the directory if it does not already exist.
  149          if not any(key == variable for key in inputs['plot_directory'].keys()):
  150              inputs['plot_directory'][variable] = default_inputs['plot_directory']
  151          if not os.path.exists(inputs['plot_directory'][variable]):
  152              os.makedirs(inputs['plot_directory'][variable])
  153          # Default for the plot names is to call it 'spatial_[var_name]', where '[var_name]' is the name of the variable.
  154          if not any(key == variable for key in inputs['plot_name'].keys()):
  155              inputs['plot_name'][variable] = os.path.join(inputs['plot_directory'][variable], 'spatial_' + variable)
  156          # Append the plot directory with the name of the p-value file.
  157              inputs['p_value_file'][variable] = os.path.join(inputs['plot_directory'][variable], inputs['p_value_file'][variable])
  158          # Default for the title of a variable is to use the column header for that variable from the Dataset.
  159          if not any(key == variable for key in inputs['title'].keys()):
  160              # Replace ^2 with $^2$ in the units for the variable, so that the intended exponent (e.g., m^2) gets rendered correctly.
  161              units = ds[variable].attrs['units'].replace('^2', '$^2$')
  162              # Replace /m2 with /m$^2$ in the units for the variable, so that it gets rendered correctly.
  163              units = units.replace('/m2', '/m$^2$')
  164              inputs['title'][variable] = rf'{variable} ({units})'
  165          # Default for the other plotting options are specified in the default_inputs dictionary.
  166          for key, value in inputs.items():
  167              if key not in ['variables', 'plot_directory', 'plot_name', 'title']:
  168                  if not any(value_key == variable for value_key in value.keys()):
  169                      inputs[key][variable] = default_inputs[key]
  170
  171      # Close the dataset now that all variable and unit information has been extracted.
  172      ds.close()
  173
  174      # Check that the grid file exists if one has been specified (required for EAM plots).
  175      grid_file = inputs['grid_file'][variables[0]] if isinstance(inputs['grid_file'], dict) else inputs['grid_file']
  176      if grid_file and not os.path.exists(grid_file):
  177          raise FileNotFoundError(f"Error: grid file not found: '{grid_file}'")
  178
  179      # Now that the dictionary has been populated with complete plotting options for each variable, separate it into a list of dictionaries,
  180      # where each of these smaller dictionaries contain the plotting options for a single variable. Return this list of dictionaries.
  181      list_of_inputs = []
  182      for variable in inputs['variables']:
  183          inputs_for_this_variable = {'variable': variable}
  184          for input_option in inputs.keys():
  185              if input_option != 'variables':
  186                  inputs_for_this_variable[input_option] = inputs[input_option][variable]
  187          list_of_inputs.append(inputs_for_this_variable)
  188      return list_of_inputs

  240      # Store the grid file in an uxarray Dataset.
  241      grid = ux.open_grid(grid_file)

  243      # We either have individual plots, in which case there could be multiple files arranged like [[file1, file2, file3, ...]],
  244      # or we could have ensemble plots, in which case there are at most two data sets, but potentially multiple files in each of the sets.
  245      # The preprocessing done earlier has formatted the files into a list of lists so that they are in the form:
  246      #   For individual plots: [[file1, file2, file3, ...]]
  247      #   For ensemble plots: [["Control", "Full feedback"], ["Control_2", "Full feedback_2"], ...]
  248      num_files_in_each_set = len(netcdf_files)
  249      num_file_sets = len(netcdf_files[0])
  250
  251      # Note that unlike time series, it is not meaningful to have a spatial plot with more than two ensembles,
  252      # which is why we are restricting the total number of data sets to 2 when producing ensemble plots.
  253      # Note that num_file_sets and num_files_in_each_set represent different things depending on whether there
  254      #    are one or two ensembles.
  255      # If there is one ensemble (of any length, including 1 file):
  256      #    num_file_sets = the number of files in the ensemble
  257      #    num_files_in_each_set = 1 (the number of ensembles)
  258      #    only mean or sum can be calculated across the files and plotted (no comparison)
  259      # If there are two ensembles (or more, but the code exits if there are more than two):
  260      #    num_file_sets = 2 (the number of ensembles)
  261      #    num_files_in_each_set = the number of files in each ensemble
  262      #    the ensemble means are calculated and can be plotted separately or as one of two difference comparisons
  263
  264      if (num_files_in_each_set == 1) or (num_file_sets == 2):
  265          pass
  266      else:
  267          error_message = "Error: For spatial plots, there can be only one or two sets of netcdf_files, " \
  268              + "and if there are two sets there must be at least two files in each set (for ensemble plots)."
  269          raise ValueError(error_message)

  271      # Read each NetCDF output file, aggregate the requested variable over the selected years, and store the results in a plain xarray Dataset.
  272      # We avoid using uxarray as the intermediate container because its Dataset constructor is incompatible with xarray >= 2025.7.1 here.
  273      # The final xarray results are converted to uxarray only at the plotting step, where the unstructured grid is needed.
  274      uxds = xr.Dataset()
  275      area_weights = None
  276      for file_set_index in range(num_file_sets):
  277          for file_index in range(num_files_in_each_set):

  280              # so the error message can report the available range rather than checking an already-filtered dataset.
  281              ds_full = xr.open_dataset(file)

  288                      ds_full.close()
  289                      raise ValueError(f"Plot start_year {start_year} not in '{file}' (available: {file_start}–{file_end}); update input json file")

  291                      ds_full.close()
  292                      raise ValueError(f"Plot end_year {end_year} not in '{file}' (available: {file_start}–{file_end}); update input json file")
  293

  295                  # Include plot_type in the name to prevent collisions between abs_diff and pct_diff workers running in parallel.
  296                  ds = ds_full.sel(year=slice(start_year, end_year))[variable]
  297                  ds_full.close()

  299                  ds.to_netcdf(wfile, 'w')

  302                  if time_calculation == 'mean':
  303                      xr_ds = xr.open_dataset(wfile).mean(dim='year')
  304                      uxda = xr_ds[variable]*multiplier
  305                      xr_ds.close()

  308                      uxda = xr_ds[variable]*multiplier
  309                      xr_ds.close()

  311                      per_time_labels = ['/year', '/month', '/day', '/hour', '/min', '/s']
  312                      time_multipliers = np.array([1, years_TO_months, years_TO_days, years_TO_hours, years_TO_mins, years_TO_s])

  314                          if per_time_label in title:
  315                              title = title.replace(per_time_label, '')

  339          columns_test_set = [column for column in df.columns if column.endswith(f'_1')]

  344          ttest = stats.ttest_ind(df_control_set, df_test_set)
  345          print_p_values(ttest, variable, p_value_threshold, p_value_file, plot_name, p_value_file_print_only_if_below_threshold)

  380          uxDataArrays_to_plot.append(convert_xarray_to_uxarray(plot_data, grid, variable=variable))

  383
  384      # This is when there are more than three files listed in one input set, so differences don't make sense.

  485  def plot_spatial_data_elm(inputs):

  488      The data for these spatial plots are stored in NetCDF files specified by the inputs dictionary.
  489
  490      Parameters:
  491          input: Dictionary containing user inputs for different plotting options, where the keys are options and values are choices for those options.
  492                 This dictionary is assumed to be complete (pre-processed).
  493
  494      Returns:
  495          N/A.

  497      # This function creates spatial plots for a single variable, and so it assumes that there is only one variable in the inputs dictionary.
  498      start_time = time.time()
  499      variable = inputs['variable']
  500
  501      # Extract all other plotting options.
  502      cbar_label_size = inputs['cbar_label_size']

  534      # or we could have ensemble plots, in which case there are at most two data sets, but potentially multiple files in each of the sets.
  535      # The preprocessing done earlier has formatted the files into a list of lists so that they are in the form:
  536      #   For individual plots: [[file1, file2, file3, ...]]
  537      #   For ensemble plots: [["Control", "Full feedback"], ["Control_2", "Full feedback_2"], ...]

  539      num_file_sets = len(netcdf_files[0])

  544      #    are one or two ensembles.
  545      # If there is one ensemble (of any length, including 1 file):

  548      #    only mean or sum can be calculated across the files and plotted (no comparison)
  549      # If there are two ensembles (or more, but the code exits if there are more than two):
  550      #    num_file_sets = 2 (the number of ensembles)

  768  def plot_spatial_data_from_netcdf_files(inputs):
  769      """
  770      Parses a dictionary of inputs (keys are options, values are choices for those options) to create spatial plots for a single variable from either
  771      E3SM atmosphere (EAM) or land (ELM) outputs. The data for these spatial plots are stored in NetCDF files specified by the inputs dictionary.
  772
  773      Parameters:
  774          input: Dictionary containing the user plotting choice inputs for different options. This dictionary is assumed to be complete (pre-processed).
  775
  776      Returns:
  777          N/A.
  778      """
  779      # EAM requires a separate grid file since the simulations are run on an unstructured grid.
  780      grid_file = inputs['grid_file']
  781      if grid_file:
  782          plot_spatial_data_eam(inputs, grid_file)
  783      else:
  784          plot_spatial_data_elm(inputs)
  785
  786  ###---------------Begin execution---------------###
  787  if __name__ == '__main__':
  788
  789      # Run this script together with the input JSON file(s) on the command line.
  790      start_time = time.time()
  791      if len(sys.argv) < 2:
  792          print('Usage: python e3sm_plot_spatial_data.py `path/to/json/input/file(s)\'')
  793          sys.exit()
  794
  795      # Read and load the JSON file(s) into a list of dictionaries.
  796      inputs = []
  797      for index in range(1, len(sys.argv)):
  798          input_file = sys.argv[index]
  799          with open(input_file) as f:
  800              inputs.extend(json.load(f))
  801
  802      # Process each dictionary to produce a list of smaller dictionaries, where each smaller dictionary specifies options for a single plot.
  803      start_time = time.time()
  804      list_of_inputs = []
  805      for index in range(len(inputs)):
  806          # Process the inputs to fill in missing plotting input choices with default values, etc., and add to the list of dictionaries.
  807          list_of_inputs.extend(process_inputs(inputs[index]))
  808
  809      # Delete all the p-value files before we do any calculations to start a fresh run.
  810      for input_options in list_of_inputs:
  811          file = input_options['p_value_file']
  812          if os.path.exists(file):
  813              os.remove(file)
  814
  815      # Use any max_processes value requested in the inputs, otherwise use the default.
