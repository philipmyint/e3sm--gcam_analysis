   62  def _safe_plot_name_part(value):
   63      """Return a filesystem-friendly name fragment for a year or category."""
   64      if isinstance(value, (list, tuple)):
   65          value = '_'.join(str(item) for item in value)

   70  def _add_suffix_to_plot_name(plot_name, *suffix_parts):
   71      """Insert non-empty suffix parts before the plot file extension."""
   72      clean_parts = [_safe_plot_name_part(part) for part in suffix_parts if part is not None]
   73      clean_parts = [part for part in clean_parts if part]
   74      if not clean_parts:
   75          return plot_name

   77      stem, extension = os.path.splitext(filename)
   78      updated_name = f"{stem}_{'_'.join(clean_parts)}{extension}"
   79      return os.path.join(directory, updated_name) if directory else updated_name

   89  def _format_units_for_label(units, use_latex=False):
   90      """Return a readable units label, or None if the units value is empty/noisy."""
   91      if units is None:
   92          return None
   93      units_str = str(units).strip()
   94      if units_str.upper() in ('', 'NA', 'NONE', 'NAN', 'NULL') or units_str in ('(', ')'):
   95          return None
   96      units_str = units_str.replace('thous km2', 'thousands km2')
   97      if use_latex:
   98          units_str = units_str.replace('^2', r'$^2$')
   99          units_str = units_str.replace('km2', r'km$^2$')
  100          units_str = units_str.replace('m2', r'm$^2$')
  101      return units_str

  104  def _aggregate(df_rows, value_label, method):
  105      """Collapse all rows for one map polygon/scenario into one plotted value."""
  106      if method == 'sum':
  107          return df_rows[value_label].sum()
  108      if method == 'area_weighted_mean':
  109          total_area = df_rows['area'].sum()
  110          return (df_rows[value_label] * df_rows['area']).sum() / total_area if total_area != 0 else 0.0
  111      return df_rows[value_label].mean()

  880      # Do not start more worker processes than there are independent plotting jobs.
  881      max_processes = get_max_processes_from_inputs(inputs, MAX_PROCESSES)
  882      num_processes = min(max_processes, max(1, len(list_of_inputs)))
  883
  884      # Create all of the spatial plots in parallel.
  885      with multiprocessing.Pool(processes=num_processes) as pool:
  886          pool.map(plot_spatial_data, list_of_inputs)
  887
  888      # Sort all the p-value files alphabetically.
  889      for input_options in list_of_inputs:
  890          file = input_options['p_value_file']
  891          if os.path.exists(file):
  892              sort_file(file)
  893
  894      # Print the total execution time to produce all the plots.
  895      end_time = time.time()
  896      elapsed_time = end_time - start_time
  897      print(f"Elapsed time for producing all plots: {elapsed_time:.2f} seconds")
