   60  def _aggregate(df_rows, value_label, method):
   61      """Collapse all rows for one map polygon/scenario into one plotted value."""
   62      if method == 'sum':
   63          return df_rows[value_label].sum()
   64      if method == 'area_weighted_mean':
   65          total_area = df_rows['area'].sum()
   66          return (df_rows[value_label] * df_rows['area']).sum() / total_area if total_area != 0 else 0.0
   67      return df_rows[value_label].mean()
