 65def _safe_plot_name_part(value):

 67    if isinstance(value, (list, tuple)):

 73def _add_suffix_to_plot_name(plot_name, *suffix_parts):
 74    """Insert non-empty suffix parts before the plot file extension."""
 75    clean_parts = [_safe_plot_name_part(part) for part in suffix_parts if part is not None]
 76    clean_parts = [part for part in clean_parts if part]
 77    if not clean_parts:
 78        return plot_name
 79    directory, filename = os.path.split(plot_name)
 80    stem, extension = os.path.splitext(filename)
 81    updated_name = f"{stem}_{'_'.join(clean_parts)}{extension}"
 82    return os.path.join(directory, updated_name) if directory else updated_name

133def _format_units_for_label(units, use_latex=False):
134    """Return a readable units label, or None if the units value is empty/noisy."""
135    if units is None:
136        return None
137    units_str = str(units).strip()
138    if units_str.upper() in ('', 'NA', 'NONE', 'NAN', 'NULL') or units_str in ('(', ')'):
139        return None
140    units_str = units_str.replace('thous km2', 'thousands km2')
141    if use_latex:
142        units_str = units_str.replace('^2', r'$^2$')
143        units_str = units_str.replace('km2', r'km$^2$')
144        units_str = units_str.replace('m2', r'm$^2$')
145    return units_str

148def _aggregate(df_rows, value_label, method):
149    """Collapse all rows for one map polygon/scenario into one plotted value."""
150    if method == 'sum':
151        return df_rows[value_label].sum()
152    if method == 'area_weighted_mean':
153        total_area = df_rows['area'].sum()
154        return (df_rows[value_label] * df_rows['area']).sum() / total_area if total_area != 0 else 0.0
155    return df_rows[value_label].mean()

171def process_inputs(inputs):
172    """ 
173    Processes a dictionary of inputs (keys are options, values are choices for those options) for creating spatial plots.
174
175    Parameters:
176        inputs: Dictionary containing the user plotting choice inputs for different options. This dictionary may be incomplete or have invalid values.
177
178    Returns:
179        Dictionary that completely specifies all plotting options. 
180        If the user did not select a plotting option fora particular category, the default choice for that plotting option will be selected.
181    """
182    # Check that the output data file and shape file exist before trying to read them.
183    # Missing data files are skipped so one absent optional plot does not stop a long plotting run.
184    if not os.path.exists(inputs['output_file']):
185        print(f"Warning: output file not found, skipping: '{inputs['output_file']}'")
186        return None
187    if os.path.getsize(inputs['output_file']) == 0:
188        print(f"Warning: output file is empty, skipping: '{inputs['output_file']}'")
189        return None
190    if not os.path.exists(inputs['shape_file']):
191        raise FileNotFoundError(f"Error: shape file not found: '{inputs['shape_file']}'")
192    df = read_file_into_dataframe(inputs['output_file'])
193
194    # If the category label (e.g., sector or landtype) has not been specified, use the default value.
195    if 'category_label' not in inputs:
196        category_label = default_inputs['category_label']
197        inputs['category_label'] = category_label
198    else:
199        category_label = inputs['category_label']
200
201    # If categories were not specified, populate them from the data. For a multi-column
202    # category_label, each category is a tuple of values from those columns.
203    if 'categories' not in inputs:
204        if isinstance(category_label, list) and all(c in df.columns for c in category_label):
205            combos = df[category_label].drop_duplicates()
206            inputs['categories'] = list(combos.itertuples(index=False, name=None))
207            if 'categories_to_exclude' in inputs:
208                exclude = {tuple(e) if isinstance(e, list) else e for e in inputs['categories_to_exclude']}
209                inputs['categories'] = [c for c in inputs['categories'] if c not in exclude]
210        elif not isinstance(category_label, list) and category_label in df.columns:
211            if 'categories_to_exclude' in inputs:
212                inputs['categories'] = [column for column in df[category_label].unique() if column not in inputs['categories_to_exclude']]
213            else:
214                # No categories were requested or excluded, so aggregate over the whole file.
215                inputs['categories'] = 'All'
216        else:
217            inputs['categories'] = 'All'
218    categories = list(inputs['categories']) if isinstance(inputs['categories'], list) else inputs['categories']
219    # A multi-column category can be supplied as a dict, for example:
220    # {"sector": ["crop"], "technology": ["Corn", "Wheat"]}. Expand that into
221    # explicit tuples so filtering works the same way everywhere below.
222    if isinstance(categories, dict) and isinstance(category_label, list):
223        per_column = [categories[col] for col in category_label]
224        categories = list(itertools.product(*per_column))
225        inputs['categories'] = categories
226    elif isinstance(categories, str):
227        categories = [categories]
228        inputs['categories'] = categories
229    elif isinstance(category_label, list):
230        # Convert any list-type categories (from JSON) to tuples.
231        categories = [tuple(c) if isinstance(c, list) else c for c in categories]
232        inputs['categories'] = categories
233
234    # plot_categories controls panel layout. When it is a dict for a multi-column
235    # category_label, expand it into one panel per requested value combination.
236    plot_categories = inputs.get('plot_categories')
237    if isinstance(plot_categories, dict) and isinstance(category_label, list):
238        per_column = [plot_categories[col] for col in category_label]
239        plot_categories = list(itertools.product(*per_column))
240        inputs['plot_categories'] = plot_categories
241
242    # Panel categories must also be present in categories, because categories drives
243    # the earlier data reduction. This matters when a panel requests an aggregate
244    # landtype group such as "crop" rather than a literal CSV value.
245    if isinstance(plot_categories, list):
246        for category in plot_categories:
247            normalized_category = tuple(category) if isinstance(category_label, list) and isinstance(category, list) else category
248            if normalized_category not in categories:
249                categories.append(normalized_category)
250        inputs['categories'] = categories
251
252    # If units were not specified explicitly, fall back to the 'units' column in the
253    # data file. Filter to selected categories first, since different categories in
254    # the same file can have different units.
255    if isinstance(category_label, list):
256        category_label_in_data = all(c in df.columns for c in category_label)
257    else:
258        category_label_in_data = category_label in df.columns
259    if 'units' not in inputs and 'units' in df.columns:
260        relevant_categories = plot_categories if plot_categories is not None else (categories if categories != 'All' else None)
261        df_units_source = pd.DataFrame()
262        if relevant_categories and category_label_in_data:
263            if isinstance(category_label, list):
264                mask = df[category_label].apply(tuple, axis=1).isin(set(relevant_categories))
265            else:
266                mask = df[category_label].isin(relevant_categories)
267            df_units_source = df[mask]
268        # Fall back to the whole file if none of the relevant categories matched (e.g., they are landtype group
269        # names like 'crop' that are aggregated from raw values rather than literal column values).
270        if df_units_source.empty:
271            df_units_source = df
272        data_units = [_format_units_for_label(unit) for unit in df_units_source['units'].dropna().unique()]
273        data_units = [unit for unit in data_units if unit]
274        if len(data_units) == 1:
275            inputs['units'] = data_units[0]
276
277    # Create the plot directory if it does not already exist. By default, put the
278    # p-value output file in the same directory as the plots.
279    if 'plot_directory' not in inputs:
280        inputs['plot_directory'] = default_inputs['plot_directory']
281    if not os.path.exists(inputs['plot_directory']):
282        os.makedirs(inputs['plot_directory'])
283    if 'p_value_file' not in inputs:
284        inputs['p_value_file'] = os.path.join(inputs['plot_directory'], default_inputs['p_value_file'])
285
286    # Use the output CSV name as the default title and as the base for generated plot names.
287    index_of_last_backslash = inputs['output_file'].rfind('/')
288    index_of_dot_csv = inputs['output_file'].find('.csv')
289    if index_of_last_backslash == -1:
290        output_file_name = inputs['output_file'][:index_of_dot_csv]
291    else:
292        output_file_name = inputs['output_file'][index_of_last_backslash+1:index_of_dot_csv]
293    if 'title' not in inputs:
294        inputs['title'] = output_file_name
295    if 'plot_name' not in inputs:
296        # Build a descriptive default name from the key plot parameters; strip
297        # "_processed" from the file base so plot filenames read like outputs,
298        # not intermediate processing artifacts.
299        file_base = output_file_name.replace('_processed', '').replace('processed_', '')
300        _plot_type_abbrevs = {'absolute_difference': 'abs_diff', 'percent_difference': 'pct_diff'}
301        _raw_plot_type = inputs.get('plot_type', default_inputs['plot_type'])
302        plot_type_str = _plot_type_abbrevs.get(_raw_plot_type, _raw_plot_type)
303
304        raw_scenarios = inputs.get('scenarios', None)
305        # Include scenario name only when there is no comparison (exactly one scenario).
306        if raw_scenarios is None or check_is_list_of_lists(raw_scenarios) or len(list(raw_scenarios)) != 1:
307            scen_part = None
308        else:
309            scen_part = str(list(raw_scenarios)[0]).replace(' ', '_')
310
311        plot_years_val = inputs.get('plot_years', default_inputs['plot_years'])
312        if plot_years_val is None:
313            time_part = 'all_years'
314        elif len(plot_years_val) == 1:
315            time_part = str(plot_years_val[0])
316        elif inputs.get('plot_categories') is not None:
317            time_part = None
318        else:
319            time_part = f'{plot_years_val[0]}-{plot_years_val[-1]}'
320
321        # For multi-year panel plots with a single selected category, include that
322        # category in the name. "All" is included only when it represents a real
323        # category dimension being aggregated over.
324        categories_val = inputs.get('categories')
325        if isinstance(category_label, list):
326            category_label_in_data = all(c in df.columns for c in category_label)
327        else:
328            category_label_in_data = category_label in df.columns
329        if (inputs.get('plot_categories') is None and isinstance(categories_val, list) and len(categories_val) == 1
330                and (categories_val[0] != 'All' or category_label_in_data)):
331            category_part = str(categories_val[0]).replace(' ', '_')
332        else:
333            category_part = None
334
335        name_parts = ['spatial', file_base]
336        if category_part:
337            name_parts.append(category_part)
338        name_parts.append(plot_type_str)
339        if scen_part:
340            name_parts.append(scen_part)
341        if time_part:
342            name_parts.append(time_part)
343        name = '_'.join(name_parts)
344        inputs['plot_name'] = os.path.join(inputs['plot_directory'], name)
345    elif 'plot_name' in inputs and '/' not in inputs['plot_name']:
346        # If the user specified only a file name (no path) for the plot name, put the plot in the plot directory.
347        inputs['plot_name'] = os.path.join(inputs['plot_directory'], inputs['plot_name'])
348
349    # Add keys for plotting options that have not been specified in the inputs dictionary and use default values for them.
350    for key in default_inputs.keys():
351        if key not in inputs:
352            inputs[key] = default_inputs[key]
353    
354    # If the scenarios are specified as a list of lists (for ensemble plots), check if they need to be transposed.
355    # Users can now specify scenarios in two formats:
356    #   Format A (organized by ensemble member - original format):
357    #       [["Control", "Full feedback"], ["Control_2", "Full feedback_2"], ...]
358    #   Format B (organized by scenario set - new user-friendly format):
359    #       [["Control", "Control_2", ...], ["Full feedback", "Full feedback_2", ...]]
360    # The plotting functions expect Format A internally, so Format B will be automatically transposed.
361    if check_is_list_of_lists(inputs['scenarios']):
362        scenario_sets = inputs.get('scenario_sets', None)
363        inputs['scenarios'], was_transposed = transpose_scenarios_if_needed(inputs['scenarios'], scenario_sets)
364        if was_transposed and inputs['notify_scenarios_transposed']:
365            print(f"Note: Scenarios were automatically transposed from 'organized by scenario set' format to 'organized by ensemble member' format.")
366
367    # If the scenarios have not been specified, use all the scenarios in the Pandas DataFrame.
368    if 'scenarios' not in inputs:
369        scenario_label = inputs['scenario_label']
370        inputs['scenarios'] = df[scenario_label].unique()
371    
372    # Check that the value_label column exists in the DataFrame.
373    value_label = inputs['value_label']
374    if value_label not in df.columns:
375        raise KeyError(f"Error: value_label '{value_label}' not found in '{inputs['output_file']}'. "
376                       f"Available columns: {list(df.columns)}")
377
378    # area_weighted_mean requires area metadata from gcam_process_extracted_data.py.
379    if inputs['aggregation_function'] == 'area_weighted_mean' and 'area' not in df.columns:
380        raise KeyError(f"Error: aggregation_function is 'area_weighted_mean' "
381                       f"but no 'area' column found in '{inputs['output_file']}'. "
382                       f"Consider using 'mean' or 'sum' instead, or add area data "
383                       f"with gcam_process_extracted_data.py and a land_allocation_file entry.")
384
385    # Warn if any specified scenario is not found in the DataFrame.
386    scenario_label = inputs['scenario_label']
387    if scenario_label in df.columns:
388        available_scenarios = set(df[scenario_label].unique())
389        specified_scenarios = inputs['scenarios']
390        if check_is_list_of_lists(specified_scenarios):
391            flat_scenarios = [s for member in specified_scenarios for s in member]
392        else:
393            flat_scenarios = list(specified_scenarios)
394        missing_scenarios = [s for s in flat_scenarios if s not in available_scenarios]
395        if missing_scenarios:
396            print(f"Warning: the following scenarios were not found in '{inputs['output_file']}' "
397                  f"and will produce empty/zero spatial plots: {missing_scenarios}")
398
399    # Verify the plot directory is writable before doing the geospatial work.
400    plot_name = inputs['plot_name']
401    plot_dir = os.path.dirname(plot_name) or '.'
402    if not os.access(plot_dir, os.W_OK):
403        raise OSError(f"Error: plot directory is not writable: '{plot_dir}'")
404
405    return inputs
