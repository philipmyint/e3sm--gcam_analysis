   15  def clean_up_dataframe(df, print_to_console=False):
   16      """
   17      Cleans up a Pandas DataFrame from junk columns that contain only NaN values and returns this cleaned-up DataFrame.
   18      The junk columns may be a result of pd.read_fwf() separating out a single column with a long label into two columns where the second column
   19      contains the junk values. For example, the column "rho A+B+C (g/cm^3)" may get separated to "rho A+B+C" and "(g/cm^3)", in which the latter
   20      column, which will contain only NaN entries, should be deleted.
   21
   22      Parameters:
   23          df: DataFrame to be cleaned.
   24          print_to_console: Indicates if the contents of the data should be printed to the console to monitor the progress.
   25
   26      Returns:
   27          df: DataFrame after cleaning operations have been applied.
   28      """
   29      if print_to_console:
   30          print(f'\nThe original columns of the DataFrame are:\n{df.columns}\n')
   31
   32      # Number of columns.
   33      num_columns = len(df.columns)
   34
   35      # List containing the labels of junk columns.
   36      columns_to_delete = []
   37
   38      # List containing the labels of non-junk columns.
   39      columns_to_keep = []
   40
   41      # Loop over the columns in the DataFrame.
   42      for i in np.arange(num_columns):
   43
   44          # Extract the label of the current column.
   45          label = df.columns[i]
   46
   47          # If this is a junk column, record its label and go to the next column.
   48          if df[label].isnull().values.all():
   49              columns_to_delete.append(label)
   50              if print_to_console:
   51                  print(f'Column {i} contains junk; its label is {label}\n')
   52              continue
   53
   54          # If the next column is a junk column and we are not in the last column, append the label of that
   55          # next column (with some modifications) to the label of the current one.
   56          if i < num_columns - 1 and df[df.columns[i+1]].isnull().values.all():
   57              # If the label of the junk column indicates the units of the quantity in the previous column
   58              # placed between parentheses, extract only the parentheses and the unit label between them.
   59              start = df.columns[i+1].find("(")
   60              if start != -1:
   61                  end = df.columns[i+1].find(")") + 1
   62                  label += " " + df.columns[i+1][start:end]
   63              # If the label of the junk column does not contain parentheses, then append that label to that of the current column.
   64              else:
   65                  label += " " + df.columns[i+1]
   66              if print_to_console:
   67                  print(f'The new label of column {i} is {label}')
   68
   69          # Record the labels of all non-junk columns in the appropriate list.
   70          columns_to_keep.append(label)
   71
   72      # Delete all junk columns.
   73      for label in columns_to_delete:
   74          df.drop(label, axis=1, inplace=True)
   75
   76      # Reassign the column labels to those that we want to keep; print these labels and return the DataFrame.
   77      df.columns = columns_to_keep
   78      if print_to_console:
   79          print(f'The new columns of the modified DataFrame are:\n{df.columns}\n')
   80      return df

   82  def get_columns_without_units_in_dataframe(df):
   83      """
   84      Returns all column names in a Pandas DataFrame after stripping out the units. For example, return 'Mass' if the column name is 'Mass (kg)'.
   85
   86      Parameters:
   87          df: The input DataFrame.
   88
   89      Returns:
   90          A list of all column names, with the units removed.
   91      """
   92      columns_without_units = []
   93      for column in df.columns:
   94          stop_index = column.find(' (')
   95          # If the column does not have unit associated with it (so that find returns a -1), just return that column unmodified.
   96          if stop_index == -1:
   97              columns_without_units.append(column)
   98          else:
   99              columns_without_units.append(column[:stop_index])
  100      return columns_without_units

  102  def get_matching_column_in_dataframe(df, variable, get_all_matches=False):
  103      """
  104      Returns the column name(s) in a Pandas DataFrame that matches the variable.
  105
  106      Parameters:
  107          df: The input DataFrame.
  108          variable: Variable name to search for in the column names.
  109          get_all_matches: If True, all columns that matches the variable will be returned. If False, only the first matching column will be returned.
  110
  111      Returns:
  112          The matching column name(s), or None if no match is found.
  113      """
  114      all_matching_columns = []
  115      for column in df.columns:
  116          units_in_column = ' (' in column
  117          is_match = (units_in_column and column.startswith(variable + ' (')) or \
  118                      (not units_in_column and variable in column)
  119          if is_match:
  120              if not get_all_matches:
  121                  return column
  122              all_matching_columns.append(column)
  123      return all_matching_columns or None

  125  def move_columns_next_to_each_other_in_dataframe(df, column1, column2):
  126      """
  127      Rearranges a Pandas DataFrame by moving column2 next to column1.
  128
  129      Parameters:
  130          column1: Column next to which column2 will be moved. The order of all columns prior to column1 will be unaffected.
  131          column2: Column to move next to column1.
  132          df: DataFrame whose columns will be rearranged.
  133
  134      Returns:
  135          A new DataFrame with column1 and column2 moved next to each other.
  136      """
  137      index = df.columns.get_loc(column1)
  138      new_columns = list(df.columns[:index]) + [column1, column2] + list(df.columns[index+1:])
  139      df = df[new_columns]
  140      # Remove the duplicate column2, keeping the first occurrence.
  141      df = df.loc[:, ~df.columns.duplicated()]
  142      return df

  144  def perform_ttest(df, columns_set_1, columns_set_2, equal_var=False):
  145      """
  146      Performs a t-test for the means of two (presumed) independent data sets.
  147
  148      Parameters:
  149          df: Pandas DataFrame containing the columns for both data sets.
  150          columns_set_1: List of columns in the DataFrame for the first data set.
  151          columns_set_2: List of columns in the DataFrame for the second data set.
  152          equal_var: Boolean that indicates if the two samples are characterized by the same variance.
  153
  154      Returns:
  155          The p-value produced by the t-test.
  156      """
  157      set_1 = df[columns_set_1]
  158      set_2 = df[columns_set_2]
  159      ttest = stats.ttest_ind(set_1, set_2, equal_var=equal_var)
  160      return ttest.pvalue

  247  def write_dataframe_to_fwf(file_name, df, keep_index_column=False):
  248      """
  249      Writes the contents of a Pandas DataFrame to an output file in fixed-width format (fwf), omitting the index column if specified to do so.
  250
  251      Parameters:
  252          file_name: Complete path and name of the output file.
  253          df: DataFrame in which the data to be written to the output file are stored.
  254          keep_index_column: Specifies whether to print the index column in the output file.
  255
  256      Returns:
  257          N/A.
  258      """
  259      # Calculate column widths dictionary to ensure minimum widths.
  260      col_space = {}
  261      for col in df.columns:
  262          if 'Year' in str(col):
  263              # Ensure Year column is wide enough for both header and data.
  264              if len(df) > 0:
  265                  max_data_width = max(len(str(val)) for val in df[col])
  266                  col_space[col] = max(4, len(str(col)), max_data_width)
  267              else:
  268                  col_space[col] = max(4, len(str(col)))
  269
  270      # Pass col_space to the to_string() function.
  271      df_str = df.to_string(index=keep_index_column, col_space=col_space if col_space else None)
  272
  273      # Write the contents to the file.
  274      with open(file_name, 'w', encoding='utf-8') as file:
  275          file.write(df_str)
