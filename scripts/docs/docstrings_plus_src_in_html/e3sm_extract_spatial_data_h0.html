   12  def get_spatial_dims_and_coords_for_h0(ds_grid, netcdf_file):
   13      """Return the spatial dims/coords used for static area and land-fraction arrays."""
   14      if 'elm.h0' in netcdf_file or 'eam.h0' in netcdf_file:
   15          area_var = 'area'
   16      else:
   17          area_var = 'AREA'
   18      area_da = ds_grid[area_var].squeeze(drop=True)
   19      spatial_dims = area_da.dims
   20      spatial_coords = {dim: area_da.coords[dim] for dim in spatial_dims if dim in area_da.coords}
   21      spatial_shape = area_da.shape
   22      return spatial_dims, spatial_coords, spatial_shape

   24  def process_inputs(inputs):
   25      """
   26      Processes a dictionary of inputs (keys are options, values are choices for those options) for extracting spatial data from E3SM-generated NetCDF
   27      files and printing that data to smaller, focused NetCDF files that contain spatial data only for the variables specified in the dictionary.
   28
   29      Parameters:
   30          inputs: Dictionary containing the user choice inputs for different options. This dictionary may be incomplete or have invalid values.
   31
   32      Returns:
   33          List of dictionaries, where each dictionary has been processed so that it is complete in all options for a single focused NetCDF file.
   34      """
   35      # If the user specified output_files to be a string (indicating a single output file), turn that string into a list containing that string.
   36      if isinstance(inputs['output_files'], (str, int, float)):
   37          inputs['output_files'] = [inputs['output_files']]
   38
   39      # These options can be specified once and applied to every output file, or specified as one entry per output file.
   40      input_types_to_modify = ['netcdf_substrings', 'variables', 'start_years', 'end_years']
   41
   42      # Inputs given per output file must have either one entry (applied to all) or one entry per output file.
   43      num_output_files = len(inputs['output_files'])
   44      mismatched = {}
   45      for input_type in input_types_to_modify:
   46          value = inputs[input_type]
   47          if isinstance(value, (str, int, float)):
   48              continue
   49          # A flat list of substrings/variables applies to all output files, so only a list of lists is checked.
   50          if input_type in ['netcdf_substrings', 'variables'] and not (value and isinstance(value[0], list)):
   51              continue
   52          if len(value) not in (1, num_output_files):
   53              mismatched[input_type] = len(value)
   54      if mismatched:
   55          print(f"Error: output_files has {num_output_files} entries, but the following inputs have a different number of entries "
   56                f"(each must have 1 or {num_output_files}):")
   57          for input_type, length in mismatched.items():
   58              print(f"  {input_type}: {length}")
   59          sys.exit(1)
   60
   61      for input_type in input_types_to_modify:
   62          if isinstance(inputs[input_type], (str, int, float)):
   63              inputs[input_type] = [inputs[input_type]]
   64          # A flat list of substrings/variables applies to every output file, so wrap it before copying it per output file.
   65          if input_type in ['netcdf_substrings', 'variables'] and not isinstance(inputs[input_type][0], list):
   66              inputs[input_type] = [inputs[input_type]]
   67          if len(inputs[input_type]) == 1:
   68              inputs[input_type] = inputs[input_type]*num_output_files
   69
   70      # Now that the dictionary has been populated with complete data extraction options for each output file, separate it into a list of dictionaries,
   71      # where each of these smaller dictionaries contain the data extraction options for a single output file. Return this list of dictionaries.
   72      list_of_inputs = []
   73      for file_index in range(len(inputs['output_files'])):
   74          output_file = inputs['output_files'][file_index]
   75          netcdf_substrings = inputs['netcdf_substrings'][file_index]
   76          variables = inputs['variables'][file_index]
   77          start_years = inputs['start_years'][file_index]
   78          end_years = inputs['end_years'][file_index]
   79          inputs_for_this_output_file = {'simulation_path': inputs['simulation_path'], 'output_files': output_file, 'netcdf_substrings': netcdf_substrings}
   80          inputs_for_this_output_file.update({'variables': variables, 'start_years': start_years, 'end_years': end_years})
   81          inputs_for_this_output_file['extra_dim_aggregation'] = inputs.get('extra_dim_aggregation', 'first')
   82          inputs_for_this_output_file['max_processes'] = inputs.get('max_processes', MAX_PROCESSES)
   83          list_of_inputs.append(inputs_for_this_output_file)
   84      return list_of_inputs

   86  def process_dataset(ds):
   87      """
   88      Processes an xarray Dataset to add a total precipitation variable (including change units from m/s to mm/year) and
   89      add a variable for the atmospheric mole fraction of CO2 in units of ppm.
   90
   91      Parameters:
   92          ds: Dataset to process.
   93
   94      Returns:
   95          The processed Dataset.
   96      """
   97      variables = list(ds.keys())
   98
   99      # Add a new variable for the total precipitation, and update the variable labels.
  100      precip_variables = ['PRECC', 'PRECL', 'PRECSC', 'PRECSL']
  101      if check_substrings_in_list(precip_variables, variables, all_or_any='all'):
  102          ds[precip_variables] *= years_TO_s*m_TO_mm
  103          for precip_variable in precip_variables:
  104              ds[precip_variable].attrs['units'] = 'mm/year'
  105          ds['PRECIP'] = ds[precip_variables].to_array(dim='variable').sum(dim='variable')
  106          ds['PRECIP'].attrs = {'units':'mm/year', 'description':'Total precipitation rate'}
  107
  108      # Calculate the atmospheric mole fraction of CO2 in units of ppm, which is defined as a mole fraction in dry air (with humidity subtracted out).
  109      pressure_variables = ['PBOT', 'PCO2', 'QBOT']
  110      if check_substrings_in_list(pressure_variables, variables, all_or_any='all'):
  111          # See derivation of H2O partial pressure formula at https://cran.r-project.org/web/packages/humidity/vignettes/humidity-measures.html.
  112          partial_pressure_H2O = ds['PBOT']*ds['QBOT']/(0.622 + (0.378*ds['QBOT']))
  113          ds['ZCO2'] = mole_fraction_TO_ppm*ds['PCO2']/(ds['PBOT'] - partial_pressure_H2O)
  114          ds['ZCO2'].attrs = {'units':'ppm', 'description':'CO2 mole fraction in dry air'}
  115
  116      # select the surface level for EAM CO2 concentration variables and calculate ppm concentration from the dry mixing ratio (kg/kg)
  117      for variable in variables:
  118          if 'lev' in ds[variable].dims and variable.startswith('CO2'):
  119              ds[variable] = ds[variable].isel(lev=-1)
  120              ds[variable] *= mole_fraction_TO_ppm*MM_ATM/MM_CO2
  121              ds[variable].attrs['units'] = 'ppm'
  122
  123      # Convert remaining /s flux variables to /month, consistent with time series extraction.
  124      avg_seconds_per_month = years_TO_s / years_TO_months
  125      for variable in list(ds.data_vars):
  126          units = ds[variable].attrs.get('units', '')
  127          if '/s' in units:
  128              ds[variable] = ds[variable] * avg_seconds_per_month
  129              ds[variable].attrs['units'] = units.replace('/s', '/month')
  130
  131      return ds

  133  def extract_spatial_data_from_netcdf_files(inputs):
  134      """
  135      Extracts time series data from E3SM-generated h0 NetCDF files of a particular type that are located in a simulation directory and puts this data
  136      into a smaller, focused NetCDF file that contains annual-mean spatial data only for the variables specified by the user in the inputs dictionary.
  137
  138      Parameters:
  139          input: Dictionary containing the user data-extraction inputs for different options. This dictionary is assumed to be complete (pre-processed).
  140
  141      Returns:
  142          N/A.
  143      """
  144      # Record the start time and extract all the user-chosen options from the inputs dictionary.
  145      start_time = time.time()
  146      simulation_path = inputs['simulation_path']
  147      output_file = inputs['output_files']
  148      netcdf_substrings = inputs['netcdf_substrings']
  149      variables = inputs['variables']
  150      start_year = inputs['start_years']
  151      end_year = inputs['end_years']
  152
  153      # Check that the simulation path exists before scanning for files.
  154      if not os.path.exists(simulation_path):
  155          print(f"Error: simulation path not found: '{simulation_path}'")
  156          return
  157
  158      # Check that start_year is not greater than end_year.
  159      if start_year > end_year:
  160          print(f"Error: start_year ({start_year}) is greater than end_year ({end_year}) for output file '{output_file}'.")
  161          return
  162
  163      # Check that the output directory exists and the output file is writable before doing any expensive processing.
  164      output_dir = os.path.dirname(os.path.abspath(output_file))
  165      if not os.path.exists(output_dir):
  166          print(f"Output directory does not exist: '{output_dir}'. Creating it now...")
  167          os.makedirs(output_dir)
  168      try:
  169          open(output_file, 'a').close()
  170      except OSError as e:
  171          print(f"Error: cannot write to output file '{output_file}': {e}")
  172          return
  173
  174      # Get all NetCDF files for this particular type that fall within the specified start and end years.
  175      netcdf_files = get_all_files_in_path(simulation_path, file_name_substrings=netcdf_substrings, file_extension='.nc')
  176      netcdf_files = get_netcdf_files_between_start_and_end_years(netcdf_files, start_year, end_year)
  177
  178      # Check that at least one matching file was found.
  179      if len(netcdf_files) == 0:
  180          print(f"Error: no NetCDF files found in '{simulation_path}' matching substrings {netcdf_substrings} "
  181                f"between years {start_year} and {end_year}.")
  182          return
  183      print(f"Found {len(netcdf_files)} file(s) for '{output_file}'. Checking variables...")
  184
  185      # Check that all requested variables exist in the first file before opening the full dataset.
  186      with xr.open_dataset(netcdf_files[0], decode_times=False) as ds_check:
  187          missing_vars = [v for v in variables if v not in ds_check.data_vars]
  188      if missing_vars:
  189          print(f"Warning: the following variables were not found in '{netcdf_files[0]}' and will be skipped: {missing_vars}")
  190          variables = [v for v in variables if v not in missing_vars]
  191          if not variables:
  192              print(f"Warning: no valid variables remain. Skipping '{output_file}'.")
  193              return
  194      print(f"Extracting {len(variables)} variable(s): {variables}")
  195
  196      # Collect the NetCDF files (one for each month between the start and end years) in an xarray Dataset and store only the specified variables.
  197      print(f"Opening {len(netcdf_files)} file(s) with xarray...")
  198      ds = xr.open_mfdataset(netcdf_files, decode_times=True, combine='nested', concat_dim='time', data_vars='minimal', parallel=True, compat='no_conflicts')[variables]
  199
  200      # For variables with extra dimensions beyond time/lat/lon/ncol (e.g. levgrnd for TSOI), select index 0 or average before computing annual means.
  201      # CO2 lev selection is handled later in process_dataset.
  202      _spatial_dims = {'time', 'lat', 'lon', 'ncol'}
  203      extra_dim_aggregation = inputs.get('extra_dim_aggregation', 'first')
  204      for variable in variables:
  205          if 'lev' in ds[variable].dims and variable.startswith('CO2'):
  206              pass
  207          else:
  208              for d in [dim for dim in ds[variable].dims if dim not in _spatial_dims]:
  209                  if extra_dim_aggregation == 'mean':
  210                      ds[variable] = ds[variable].mean(dim=d)
  211                  else:
  212                      ds[variable] = ds[variable].isel({d: 0})
  213
  214      # Shift output back by one month to get rid of the extra month (January in the next year after end_year) that somehow gets added.
  215      ds['time'] = xr.CFTimeIndex(ds.get_index('time').shift(-1, 'ME'))
  216
  217      # Take a month-length-weighted mean in each year so annual means account for different month lengths.
  218      month_weights = ds['time'].dt.days_in_month.groupby('time.year') / ds['time'].dt.days_in_month.groupby('time.year').sum()
  219      ds = (ds * month_weights).groupby('time.year').sum()
  220
  221      # Add static grid-cell area and land fraction from one NetCDF file (these do not vary across time).
  222      areas_m2, ds_grid, landfrac, _ = find_gridcell_areas_in_netcdf_file(netcdf_files[0], variables_to_keep=variables)
  223      spatial_dims, spatial_coords, spatial_shape = get_spatial_dims_and_coords_for_h0(ds_grid, netcdf_files[0])
  224      ds_grid.close()
  225
  226      area_values = areas_m2.reshape(spatial_shape)
  227      landfrac_values = landfrac.reshape(spatial_shape)
  228
  229      ds['AREA_H0 (km^2)'] = xr.DataArray(area_values/km2_TO_m2, dims=spatial_dims, coords=spatial_coords)
  230      ds['AREA_H0 (km^2)'].attrs = {'units': 'km^2', 'description': 'Grid-cell area from h0 file'}
  231      ds['LANDFRAC_H0'] = xr.DataArray(landfrac_values, dims=spatial_dims, coords=spatial_coords)
  232      ds['LANDFRAC_H0'].attrs = {'units': '1', 'description': 'Land fraction from h0 file'}
  233
  234      # Add total precipitation (in units of mm/year) and CO2 concentration variables to the Dataset.
  235      ds = process_dataset(ds)
  236
  237      # Clear the unlimited_dims encoding entirely since the output is a static spatial file and does not need any unlimited dimension.
  238      ds.encoding['unlimited_dims'] = set()
  239
  240      # Write the Dataset to a NetCDF file.
  241      print(f"Writing output to '{output_file}'...")
  242      ds.to_netcdf(output_file, mode='w')
  243
  244      # Print the time needed to create the smaller NetCDF file.
  245      end_time = time.time()
  246      elapsed_time = end_time - start_time
  247      print(f"Elapsed time for {output_file}: {elapsed_time:.2f} seconds")
