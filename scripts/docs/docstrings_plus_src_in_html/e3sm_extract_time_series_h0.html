   14  def get_seconds_for_each_month(df, NUM_SECONDS_IN_MONTHS):
   15      """
   16      Get the number of seconds for each row based on the actual month in that row.
   17
   18      This function properly handles missing months by using the actual Month column
   19      rather than assuming data starts in January.
   20
   21      Parameters:
   22          df: DataFrame with a 'Month' column containing values 1-12
   23          NUM_SECONDS_IN_MONTHS: List/array of seconds for each month (12 values)
   24                                 [Jan_secs, Feb_secs, ..., Dec_secs]
   25
   26      Returns:
   27          Array of shape (len(df), 1) with correct seconds for each row's month
   28
   29      Example:
   30          If df has months [2, 3, 4] (Feb, Mar, Apr), returns:
   31          [[Feb_secs],
   32           [Mar_secs],
   33           [Apr_secs]]
   34      """
   35      # Create mapping: Month 1 -> NUM_SECONDS_IN_MONTHS[0], Month 2 -> NUM_SECONDS_IN_MONTHS[1], etc.
   36      month_to_seconds = dict(enumerate(NUM_SECONDS_IN_MONTHS, start=1))
   37
   38      # Map each row's month to its corresponding seconds value.
   39      seconds_array = df['Month'].map(month_to_seconds).values
   40
   41      # Reshape to column vector for broadcasting with DataFrame columns.
   42      return seconds_array.reshape(-1, 1)

   44  def process_dataframe(df):
   45      """
   46      Processes a Pandas DataFrame by adding new columns (e.g., total precipitation, mole fraction CO2) or changing the units (e.g., Pg instead of g).
   47
   48      Parameters:
   49          df: DataFrame to be processed.
   50
   51      Returns:
   52          The processed DataFrame.
   53      """
   54      # Convert precipitation variables from m/s to mm/month, add a new column for the total precipitation, and update the column labels.
   55      substrings = ['PRECC', 'PRECL', 'PRECSC', 'PRECSL']
   56      if check_substrings_in_list(substrings, df.columns, all_or_any='all'):
   57          columns_to_modify = [label for label in df.columns for substring in substrings if substring in label]
   58          # Use actual Month column to get correct seconds for each row.
   59          seconds_in_months_tiled = get_seconds_for_each_month(df, NUM_SECONDS_IN_MONTHS)
   60          df[columns_to_modify] *= seconds_in_months_tiled*m_TO_mm
   61          df['PRECIP (mm/month)'] = df[columns_to_modify].sum(axis=1)
   62          condition = lambda x: any(substring in x for substring in ['PRECC', 'PRECL', 'PRECSC', 'PRECSL'])
   63          new_column_label_function = lambda x: x.replace('(m/s)', '(mm/month)')
   64          df.columns = modify_list_based_on_condition(df.columns, condition, new_column_label_function)
   65
   66      # Calculate the atmospheric mole fraction of CO2 in units of ppm, which is defined as a mole fraction in dry air (with humidity subtracted out).
   67      substrings = ['PBOT', 'PCO2', 'QBOT']
   68      if check_substrings_in_list(substrings, df.columns, all_or_any='all'):
   69          columns_to_modify = [label for label in df.columns for substring in substrings if substring in label]
   70          # See derivation of H2O partial pressure formula at https://cran.r-project.org/web/packages/humidity/vignettes/humidity-measures.html.
   71          partial_pressure_H2O = df['PBOT (Pa)']*df['QBOT (kg/kg)']/(0.622 + (0.378*df['QBOT (kg/kg)']))
   72          df['ZCO2 (ppm)'] = mole_fraction_TO_ppm*df['PCO2 (Pa)']/(df['PBOT (Pa)'] - partial_pressure_H2O)
   73          df = move_columns_next_to_each_other_in_dataframe(df, 'PCO2 (Pa)', 'ZCO2 (ppm)')
   74
   75      # Convert fluxes and stocks that have units of g or kg to Pg.
   76      old_labels = ['(gC', '(g/', '(kg']
   77      new_labels = ['(PgC', '(Pg/', '(Pg']
   78      multipliers = [1/Pg_TO_g, 1/Pg_TO_g, 1/Pg_TO_kg]
   79      for index, old_label in enumerate(old_labels):
   80          # Make sure that the quantity is actually a flux or stock and not a mass fraction (i.e., does not have units of kg/kg).
   81          columns_to_modify = [label for label in df.columns if (old_label in label and '(kg/kg)' not in label)]
   82          condition = lambda x: old_label in x and '(kg/kg)' not in x
   83          new_column_label_function = lambda x: x.replace(old_label, new_labels[index])
   84          df[columns_to_modify] *= multipliers[index]
   85          df.columns = modify_list_based_on_condition(df.columns, condition, new_column_label_function)
   86
   87      # Convert fluxes from per second to per month.
   88      old_label = '/s)'
   89      columns_to_modify = [label for label in df.columns if old_label in label]
   90      # Use actual Month column to get correct seconds for each row.
   91      seconds_in_months_tiled = get_seconds_for_each_month(df, NUM_SECONDS_IN_MONTHS)
   92      df[columns_to_modify] *= seconds_in_months_tiled
   93      condition = lambda x: old_label in x
   94      new_column_label_function = lambda x: x.replace(old_label, '/month)')
   95      df.columns = modify_list_based_on_condition(df.columns, condition, new_column_label_function)
   96
   97      # Convert CO2 fluxes and stocks from PgCO2 to PgC.
   98      substrings = ['SFCO2', 'TMCO2']
   99      columns_to_modify = [label for label in df.columns if check_substrings_in_string(substrings, label, all_or_any='any')]
  100      df[columns_to_modify] *= MM_C/MM_CO2
  101      old_label = 'Pg'
  102      condition = lambda x: old_label in x and check_substrings_in_string(substrings, x, all_or_any='any')
  103      new_column_label_function = lambda x: x.replace(old_label, 'PgC')
  104      df.columns = modify_list_based_on_condition(df.columns, condition, new_column_label_function)
  105
  106      # Convert EAM CO2 concentration variables to ppm from dry mixing ratio (kg/kg).
  107      columns_to_modify = [label for label in df.columns if label.startswith('CO2') and '(kg/kg)' in label]
  108      df[columns_to_modify] *= mole_fraction_TO_ppm*MM_ATM/MM_CO2
  109      condition = lambda x: x.startswith('CO2') and '(kg/kg)' in x
  110      new_column_label_function = lambda x: x.replace('(kg/kg)', '(ppm)')
  111      df.columns = modify_list_based_on_condition(df.columns, condition, new_column_label_function)
  112
  113      return df

  115  def extract_netcdf_file_into_dataframe(file, variables, lat_lon_aggregation_type, region, extra_dim_aggregation='first'):
  116      """
  117      Extracts the specified variables from an E3SM-generated NetCDF h0 file into a Pandas DataFrame and performs the indicated lat/lon aggregation.
  118      Each NetCDF file contains simulation results for one month in a particular year from either EAM (atmosphere model) or ELM (land model).
  119      One choice of aggregation type is to perform an area-weighted mean over the latitude/longitude coordinates for the given month.
  120
  121      Parameters:
  122          file: Complete path and name of the NetCDF file.
  123          variables: List of variables that we want to extract from the NetCDF file.
  124          lat_lon_aggregation_type: String that indicates how we want to perform the aggregation over the lat/lon coordinates to form the time series.
  125          region: String for the region of interest. If not specified or not recognized, then there will be no restrictions on the lat/lon coordinates.
  126          extra_dim_aggregation: How to collapse variable dimensions beyond time and horizontal space. Use 'first' to select index 0 or 'mean' to
  127                                 average over each extra dimension. CO2 variables with a lev dimension are handled separately by selecting the
  128                                 surface level.
  129
  130      Returns:
  131          DataFrame containing one column for each of the variables, plus year and month columns.
  132      """
  133      # Extract the area as a function of lat/lon coordinate from the NetCDF file and forms an xarray Dataset for the variables.
  134      areas, ds, landfrac, non_landfrac = find_gridcell_areas_in_netcdf_file(file, region=region, variables_to_keep=variables)
  135
  136      # Convert the Dataset to create an overall DataFrame that stores all variables except those for land units and plant-functional types (PFTs).
  137      # If land units and/or PFTs are also of interest, create a second DataFrame to store them.
  138      variables_except_landunits_and_pfts = variables.copy()
  139      variables_landunits_and_pfts = []
  140      for variable in ['PCT_LANDUNIT', 'PCT_NAT_PFT']:
  141          if variable in variables:
  142              variables_except_landunits_and_pfts.remove(variable)
  143              variables_landunits_and_pfts.append(variable)
  144      # PCT_CFT gives crop functional type percentages on the crop land unit. Handle it separately from ordinary variables
  145      # and from PCT_NAT_PFT so xarray does not broadcast crop types against land-unit or natural-PFT dimensions.
  146      if 'PCT_CFT' in variables:
  147          variables_except_landunits_and_pfts.remove('PCT_CFT')
  148      if variables_except_landunits_and_pfts:
  149          # For CO2 concentration variables that have a lev dimension, select only the surface level (index -1, i.e. the lowest/highest-pressure level)
  150          # before converting to a DataFrame, so that all variables share the same (time, ncol) dimensions.
  151          # For any other variable with extra dimensions beyond time/lat/lon/ncol, collapse each extra dimension according to extra_dim_aggregation:
  152          # select index 0 by default (e.g., surface soil for levgrnd), or average across the dimension when extra_dim_aggregation is 'mean'.
  153          _spatial_dims = {'time', 'lat', 'lon', 'ncol'}
  154          for variable in variables_except_landunits_and_pfts:
  155              if 'lev' in ds[variable].dims and variable.startswith('CO2'):
  156                  ds[variable] = ds[variable].isel(lev=-1)
  157              else:
  158                  for d in [dim for dim in ds[variable].dims if dim not in _spatial_dims]:
  159                      if extra_dim_aggregation == 'mean':
  160                          ds[variable] = ds[variable].mean(dim=d)
  161                      else:
  162                          ds[variable] = ds[variable].isel({d: 0})
  163          df = ds[variables_except_landunits_and_pfts].to_dataframe()
  164          # Remove the time index since we do not use it.
  165          df = df.reset_index(level='time', drop=True)
  166          # Add units to the DataFrame column header.
  167          units = [ds[variable].attrs['units'] for variable in ds[variables_except_landunits_and_pfts].data_vars]
  168          df.columns = add_lists_elementwise(df.columns, units, list2_are_units=True)
  169      if variables_landunits_and_pfts:
  170          df_landunits_and_pfts = ds[variables_landunits_and_pfts].to_dataframe()
  171
  172      def _per_row_spatial_values(arr, df_ref):
  173          # Return one area/fraction value for each DataFrame row. For lat/lon-indexed data, align by coordinate;
  174          # otherwise assume ncol-style row ordering and repeat the flattened gridcell values.
  175          flat = arr.flatten()
  176          if 'lat' in df_ref.index.names and 'lon' in df_ref.index.names:
  177              la, lo = np.meshgrid(ds['lat'].values, ds['lon'].values, indexing='ij')
  178              lookup = pd.Series(flat, index=pd.MultiIndex.from_arrays(
  179                  [la.flatten(), lo.flatten()], names=['lat', 'lon']))
  180              extra = [n for n in df_ref.index.names if n not in ('lat', 'lon')]
  181              idx = df_ref.index.droplevel(extra) if extra else df_ref.index
  182              return lookup.reindex(idx).values
  183          n_reps = len(df_ref) // len(flat)
  184          return np.tile(flat, n_reps)
  185
  186      # 9 land units: vegetation, crop, ice, multiple ice, lake, wetland, urban tbd, urban hd, urban md.
  187      # Use vegetation (index 0) for natural PFT areas, and crop (index 1) for CFT areas when PCT_CFT is requested.
  188      if 'PCT_LANDUNIT' in variables:
  189          df_landunits_and_pfts = df_landunits_and_pfts.reset_index(level='ltype')
  190          # The crop land unit (index 1) provides the crop-land fraction needed to turn PCT_CFT percentages into areas.
  191          if 'PCT_CFT' in variables:
  192              frac_crop = df_landunits_and_pfts[df_landunits_and_pfts['ltype'] == 1]['PCT_LANDUNIT'].groupby(['lat', 'lon']).mean().fillna(0)/100
  193          df_landunits_and_pfts = df_landunits_and_pfts[df_landunits_and_pfts['ltype'] == 0].drop(columns=['ltype'])
  194          # Divide the vegetation percent by 100 to change it to a fraction and update the label accordingly. For each lat/lon, there could be many
  195          # rows, corresponding to a different value of natpft (see below). The land unit fraction will be the same for all natpft values (same value
  196          # across all such rows), so just take the mean to extract this land unit fraction value at each lat/lon coordinate.
  197          if variables_except_landunits_and_pfts:
  198              # Add the vegetation fraction to the overall DataFrame that will store all variables except land units and PFTs.
  199              df['FRAC_VEG_H0'] = df_landunits_and_pfts['PCT_LANDUNIT'].groupby(['lat', 'lon']).mean().fillna(0)/100
  200          else:
  201              # If the overall DataFrame is empty (no variables other than land units and PFTs were specified in the JSON files), create the DataFrame.
  202              df = df_landunits_and_pfts['PCT_LANDUNIT'].groupby(['lat', 'lon']).mean().fillna(0)/100
  203              df = df.to_frame()
  204              df = df.rename(columns={'PCT_LANDUNIT': 'FRAC_VEG_H0'})
  205          if 'PCT_CFT' in variables:
  206              df['FRAC_CROP_H0'] = frac_crop
  207          # Add a column for the area at each lat/lon coordinate to the overall DataFrame.
  208          df['AREA_H0 (km^2)'] = _per_row_spatial_values(areas / km2_TO_m2, df)
  209
  210      # 17 PFTs in order: 1 bare, 8 tree, 3 shrub, 3 grass, 1 crop, 1 empty. These can be further aggregated into subgroups as follows:
  211      # Bare soil (index 0), forest (the 8 trees, indices 1--8); shrub (indices 9--11); grass (indices 12--14), crop (index 15). Ignore the empty PFT.
  212      if 'PCT_LANDUNIT' in variables and 'PCT_NAT_PFT' in variables:
  213          df_landunits_and_pfts = df_landunits_and_pfts.reset_index(level='natpft')
  214          # Get data for the individual PFTs (again ignoring the empty one), as well as the aggregate subgroups.
  215          pft_labels = [f'PFT_{i}_AREA_H0 (km^2)' for i in range(16)]
  216          pft_labels.extend(['BARE_AREA_H0 (km^2)', 'FOREST_AREA_H0 (km^2)', 'SHRUB_AREA_H0 (km^2)', 'GRASS_AREA_H0 (km^2)', 'CROP_AREA_H0 (km^2)'])
  217          pft_min_max_indices = [(i, i) for i in range(16)]
  218          pft_min_max_indices.extend([(0, 0), (1, 8), (9, 11), (12, 14), (15, 15)])
  219          # Select only the rows that pertain to this particular PFT category and for each lat/lon coordinate, sum over all PFTs if a subgroup.
  220          for index, pft_label in enumerate(pft_labels):
  221              pft_min_index, pft_max_index = pft_min_max_indices[index][0], pft_min_max_indices[index][1]
  222              df_this_pft = df_landunits_and_pfts[(df_landunits_and_pfts['natpft'] >= pft_min_index) &
  223                                                  (df_landunits_and_pfts['natpft'] <= pft_max_index)]
  224              # Add up the percentages over all PFTs in the subgroup and divide that total percent by 100 to change it to a fraction.
  225              df_this_pft = df_this_pft['PCT_NAT_PFT'].groupby(['lat', 'lon']).sum().fillna(0)/100
  226              # Add a column to the overall DataFrame to record the area of this PFT category (individual or subgroup) at each lat/lon coordinate.
  227              df[pft_label] = df['AREA_H0 (km^2)']*df['FRAC_VEG_H0']*df_this_pft
  228
  229      # When crops are resolved by crop functional type (CFT), PCT_CFT gives the percent of each CFT on the crop land unit.
  230      # Convert those percentages to areas and use their sum as the total crop area, since the natural-PFT crop slot is empty.
  231      if 'PCT_LANDUNIT' in variables and 'PCT_CFT' in variables:
  232          df_cft = ds['PCT_CFT'].isel(time=0).to_dataframe()['PCT_CFT'].unstack('cft').fillna(0)/100
  233          cft_areas = df_cft.multiply(df['AREA_H0 (km^2)']*df['FRAC_CROP_H0'], axis=0)
  234          cft_areas.columns = [f'CFT_{cft}_AREA_H0 (km^2)' for cft in cft_areas.columns]
  235          df = pd.concat([df, cft_areas], axis=1)
  236          df['CROP_AREA_H0 (km^2)'] = cft_areas.sum(axis=1)
  237
  238      if lat_lon_aggregation_type == 'area_weighted_mean_or_sum':
  239          # Calculate an area-weighted mean or sum over all latitude/longitude coordinates for each variable.
  240
  241          # First, multiply all variables in the DataFrame except for the areas by the grid cell areas at each lat/lon coordinate.
  242          columns_to_multipy_by_areas = [label for label in df.columns if 'AREA' not in label]
  243          df[columns_to_multipy_by_areas] = df[columns_to_multipy_by_areas].multiply(
  244              _per_row_spatial_values(areas, df), axis=0)
  245
  246          # For EAM variables that correspond specifically to land or non-land (ocean) quantities, multiply by the land or non-land fractions.
  247          df[[label for label in df.columns if '_LND' in label]] = df[[label for label in df.columns if '_LND' in label]].multiply(
  248              _per_row_spatial_values(landfrac, df), axis=0)
  249          df[[label for label in df.columns if '_OCN' in label]] = df[[label for label in df.columns if '_OCN' in label]].multiply(
  250              _per_row_spatial_values(non_landfrac, df), axis=0)
  251
  252          # Sum over all latitude/longitude coordinates to get an area-weighted sum for each variable.
  253          df = df.sum().to_frame().T
  254
  255          # Variables that are not fluxes and stocks (so that they are not per-area quantities), should be global area-weighted means
  256          # rather than area-weighted sums, and therefore we need to divide these sums (which were computed a few lines above) by the total area.
  257          # Temperature variables are examples of such non-flux/non-stock variables.
  258          columns_for_means = [label for label in columns_to_multipy_by_areas
  259                               if not check_substrings_in_string(['/m^2', '/m2'], label, all_or_any='any')]
  260          # LND and OCN refer to certain types of EAM output for which we need to multiply the total grid cell areas by the land or non-land fractions.
  261          columns_for_means_LND = [label for label in columns_for_means if '_LND' in label]
  262          total_area = np.sum(areas*landfrac)
  263          df[columns_for_means_LND] /= total_area
  264          columns_for_means_OCN = [label for label in columns_for_means if '_OCN' in label]
  265          total_area = np.sum(areas*non_landfrac)
  266          df[columns_for_means_OCN] /= total_area
  267          # Variables that are not LND or OCN refer to outputs where we need to work with the full area of each grid cell.
  268          columns_for_means = [label for label in columns_for_means if (label not in columns_for_means_LND and label not in columns_for_means_OCN)]
  269          total_area = np.sum(areas)
  270          df[columns_for_means] /= total_area
  271
  272          # Fluxes and stocks are global area-weighted sums and have been multiplied by areas, so we must update the labels to remove the '/m^2' part.
  273          for old_label in ['/m^2', '/m2']:
  274              condition = lambda x: old_label in x
  275              new_column_label_function = lambda x: x.replace(old_label, '')
  276              df.columns = modify_list_based_on_condition(df.columns, condition, new_column_label_function)
  277
  278      elif lat_lon_aggregation_type == 'sum':
  279          # Perform a sum over all latitude/longitude coordinates for each variable.
  280          df = df.sum().to_frame().T
  281      elif lat_lon_aggregation_type == 'mean':
  282          # Calculate a mean over all latitude/longitude coordinates for each variable.
  283          df = df.mean().to_frame().T
  284
  285      # Add year and month columns.
  286      year, month = extract_year_and_month_from_name_of_netcdf_file(file)
  287      column_names_with_year_and_month_first = ['Year', 'Month']
  288      column_names_with_year_and_month_first.extend(df.columns)
  289      df['Year'] = year
  290      df['Month'] = month
  291      ds.close()
  292      print(f"  Extracted {os.path.basename(file)} ({year}-{month:02d})")
  293      return df[column_names_with_year_and_month_first]

  299  def extract_time_series_from_netcdf_files(simulation_path, output_file, netcdf_substrings, variables,
  300                  lat_lon_aggregation_types=None, regions=None, extra_dim_aggregation_types=None, process_variables=True, start_year=2015, end_year=2100, write_to_csv=False, max_processes=None):
  301      """
  302      Extracts time series data from E3SM-generated NetCDF h0 files in a simulation directory into a Pandas DataFrame and writes it to an output file.
  303      The NetCDF files can be of more than one type (e.g., one set generated from the ELM model and another set from the EAM model in E3SM).
  304      Each NetCDF file contains simulation results for one month in a particular year.
  305
  306      Parameters:
  307          simulation_path: Complete path of the directory containing the NetCDF output files from running a simulation.
  308          output_file: Name of the output file where the contents of the DataFrame will be written.
  309          netcdf_substrings: List where each element is itself a list of substrings. Each list corresponds to a particular type of NetCDF file and
  310                             indicates the substrings that must be in the names of that NetCDF file type.
  311          variables: List where each element is itself a list of variables we want to extract for each type of NetCDF file.
  312                     The aggregate of all variables contained in these lists will together form the columns of the DataFrame.
  313          lat_lon_aggregation_types: List of strings that indicate what type of lat/lon aggregation we want to perform to produce the time series data
  314                             for each type of NetCDF file. Current options include 'area_weighted_mean_or_sum', 'mean', and 'sum'.
  315          regions: List of strings for the region of interest. If not specified or not recognized, then there will be no restrictions on the
  316                   lat/lon coordinates (the entire globe will be used).
  317          extra_dim_aggregation_types: List of strings, one per NetCDF file type, controlling how variables with extra non-spatial dimensions are
  318                             collapsed before spatial aggregation. Current options are 'first' and 'mean'. If omitted, 'first' is used.
  319          process_variables: Boolean indicating if further processing is to be done on the outputs in the DataFrame. This includes adding new variables
  320                             not in the NetCDF files (e.g., total precipitation, mole fraction CO2) or changing the units (e.g., Pg instead of g).
  321          start_year: First year in the extracted time series data.
  322          end_year: Last year in the extracted time series data.
  323          max_processes: Maximum number of parallel processes to use. If not specified, defaults to all available CPUs minus one, with a minimum of one.
  324
  325      Returns:
  326          N/A.
  327      """
  328      # Each per-file-type input must have one entry for every NetCDF file type, including optional aggregation choices.
  329      per_file_type_inputs = {'netcdf_substrings': netcdf_substrings, 'variables': variables,
  330                              'lat_lon_aggregation_types': lat_lon_aggregation_types, 'regions': regions,
  331                              'extra_dim_aggregation_types': extra_dim_aggregation_types}
  332      lengths = {name: len(value) for name, value in per_file_type_inputs.items() if value}
  333      if len(set(lengths.values())) > 1:
  334          print("Error: the following inputs must have the same number of entries (one per NetCDF file type), but they differ:")
  335          for name, length in lengths.items():
  336              print(f"  {name}: {length}")
  337          sys.exit(1)
  338
  339      # Verify the output file is writable before doing any processing, so a bad path or permissions problem surfaces immediately.
  340      output_dir = os.path.dirname(os.path.abspath(output_file))
  341      if not os.path.exists(output_dir):
  342          print(f"Output directory does not exist: '{output_dir}'. Creating it now...")
  343          os.makedirs(output_dir)
  344      try:
