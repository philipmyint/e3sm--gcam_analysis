   13      """

   27  def extract_netcdf_file_into_dataframe_single_year(ds, areas, variables, region, year):
   28      """
   29      For the given year, extracts the specified variables from the E3SM human component (EHC) land surface data file into a Pandas DataFrame.
   30
   31      Parameters:
   32          ds: xarray Dataset for the NetCDF file generated by the EHC during run time. It contains data on land surface variables like harvest
   33          and grazing.
   34          areas: Grid cell areas.
   35          variables: List of variables we want to extract from the NetCDF file.
   36          region: String for the region of interest. If not specified or not recognized, then there will be no restrictions on the lat/lon coordinates.
   37          year: Year of interest.
   38
   39      Returns:
   40          DataFrame containing one column for each of the variables, plus a column for the specific year of interest.
   41      """
   42      # Make a local copy to avoid modifying the shared list.
   43      variables = variables.copy()
   44
   45      # If grazing or harvest are of interest, make sure to also extract the land fractions for vegetation and the plant-functional types (PFTs).
   46      if 'GRAZING' in variables or any('HARVEST_' in variable for variable in variables):
   47          if 'PCT_NAT_PFT' not in variables:
   48              variables.append('PCT_NAT_PFT')
   49          if 'PCT_NATVEG' not in variables:
   50              variables.append('PCT_NATVEG')
   51
   52      # CFT areas require the crop land unit fraction.
   53      if 'PCT_CFT' in variables and 'PCT_CROP' not in variables:
   54          variables.append('PCT_CROP')
   55
   56      # Convert the Dataset to create an overall DataFrame that stores all variables except those for land fractions and PFTs.
   57      # Create a second DataFrame to store the land fractions and PFTs.
   58      variables_except_pfts = variables.copy()
   59      variables_pfts = []
   60      if 'PCT_NAT_PFT' in variables:
   61          variables_except_pfts.remove('PCT_NAT_PFT')
   62          variables_pfts.append('PCT_NAT_PFT')
   63      # PCT_CFT is handled separately to avoid broadcasting it against the other variables.
   64      if 'PCT_CFT' in variables:
   65          variables_except_pfts.remove('PCT_CFT')
   66      if variables_except_pfts:
   67          # Select the year before converting to a DataFrame to avoid materializing all years on every call.
   68          df = ds[variables_except_pfts].sel(time=year).to_dataframe()
   69      if variables_pfts:
   70          df_pfts = ds[variables_pfts].sel(time=year).to_dataframe()
   71
   72      # Ensure df exists even if only PFT variables were requested.
   73      if not variables_except_pfts:
   74          # Create empty DataFrame with the correct spatial index from df_pfts.
   75          if variables_pfts:
   76              df = pd.DataFrame(index=df_pfts.index.droplevel('natpft').unique())
   77          else:
   78              raise ValueError('No variables requested!')
   79
   80      # Divide the vegetation percent by 100 to change it to a fraction and update the label accordingly.
   81      if 'PCT_NATVEG' in variables:
   82          df['PCT_NATVEG'] = df['PCT_NATVEG'].fillna(0)/100
   83          df = df.rename(columns={'PCT_NATVEG': 'FRAC_VEG'})
   84
   85          # SAFETY: Validate FRAC_VEG is a valid fraction.
   86          if (df['FRAC_VEG'] < -0.001).any() or (df['FRAC_VEG'] > 1.001).any():
   87              print(f'ERROR in year {year}: FRAC_VEG outside [0,1] range!')
   88              print(f'  min: {df["FRAC_VEG"].min():.6f}, max: {df["FRAC_VEG"].max():.6f}')
   89              raise ValueError(f'Invalid FRAC_VEG values in year {year}')
   90
   91      # Divide the crop land unit percent by 100 to change it to a fraction.
   92      if 'PCT_CROP' in variables:
   93          df['PCT_CROP'] = df['PCT_CROP'].fillna(0)/100
   94          df = df.rename(columns={'PCT_CROP': 'FRAC_CROP'})
   95          if (df['FRAC_CROP'] < -0.001).any() or (df['FRAC_CROP'] > 1.001).any():
   96              print(f'ERROR in year {year}: FRAC_CROP outside [0,1] range!')
   97              print(f'  min: {df["FRAC_CROP"].min():.6f}, max: {df["FRAC_CROP"].max():.6f}')
   98              raise ValueError(f'Invalid FRAC_CROP values in year {year}')
   99
  100      # Add a column for the area at each lat/lon coordinate to the overall DataFrame.
  101      df['AREA (km^2)'] = areas
  102
  103      # SAFETY: Validate grid cell areas
  104      if (areas < 0).any():
  105          print(f'ERROR in year {year}: Negative areas detected!')
  106          raise ValueError(f'Negative grid cell areas in year {year}')
  107
  108      if df['AREA (km^2)'].max() > 1e8:  # Earth's largest grid cell ~100,000 km²
  109          print(f'ERROR in year {year}: Unreasonably large areas!')
  110          print(f'  Max area: {df["AREA (km^2)"].max():.2e} km² (Earth surface = 5.1e8 km²)')
  111          raise ValueError(f'Invalid areas in year {year}')
  112
  113      # 17 PFTs in order: 1 bare, 8 tree, 3 shrub, 3 grass, 1 crop, 1 empty. These can be further subgrouped as follows:
  114      # Bare soil (index 0), forest (the 8 trees, indices 1--8); shrub (indices 9--11); grass (indices 12--14), crop (index 15). Ignore the empty PFT.
  115      if 'PCT_NAT_PFT' in variables:
  116
  117          df_pfts = df_pfts.reset_index(level='natpft')
  118
  119          # Get data for the individual PFTs (again ignoring the empty one), as well as the aggregate subgroups.
  120          pft_labels = [f'PFT_{i}_AREA (km^2)' for i in range(16)]
  121          pft_labels.extend(['BARE_AREA (km^2)', 'FOREST_AREA (km^2)', 'SHRUB_AREA (km^2)', 'GRASS_AREA (km^2)', 'CROP_AREA (km^2)'])
  122          pft_min_max_indices = [(i, i) for i in range(16)]
  123          pft_min_max_indices.extend([(0, 0), (1, 8), (9, 11), (12, 14), (15, 15)])
  124
  125          # Select only the rows that pertain to this particular PFT category and for each lat/lon coordinate, sum over all PFTS if a subgroup.
  126          for index, pft_label in enumerate(pft_labels):
  127              pft_min_index, pft_max_index = pft_min_max_indices[index][0], pft_min_max_indices[index][1]
  128              df_this_pft = df_pfts[(df_pfts['natpft'] >= pft_min_index) & (df_pfts['natpft'] <= pft_max_index)]
  129              # Add up the percentages over all PFTs in the subgroup and divide that total percent by 100 to change it to a fraction.
  130              df_this_pft = df_this_pft['PCT_NAT_PFT'].groupby(['lsmlat', 'lsmlon']).sum().fillna(0)/100
  131              # Add a column to the overall DataFrame to record the area of this PFT category (individual or subgroup) at each lat/lon coordinate.
  132              df[pft_label] = df['AREA (km^2)'] * df['FRAC_VEG'] * df_this_pft
  133
  134          # SAFETY: Validate PFT areas are reasonable.
  135          pft_area_cols = [col for col in df.columns if 'AREA (km^2)' in col and 'PFT' in col]
  136          for col in pft_area_cols:
  137              if (df[col] < 0).any():
  138                  print(f'ERROR in year {year}: Negative {col} detected!')
  139                  raise ValueError(f'Negative PFT area in year {year}')
  140              max_val = df[col].max()
  141              if max_val > 1e8:  # No PFT area should exceed max grid cell
  142                  print(f'ERROR in year {year}: {col} too large: {max_val:.2e} km²')
  143                  raise ValueError(f'Invalid PFT area in year {year}')
  144
  145      # Crops are on their own land unit (PCT_CROP), split by crop functional type (PCT_CFT), rather than in PCT_NAT_PFT.
  146      if 'FRAC_CROP' in df.columns:
  147          df['CROP_AREA (km^2)'] = df['AREA (km^2)']*df['FRAC_CROP']
  148      if 'PCT_CFT' in variables:
  149          df_cft = ds['PCT_CFT'].sel(time=year).to_dataframe()['PCT_CFT'].unstack('cft').fillna(0)/100
  150          cft_areas = df_cft.multiply(df['CROP_AREA (km^2)'], axis=0)
  151          cft_areas.columns = [f'CFT_{cft}_AREA (km^2)' for cft in cft_areas.columns]
  152          df = pd.concat([df, cft_areas], axis=1)
  153
  154      # Convert the grazing and harvest variables from a unitless fraction into an area by multiplying with the area of the PFT aggregate subgroup.
  155      grazing_harvest_variables = ['GRAZING', 'HARVEST_SH1', 'HARVEST_SH2', 'HARVEST_SH3', 'HARVEST_VH1', 'HARVEST_VH2']
  156      pft_area_columns = ['GRASS_AREA (km^2)', 'FOREST_AREA (km^2)', 'FOREST_AREA (km^2)', 'FOREST_AREA (km^2)', 'FOREST_AREA (km^2)',
  157                            'FOREST_AREA (km^2)']
  158      for index in range(len(grazing_harvest_variables)):
  159          grazing_harvest_variable = grazing_harvest_variables[index]
  160          # Only process this variable if it was actually requested and exists in the DataFrame.
  161          if grazing_harvest_variable in df.columns:
  162              df[grazing_harvest_variable] = df[grazing_harvest_variable]*df[pft_area_columns[index]]
  163              df = df.rename(columns={grazing_harvest_variable: grazing_harvest_variable + '_AREA (km^2)'})
  164
  165      # Create another column for the total harvest, which is the sum of all the harvest variables present in the DataFrame.
  166      harvest_variables = ['HARVEST_SH1_AREA (km^2)', 'HARVEST_SH2_AREA (km^2)', 'HARVEST_SH3_AREA (km^2)', 'HARVEST_VH1_AREA (km^2)',
  167                           'HARVEST_VH2_AREA (km^2)']
  168      harvest_variables = [x for x in harvest_variables if x in df.columns]
  169      if harvest_variables:
  170          df['HARVEST_AREA (km^2)'] = df[harvest_variables].sum(axis=1)
  171          df = move_columns_next_to_each_other_in_dataframe(df, harvest_variables[len(harvest_variables)-1], 'HARVEST_AREA (km^2)')
  172
  173      # Sum over all latitude/longitude coordinates to get an area-weighted sum for each variable.
  174      for frac_column in ['FRAC_VEG', 'FRAC_CROP']:
  175          if frac_column in df.columns:
  176              df[frac_column] *= df['AREA (km^2)']
  177
  178      # SAFETY: Check for inf/NaN before summing.
  179      if df.isin([np.inf, -np.inf]).any().any():
  180          inf_cols = df.columns[df.isin([np.inf, -np.inf]).any()].tolist()
  181          print(f'ERROR in year {year}: Infinite values in columns: {inf_cols}')
  182          raise ValueError(f'Infinite values detected in year {year}')
  183
  184      df = df.sum(skipna=True).to_frame().T
  185
  186      # To get global land unit fractions divide their area sums by the total land area.
  187      for frac_column in ['FRAC_VEG', 'FRAC_CROP']:
  188          if frac_column in df.columns:
  189              df[frac_column] /= np.sum(areas)
  190
  191      # Add year column.
  192      column_names_with_year_first = ['Year']
  193      column_names_with_year_first.extend(df.columns)
  194      df['Year'] = year
  195
  196      # SAFETY: Final validation of output (allow tiny floating-point errors).
  197      if (df.select_dtypes(include=[np.number]) < -0.001).any().any():
  198          numeric_cols = df.select_dtypes(include=[np.number]).columns
  199          neg_cols = [col for col in numeric_cols if (df[col] < -0.001).any()]
  200          print(f'ERROR in year {year}: Negative values in final output: {neg_cols}')
  201          raise ValueError(f'Negative values in final output for year {year}')
  202
  203      return df[column_names_with_year_first]

  205  def extract_time_series_from_netcdf_file(inputs):
  206      """
  207      Extracts the specified variables from an E3SM-generated NetCDF file into a Pandas DataFrame and performs the indicated operation on the variables.
  208      This NetCDF file is generated dynamically during run time by the E3SM human component (EHC) and contains land surface data like harvest, grazing.
  209
  210      Parameters:
  211          inputs: Dictionary containing the user choice inputs for different options, such as the variables that they want to extract from the file.
  212
  213      Returns:
  214          DataFrame containing one column for each of the variables, plus a column for the year.
  215      """
  216      # Start timing for this file.
  217      file_start_time = time.time()
  218
  219      # Extract all user selections, some being supplied with default choices. landuse_file is the full path and name of the land use NetCDF file.
  220      landuse_file = inputs['landuse_file']
  221      output_file = inputs['output_file']
  222      variables = inputs['variables']
  223      region = inputs.get('region', None)
  224      write_to_csv = inputs.get('write_to_csv', False)
  225      start_year = inputs.get('start_year', 2015)
  226      end_year = inputs.get('end_year', 2100)
  227
  228      # Verify the output file is writable before doing any processing, so a bad path or permissions problem surfaces immediately.
  229      output_dir = os.path.dirname(os.path.abspath(output_file))
  230      if not os.path.exists(output_dir):
  231          print(f"Output directory does not exist: '{output_dir}'. Creating it now...")
  232          os.makedirs(output_dir)
  233      try:
  234          open(output_file, 'a').close()
  235      except OSError as e:
  236          print(f"Error: cannot write to output file '{output_file}': {e}")
  237          return
  238
  239      # Load file. Do it in this function (outside of the multiprocessing steps) so that the file is loaded only once.
  240      # Prefilter to only load variables that are needed: the user-requested variables, those that may be added implicitly
  241      # for grazing/harvest calculations, and the auxiliary variables required for area/fraction calculations.
  242      # This reduces memory usage and I/O time by avoiding loading unnecessary variables from the file.
  243      print(f"Loading {landuse_file}...")
  244      auxiliary_vars = ['AREA', 'LANDFRAC_PFT', 'PFTDATA_MASK']
  245      implicit_vars = ['PCT_NAT_PFT', 'PCT_NATVEG', 'PCT_CROP']
  246      vars_to_retain = set(variables) | set(auxiliary_vars) | set(implicit_vars)
  247      all_vars_in_file = list(xr.open_dataset(landuse_file, decode_times=False).data_vars)
  248      drop_vars = [v for v in all_vars_in_file if v not in vars_to_retain]
  249      ds = xr.open_dataset(landuse_file, drop_variables=drop_vars)
  250      # Drop duplicate 'time' coordinate values (e.g., that get generated during restarts), keeping the data that correspond to the last occurrence.
  251      ds = ds.drop_duplicates(dim='time', keep='last')
  252
  253      # Calculate grid cell areas, with regional filtering if needed
  254      print("Calculating grid cell areas...")
  255      areas, ds, _, _ = find_gridcell_areas_in_netcdf_file_ds(ds, region=region, file_type='landuse_timeseries')
  256
  257      # Check that all requested variables exist in the dataset before doing any per-year processing.
  258      missing_vars = [v for v in variables if v not in ds.data_vars
  259                      and v not in ('PCT_NAT_PFT', 'PCT_NATVEG')]  # these may be added implicitly
  260      if missing_vars:
  261          print(f"Warning: the following variables were not found in {landuse_file} and will be skipped: {missing_vars}")
  262          variables = [v for v in variables if v not in missing_vars]
  263          if not variables:
  264              print(f"Warning: no valid variables remain. Skipping '{output_file}'.")
  265              ds.close()
  266              return
  267      print(f"Variables to extract ({len(variables)}): {variables}")
  268
  269      # Check which years actually exist in the file.
  270      print(f'Checking available years in {landuse_file}...')
  271      if 'YEAR' in ds:
  272          available_years = sorted([int(y) for y in ds['YEAR'].values])
  273      else:
  274          available_years = sorted([int(y) for y in ds['time'].values])
  275
  276      years_to_process = [y for y in available_years if start_year <= y <= end_year]
  277
  278      if len(years_to_process) == 0:
  279          print(f'WARNING: No years in range {start_year}-{end_year} found in file!')
  280          return
  281
  282      num_years = len(years_to_process)
  283
  284      # Historical per-year multiprocessing attempt. This is intentionally disabled because
  285      # sharing an open xarray Dataset across worker processes has been unreliable.
  286      # The active code below processes years sequentially within this one landuse file.
  287      '''
  288      arguments = list(zip([ds]*num_years, [areas]*num_years, [variables]*num_years,
  289                       [region]*num_years, years_to_process))
  290      num_processes = min(MAX_PROCESSES, num_years)
  291      print(f'Processing {num_years} years with multiprocessing...')
  292      with multiprocessing.Pool(processes=num_processes) as pool:
  293          dataframes_for_each_year = list(pool.starmap(extract_netcdf_file_into_dataframe_single_year, arguments))
  294      ds.close()
  295      '''
  296
  297      # Process each year sequentially within this landuse file.
  298      print(f"Processing {num_years} years sequentially...")
  299      dataframes_for_each_year = []
  300      for i, year in enumerate(years_to_process, 1):
  301          print(f" {landuse_file}: Processing year {year} [{i}/{num_years}]...")
  302          df = extract_netcdf_file_into_dataframe_single_year(ds, areas, variables, region, year)
  303          dataframes_for_each_year.append(df)
  304
  305      # Concatenate all DataFrames in the list together to form a single DataFrame over all years. Sort by year.
  306      df = pd.concat(dataframes_for_each_year)
  307      df.sort_values(['Year'], inplace=True)
  308
  309      # Check for missing years in the concatenated DataFrame and warn the user about any gaps.
  310      actual_years = set(df['Year'])
  311      missing_years = sorted(set(years_to_process) - actual_years)
  312      if missing_years:
  313          print(f"Warning: {len(missing_years)} missing years in output: {missing_years}")
  314
  315      # Close the dataset now that all years have been processed.
  316      ds.close()
  317
  318      # Write the DataFrame to the specified output file.
  319      if write_to_csv or output_file.endswith('.csv'):
  320          df.to_csv(output_file, index=False)
  321      else:
  322          write_dataframe_to_fwf(output_file, df)
  323
  324      # Print elapsed time for this file.
  325      file_elapsed_time = time.time() - file_start_time
  326      print(f'Elapsed time for {output_file}: {file_elapsed_time:.2f} seconds')
  327
  328
  329  ###---------------Begin execution---------------###
  330  if __name__ == '__main__':
  331
  332      # Run this script together with the input JSON file(s) on the command line.
  333      start_time_total = time.time()
  334      overwrite = False
  335      input_files = []
  336      for arg in sys.argv[1:]:
  337          if arg == '--overwrite':
  338              overwrite = True
  339          elif arg in ['-h', '--help']:
  340              print('Usage: python e3sm_extract_time_series_landuse.py [--overwrite] `path/to/json/input/file(s)\'')
  341              sys.exit()
  342          elif arg.startswith('-'):
  343              print(f"Error: unknown option: '{arg}'")
  344              print('Usage: python e3sm_extract_time_series_landuse.py [--overwrite] `path/to/json/input/file(s)\'')
  345              sys.exit(1)
  346          else:
  347              input_files.append(arg)
  348
  349      if not input_files:
  350          print('Usage: python e3sm_extract_time_series_landuse.py [--overwrite] `path/to/json/input/file(s)\'')
  351          sys.exit()
  352
  353      # Read and load the JSON file(s) into a list of dictionaries. Each block in a JSON file represents one time series output file.
  354      list_of_inputs = []
  355      for input_file in input_files:
  356          with open(input_file) as f:
  357              list_of_inputs.extend(json.load(f))
  358
  359      # Check upfront whether any output files already exist; report all conflicts before exiting unless overwrite was requested.
  360      if not overwrite:
  361          existing_files = [i['output_file'] for i in list_of_inputs if os.path.exists(i['output_file'])]
  362          if existing_files:
  363              for f in existing_files:
  364                  print(f"Error: output file already exists: '{f}'")
  365              print("Error: some files in output_file already exist. To create new output_files either delete/rename existing files or "
  366                    "rename project files in input json file. No new files have been created")
  367              print("Use --overwrite to replace existing output files.")
  368              sys.exit(1)
  369
  370      try:
  371          max_processes = get_max_processes_from_inputs(list_of_inputs, MAX_PROCESSES)
  372      except ValueError as e:
  373          print(f"Error: {e}")
  374          sys.exit(1)
  375
  376      # Multiprocessing is applied across JSON blocks/landuse files, not across years within a file.
  377      # The max_processes value caps how many landuse files are processed at the same time.
  378      print(f"\nProcessing {len(list_of_inputs)} file(s)...")
  379
  380      if len(list_of_inputs) == 1:
  381          # Single file: no need for multiprocessing.
  382          extract_time_series_from_netcdf_file(list_of_inputs[0])
  383      else:
  384          # Multiple files: process them in parallel.
  385          num_processes = min(max_processes, len(list_of_inputs))
  386          with multiprocessing.Pool(processes=num_processes) as pool:
  387              pool.map(extract_time_series_from_netcdf_file, list_of_inputs)
  388
  389      # Print the total execution time needed to complete all data extraction operations.
  390      end_time = time.time()
  391      elapsed_time = end_time - start_time_total
  392      print(f"Elapsed time for extracting all time series data: {elapsed_time:.2f} seconds")
