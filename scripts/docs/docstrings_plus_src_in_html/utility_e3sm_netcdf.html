    5  def extract_year_and_month_from_name_of_netcdf_file(file):
    6      """
    7      Finds the year and month from the name of a given E3SM-generated NetCDF file.
    8
    9      Parameters:
   10          file: NetCDF file containing data for one month in a particular year.
   11
   12      Returns:
   13          The year and month indicated in the name of the NetCDF file.
   14      """
   15      # Files are named *YYYY-MM.nc, so the year starts 7 characters before the .nc extension.
   16      index = file.find('.nc')
   17      index -= len('YYYY-MM')
   18      year = int(file[index:index+len('YYYY')])
   19      month = int(file[index+len('YYYY-'):index+len('YYYY-MM')])
   20      return year, month

   22  def extract_year_and_month_from_netcdf_file(file):
   23      """
   24      Finds the year and month from reading the contents of a given E3SM-generated NetCDF file.
   25
   26      Parameters:
   27          file: NetCDF file containing data for one month in a particular year.
   28
   29      Returns:
   30          The year and month indicated in the name of the NetCDF file.
   31      """
   32      ds = xr.open_dataset(file)
   33      year = ds['time'].dt.year.values[0]
   34      month = ds['time'].dt.month.values[0]
   35      return year, month

   37  def find_gridcell_areas_in_netcdf_file(file, region=None, variables_to_keep=None):
   38      """
   39      Obtains the grid cell areas of all latitude/longitude coordinates in an E3SM-generated (EAM or ELM or EHC) NetCDF file for the given region.
   40
   41      Parameters:
   42          file: NetCDF file.
   43          region: String for the region of interest. If not specified or not recognized, then there will be no restrictions on the lat/lon coordinates.
   44          variables_to_keep: List of user-requested variables to retain when opening the dataset. All other variables not needed for area/fraction
   45                             calculations will be dropped to reduce memory usage and I/O time. If None, all variables are loaded.
   46
   47      Returns:
   48          NumPy array containing the grid cell areas of all coordinates in units of m^2 and an xarray Dataset containing data from the file.
   49          Also returns the land and non-land (which is defined as ocean if in the case of EAM) fractions.
   50      """
   51      # Determine file type from filename so we know which auxiliary variables (areas, fractions, masks) must be retained.
   52      if 'elm.h0' in file:
   53          file_type = 'elm.h0'
   54          auxiliary_vars = ['area', 'landfrac', 'landmask', 'pftmask']
   55      elif 'eam.h0' in file:
   56          file_type = 'eam.h0'
   57          auxiliary_vars = ['area', 'LANDFRAC', 'OCNFRAC']
   58      elif 'surfdata_iESM_dyn' in file or 'landuse' in file:
   59          file_type = 'landuse_timeseries'
   60          auxiliary_vars = ['AREA', 'LANDFRAC_PFT', 'PFTDATA_MASK']
   61      else:
   62          file_type = 'landuse_timeseries'  # Default
   63          auxiliary_vars = ['AREA', 'LANDFRAC_PFT', 'PFTDATA_MASK']
   64
   65      # Load the file into an xarray Dataset, dropping all variables that are neither auxiliary (needed for area/fraction calculations)
   66      # nor user-requested. This reduces memory usage and I/O time by avoiding loading unnecessary variables from the file.
   67      if variables_to_keep is not None:
   68          all_vars_in_file = list(xr.open_dataset(file, decode_times=False).data_vars)
   69          vars_to_retain = set(auxiliary_vars) | set(variables_to_keep)
   70          drop_vars = [v for v in all_vars_in_file if v not in vars_to_retain]
   71          ds = xr.open_dataset(file, drop_variables=drop_vars)
   72      else:
   73          ds = xr.open_dataset(file)
   74
   75      # Employ method chaining to call the version of this function that assumes the ds has been loaded already
   76      return find_gridcell_areas_in_netcdf_file_ds(ds, region, file_type)

   78  def find_gridcell_areas_in_netcdf_file_ds(ds, region=None, file_type='landuse_timeseries'):
   79      """
   80      Obtains the grid cell areas of all latitude/longitude coordinates in an E3SM-generated (EAM or ELM or EHC) NetCDF file for the given region.
   81      This function assumes that the file has been pre-loaded into an xarray Dataset.
   82
   83      Parameters:
   84          ds: Pre-loaded xarray Dataset.
   85          region: String for the region of interest. If not specified or not recognized, then there will be no restrictions on the lat/lon coordinates.
   86          file_type: Type of file. Options include: 'elm.h0', 'eam.h0', or 'landuse_timeseries'.
   87
   88      Returns:
   89          NumPy array containing the grid cell areas of all coordinates in units of m^2 and an xarray Dataset containing data from the file.
   90          Also returns the land and non-land (which is defined as ocean if in the case of EAM) fractions.
   91      """
   92      # If a region has been specified, get the bounds on the lat/lon coordinates and apply these bounds to restrict the lat/lon coordinates.
   93      if region:
   94          # Lon and lat bounds from get_regional_bounds(region) are in the range 0 to 360 and -90 to 90, respectively.
   95          bounds = get_regional_bounds(region)
   96          lon_bounds = bounds[:2]
   97          lat_bounds = bounds[2:]
   98
   99          # If the lon and lat coordinates in the file are in the range of -180 to 180 and 0 to 180, respectively,
  100          # then convert the bounds to be in the same range as the coordinates in the file.
  101          # Then apply the bounds to restrict the lat/lon coordinates in the Dataset.
  102          if file_type == 'elm.h0' or file_type == 'eam.h0':
  103              if file_type == 'eam.h0':
  104                  ds.load()
  105              if ds.lon.min() < 0:
  106                  lon_bounds = lon_bounds - 180
  107              if ds.lat.max() > 90:
  108                  lat_bounds = lat_bounds + 90
  109              ds = ds.where((ds.lon >= lon_bounds[0]) & (ds.lon <= lon_bounds[1]), drop=True)
  110              ds = ds.where((ds.lat >= lat_bounds[0]) & (ds.lat <= lat_bounds[1]), drop=True)
  111          elif file_type == 'landuse_timeseries':
  112              if ds.LONGXY.min() < 0:
  113                  lon_bounds = lon_bounds - 180
  114              if ds.LATIXY.max() > 90:
  115                  lat_bounds = lat_bounds + 90
  116              ds = ds.where((ds.LONGXY >= lon_bounds[0]) & (ds.LONGXY <= lon_bounds[1]), drop=True)
  117              ds = ds.where((ds.LATIXY >= lat_bounds[0]) & (ds.LATIXY <= lat_bounds[1]), drop=True)
  118
  119      if file_type == 'elm.h0':
  120          # If the NetCDF file is produced by the ELM model, multiply the areas by the land fraction, plus additional land and pft masks.
  121          areas = create_numpy_array_from_ds(ds, ['area'], [0])
  122          variables = ['landfrac', 'landmask', 'pftmask']
  123          landfrac, landmask, pftmask = create_numpy_array_from_ds(ds, variables, [0]*len(variables))
  124          areas *= landfrac*landmask*pftmask*km2_TO_m2
  125          landfrac *= landmask*pftmask
  126          non_landfrac = 1 - landfrac
  127      elif file_type == 'eam.h0':
  128          # If the NetCDF file is produced by the EAM model, multiply the areas by the Earth surface area so that the areas are in units of m^2.
  129          areas = create_numpy_array_from_ds(ds, ['area'], [0])
  130          areas *= SURF_AREA
  131          # Find the land and ocean fractions.
  132          variables = ['LANDFRAC', 'OCNFRAC']
  133          landfrac, non_landfrac = create_numpy_array_from_ds(ds, variables, [0]*len(variables))
  134      elif file_type == 'landuse_timeseries':
  135          # Case where the NetCDF file is the land surface data file produced dynamically by the E3SM human component (EHC) model during run time.
  136          areas = create_numpy_array_from_ds(ds, ['AREA'], [0])
  137          variables = ['LANDFRAC_PFT', 'PFTDATA_MASK']
  138          landfrac, pftmask = create_numpy_array_from_ds(ds, variables, [0]*len(variables))
  139          areas *= landfrac*pftmask
  140          landfrac *= pftmask
  141          non_landfrac = 1 - landfrac
  142      return areas, ds, landfrac, non_landfrac

  144  def get_netcdf_files_between_start_and_end_years(files, start_year, end_year):
  145      """
  146      Finds all E3SM-generated NetCDF files in a given list that fall between the start and end years.
  147
  148      Parameters:
  149          files: List of NetCDF files.
  150          start_year: Start year.
  151          end_year: End year.
  152
  153      Returns:
  154          List containing the NetCDF files that fall between the start and end years.
  155      """
  156      new_file_paths = []
  157      for file in files:
  158          year, _ = extract_year_and_month_from_name_of_netcdf_file(file)
  159          if year >= start_year and year <= end_year:
  160              new_file_paths.append(file)
  161      return new_file_paths
