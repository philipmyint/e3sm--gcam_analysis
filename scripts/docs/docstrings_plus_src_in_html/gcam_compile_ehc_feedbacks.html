   12  """
   13  Compile EHC feedback files into the same long-table format used by the GCAM
   14  plotting scripts.
   15
   16  The E3SM human component writes several small CSV files while it runs. Those
   17  files are organized by scenario and year, and some of them contain multiple
   18  physical quantities in separate columns. This script combines the per-year files

   26      Writes two output files: one for vegetation scalars and one for soil scalars, each with a 'value' column and
   27      optional 'area' and 'area_units' columns added from a processed land allocation file.

   32                  corresponding to one directory.

   35          N/A.

   43      # Scenarios not present in the dict fall back to the default root name 'scalars'.

   49      # quantity the same way as other processed GCAM tables: one 'value' column
   50      # plus metadata columns.

   52      soil_output_file = os.path.join(output_dir, 'soil_scalars.csv')
   53
   54      # Check that input_directories and scenarios are the same length.
   55      if len(input_directories) != len(scenarios):
   56          print(f"Error: 'input_directories' (length {len(input_directories)}) and 'scenarios' "
   57                f"(length {len(scenarios)}) must be the same length.")
   58          return
   59

   61      for out_file in [vegetation_output_file, soil_output_file]:
   62          try:
   63              open(out_file, 'a').close()
   64          except OSError as e:
   65              print(f"Error: cannot write to output file '{out_file}': {e}")
   66              return

  125

  130              return

  133      df = df[key_columns + ['vegetation', 'soil']]
  134
  135      # Add areas before standardizing crop names. The scalar files still use the
  136      # original GCAM crop names here, so the land allocation file should be the
  137      # processed land allocation table that also keeps original crop names
  138      # (call_modify_crop_names = false in gcam_process_extracted_data.json).

  146          return

  148      df = add_areas_to_dataframe(df, df_land)

  150      # Standardize crop names after adding areas so the merge above uses original names.
  151      if call_modify_crop_names:
  152          df = modify_crop_names(df, key_columns, 'area_weighted_mean')
  153
  154      # Keep area metadata at the end of the output tables if it was successfully
  155      # added. Plotting scripts can use the area column for area-weighted means.
  156      trailing = [c for c in ['area', 'area_units'] if c in df.columns]
  157

  159      # data column to 'value' lets plotting JSON omit value_label or set it to the
  160      # default 'value'.

  162      veg_df['units'] = 'Unitless'

  166      # Write soil scalars with the same structure as vegetation_scalars.csv.

  168      soil_df['units'] = 'Unitless'
  169      soil_df = soil_df[key_columns + ['value', 'units'] + trailing]

  171
  172      end_time = time.time()
  173      elapsed_time = end_time - start_time
  174      print(f"Elapsed time processing/compiling the data for {vegetation_output_file} and {soil_output_file}: {elapsed_time:.2f} seconds")

  426  def compile_ehc_feedbacks(inputs):
  427      """Dispatcher that calls compile functions based on the 'file_type' key.
  428      'file_type' can be:
  429        'all' (default) — runs all three types with their default root names
  430        a string        — runs that type, using the string as both type and root name
  431        a list          — runs those types, each string as both type and root name
  432        a dict          — keys are type identifiers, values are the root names to search for
  433      Valid type identifiers: 'scalars', 'hdd_cdd_scalars', 'scaled_hdd_cdd_diag'."""
  434      # Map each file_type identifier to the function that knows that input
  435      # layout. The JSON file_type field controls which of these readers runs for
  436      # a given input block.
  437      all_types = ['scalars', 'hdd_cdd_scalars', 'scaled_hdd_cdd_diag']
  438      type_to_func = {
  439          'scalars': compile_ehc_scalars,
  440          'hdd_cdd_scalars': compile_hdd_cdd_scalars,
  441          'scaled_hdd_cdd_diag': compile_scaled_hdd_cdd_diag,
  442      }
  443      # Normalize file_type into (type identifier, file prefix) pairs:
  444      #   "scalars" runs the scalar reader on files starting with "scalars";
  445      #   ["scalars", "hdd_cdd_scalars"] runs both default prefixes;
  446      #   {"scalars": "my_scalars"} runs the scalar reader with a custom prefix.
  447      raw = inputs.get('file_type', 'all')
  448      if raw == 'all':
  449          type_root_pairs = [(t, t) for t in all_types]
  450      elif isinstance(raw, dict):
  451          type_root_pairs = list(raw.items())
  452      elif isinstance(raw, str):
  453          type_root_pairs = [(raw, raw)]
  454      else:
  455          type_root_pairs = [(t, t) for t in raw]
  456      for file_type, root_name in type_root_pairs:
  457          if file_type not in type_to_func:
  458              print(f"Error: unknown file_type '{file_type}'. Valid types: {all_types}.")
  459              return
  460          modified_inputs = dict(inputs)
  461          # Store the chosen root name in a private key so the lower-level compile
  462          # function can use the same input dictionary without changing its public
  463          # JSON interface.
  464          modified_inputs['_file_root_name'] = root_name
  465          type_to_func[file_type](modified_inputs)
  466
  467
  468  ###---------------Begin execution---------------###
  469  if __name__ == '__main__':
  470
  471      # Run this script together with the input JSON file(s) on the command line.
  472      start_time = time.time()
  473      if len(sys.argv) < 2:
  474          print('Usage: python gcam_compile_ehc_feedbacks.py `path/to/json/input/file(s)\'')
  475          sys.exit()

  477      # Read and load the JSON file(s) into a list of dictionaries.
  478      list_of_inputs = []
  479      for i in range(1, len(sys.argv)):
  480          input_file = sys.argv[i]
  481          with open(input_file) as f:
  482              list_of_inputs.extend(json.load(f))
  483
  484      # Use any max_processes value requested in the inputs, otherwise use the default.
  485      # Do not start more worker processes than there are independent input blocks.
  486      max_processes = get_max_processes_from_inputs(list_of_inputs, MAX_PROCESSES)
  487      num_processes = min(max_processes, max(1, len(list_of_inputs)))
  488
  489      # Each top-level JSON block is independent, so multiple blocks can be
  490      # compiled in parallel. Inside each block, the files for that block are read
  491      # serially to keep the per-output ordering and error messages simple.
  492      with multiprocessing.Pool(processes=num_processes) as pool:
  493          pool.map(compile_ehc_feedbacks, list_of_inputs)
  494
  495      # Print the total execution time needed to process/compile all requested
  496      # feedback files.
  497      end_time = time.time()
  498      elapsed_time = end_time - start_time
  499      print(f"Elapsed time processing/compiling the data for all files: {elapsed_time:.2f} seconds")
