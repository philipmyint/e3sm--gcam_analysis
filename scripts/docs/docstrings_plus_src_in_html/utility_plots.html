   64      """
   65      Creates a contour plot.
   66
   67      Parameters:

   74          N/A.

   76      fig, ax = plt.subplots(nrows=1, ncols=1)

   79      plot = ax.contourf(x, y, z, levels=levels)
   80      cbar = fig.colorbar(plot)
   81      cbar.ax.tick_params(labelsize=tick_label_size_default)
   82      cbar_label = options.get('cbar_label', None)
   83      if cbar_label:
   84          cbar.set_label(fr'{cbar_label}', fontsize=axis_label_size_default)
   85      cbar.set_ticks(cbar.locator.tick_values(z.min(), z.max()))
   86      set_figure_options(fig, ax, options)

   89      """
   90      Saves a figure as either a .pdf (default) or .png (if figure name ends with .png or if produce_png in the options dictionary is True).

   92      Parameters:
   93          name: Name of the figure.

   96
   97      Returns:

  100      produce_png = options.get('produce_png', produce_png_default)
  101      bbox_inches = options.get('bbox_inches', bbox_inches_default)
  102      if produce_png or name.endswith('.png'):
  103          if name.endswith('.png'):

  105          else:

  107      else:
