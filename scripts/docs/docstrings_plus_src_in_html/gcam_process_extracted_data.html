   13  def process_extracted_data(inputs):
   14      """
   15      Processes a file containing data extracted from GCAM project files using XML queries. The data are rearranged, split, aggregated in various ways,
   16      including potentially aggregating landtype groups, and the resulting processed data then gets written as an output file in .dat or .csv format.
   17
   18      Parameters:
   19          inputs: Dictionary with user-specified inputs, like the names of the input file (the one extracted from GCAM) and the output (processed) file.
   20
   21      Returns:
   22          True if processing completed successfully; False otherwise.
   23      """
   24      start_time = time.time()
   25      input_file = inputs['input_file']
   26      output_file = inputs['output_file']
   27      columns_to_drop = inputs.get('columns_to_drop', None)
   28      columns_to_split = inputs.get('columns_to_split', None)
   29      columns_to_split_on_substring = inputs.get('columns_to_split_on_substring', None)
   30      key_columns = [c.lower() for c in inputs['key_columns']] if inputs.get('key_columns') else None
   31      aggregate_function = inputs.get('aggregate_function', "sum")
   32      call_modify_crop_names = inputs.get('call_modify_crop_names', False)
   33      land_allocation_file = inputs.get('land_allocation_file', None)
   34
   35      # Verify input file exists and output file is writable before doing any processing.
   36      if not os.path.exists(input_file):
   37          print(f"Error: input file '{input_file}' does not exist.")
   38          return False
   39      if os.path.getsize(input_file) == 0:
   40          print(f"Warning: input file '{input_file}' is empty. Skipping.")
   41          return False
   42      output_dir = os.path.dirname(os.path.abspath(output_file)) or '.'
   43      if not os.path.isdir(output_dir):
   44          print(f"Error: output directory does not exist: '{output_dir}'")
   45          return False
   46      if not os.access(output_dir, os.W_OK):
   47          print(f"Error: output directory is not writable: '{output_dir}'")
   48          return False
   49
   50      df = read_file_into_dataframe(input_file)
   51      if df.empty or len(df.columns) == 0:
   52          print(f"Warning: input file '{input_file}' contains no data. Skipping.")
   53          return False
   54
   55      # Check that columns_to_drop columns actually exist before dropping, and warn about any that are missing.
   56      if columns_to_drop:
   57          missing_drop = [c for c in columns_to_drop if c not in df.columns]
   58          if missing_drop:
   59              print(f"Warning: the following columns_to_drop were not found in '{input_file}' and will be skipped: {missing_drop}")
   60          columns_to_drop_existing = [c for c in columns_to_drop if c in df.columns]
   61          df = df.drop(columns_to_drop_existing, axis=1)
   62
   63      # Lowercase all column names to avoid case sensitivity issues.
   64      df.columns = df.columns.str.lower()
   65
   66      # Check that the 'year' column exists after lowercasing before sorting by it.
   67      if 'year' not in df.columns:
   68          print(f"Error: no 'year' column found in '{input_file}' after lowercasing. Available columns: {list(df.columns)}")
   69          return False
   70      df = df.sort_values(by='year')
   71
   72      # For carbon-density files, replace 'None Specified' in the 'units' column with 'MgC/ha'.
   73      if 'carbon_density' in input_file and 'units' in df.columns:
   74          df.loc[df['units'] == 'None Specified', 'units'] = 'MgC/ha'
   75
   76      # For scaler files, replace NA in the 'units' column with 'Unitless'.
   77      if 'scaler' in input_file and 'units' in df.columns:
   78          df['units'] = df['units'].fillna('Unitless')
   79
   80      # Split each specified column (if any) separated by underscores into a set of new columns.
   81      # Rows with no '_' have the full value placed in the first new column and remaining new
   82      # columns set to "". Rows with '_' are split across the new columns; extra new columns
   83      # beyond the number of split elements are set to "". It is an error if any row produces
   84      # more split elements than there are new columns.
   85      if columns_to_split:
   86          for column_to_split, new_columns in columns_to_split.items():
   87              # Check that the column to split exists after lowercasing.
   88              if column_to_split not in df.columns:
   89                  print(f"Error: columns_to_split column '{column_to_split}' not found in '{input_file}' after lowercasing. "
   90                        f"Available columns: {list(df.columns)}")
   91                  return False
   92              mask = df[column_to_split].str.contains('_', na=False)
   93              # Initialize all new columns to "".
   94              for col_name in new_columns:
   95                  df[col_name] = ""
   96              if mask.any():
   97                  split_result = df.loc[mask, column_to_split].str.split('_', expand=True)
   98                  if split_result.shape[1] > len(new_columns):
   99                      print(f"Error: columns_to_split column '{column_to_split}' in '{input_file}' produces "
  100                            f"{split_result.shape[1]} split elements but only {len(new_columns)} new columns were specified.")
  101                      return False
  102                  for i, col_name in enumerate(new_columns):
  103                      if i < split_result.shape[1]:
  104                          df.loc[mask, col_name] = split_result.iloc[:, i].fillna('')
  105              # Rows without '_': the full value goes into the first new column.
  106              df.loc[~mask, new_columns[0]] = df.loc[~mask, column_to_split]
  107
  108      # Split each specified column on the first occurrence of a substring, placing text before it
  109      # in the first new column and text from the substring onward in the second new column.
  110      if columns_to_split_on_substring:
  111          for col, spec in columns_to_split_on_substring.items():
  112              col_lower = col.lower()
  113              if col_lower not in df.columns:
  114                  print(f"Error: columns_to_split_on_substring column '{col}' not found in '{input_file}' "
  115                        f"after lowercasing. Available columns: {list(df.columns)}")
  116                  return False
  117              split_before = spec['split_before']
  118              new_columns = spec['new_columns']
  119              pattern = re.compile(r'(?=' + re.escape(split_before) + r')')
  120              parts = df[col_lower].str.split(pattern, n=1, expand=True)
  121              # Drop before assigning new columns to avoid clobbering a new column that shares the source name.
  122              df = df.drop(columns=[col_lower])
  123              if parts.shape[1] == 1:
  124                  # Substring not found in any row; full value goes in first new column.
  125                  df[new_columns[0]] = parts[0]
  126                  for nc in new_columns[1:]:
  127                      df[nc] = ''
  128              else:
  129                  df[new_columns[0]] = parts[0]
  130                  df[new_columns[1]] = parts[1].fillna('')
  131
  132      # Sort columns by a set of keys if specified to do so.
  133      if key_columns:
  134          # Check that all key_columns exist (including any newly created columns from columns_to_split).
  135          missing_keys = [c for c in key_columns if c not in df.columns]
  136          if missing_keys:
  137              print(f"Error: the following key_columns were not found in '{input_file}' after all processing steps: {missing_keys}. "
  138                    f"Available columns: {list(df.columns)}")
  139              return False
  140          # Check that a 'value' column exists before selecting key_columns + ['value'].
  141          if 'value' not in df.columns:
  142              print(f"Error: no 'value' column found in '{input_file}'. Available columns: {list(df.columns)}")
  143              return False
  144          df = df.sort_values(by=key_columns)
  145          df = df[key_columns + ['value']]
  146
  147          # Add areas from the processed land allocation file.
  148          # Skip this for land allocation data itself, since those files ARE the area data.
  149          if 'land_allocation' not in input_file:
  150              if land_allocation_file is None:
  151                  print(f"Error: A 'land_allocation_file' file must be specified in the JSON input")
  152                  print(f"Error: So first the 'detailed_land_allocation_processed.csv' must be created by running "
  153                        f"the 'gcam_process_extracted_data.py' script on the 'detailed_land_allocation.csv' file."
  154                        f"without modifying the crop names (call_modify_crop_names = false).")
  155                  print(f"Error: The default 'land_allocation_file' is called 'detailed_land_allocation_processed.csv' "
  156                        f"and is should be located in the 'output/gcam_csv/' directory.")
  157                  return False
  158              if not os.path.exists(land_allocation_file):
  159                  print(f"Error: land_allocation_file not found: '{land_allocation_file}'")
  160                  return False
  161              if os.path.getsize(land_allocation_file) == 0:
  162                  print(f"Error: land_allocation_file is empty: '{land_allocation_file}'")
  163                  return False
  164              df_land = read_file_into_dataframe(land_allocation_file)
  165              df = add_areas_to_dataframe(df, df_land)
  166
  167      # Perform either a sum (default) or a weighted mean if there happens to be more than one row that agrees on all the table keys
  168      #    (i.e., belongs to the same group).
  169      # This also works for means where the areas are the same, e.g., groups that are in the same area unit,
  170      #  in which case the weighted mean is equivalent to a simple mean.
  171      if aggregate_function == 'sum' and key_columns:
  172          # Save string non-key columns (e.g. area_units) to rejoin after aggregation,
  173          # since groupby().sum() cannot aggregate string columns.
  174          str_non_key = [c for c in df.columns if c not in key_columns and df[c].dtype == object]
  175          str_first = df.groupby(key_columns)[str_non_key].first().reset_index() if str_non_key else None
  176          df = df.drop(columns=str_non_key).groupby(key_columns).sum().reset_index()
  177          if str_first is not None:
  178              df = df.merge(str_first, on=key_columns, how='left')
  179      elif aggregate_function == 'area_weighted_mean' and key_columns:
  180          # Save string non-key columns (e.g. area_units) to rejoin after aggregation.
  181          str_non_key = [c for c in df.columns if c not in key_columns and df[c].dtype == object]
  182          str_first = df.groupby(key_columns)[str_non_key].first().reset_index() if str_non_key else None
  183          df_numeric = df.drop(columns=str_non_key)
  184          # Compute area-weighted mean: for each group, sum(value * area) / sum(area).
  185          df_numeric['weighted_value'] = df_numeric['value'] * df_numeric['area']
  186          grouped = df_numeric.groupby(key_columns)[['weighted_value', 'area']].sum().reset_index()
  187          grouped['value'] = grouped['weighted_value'] / grouped['area'].replace(0, float('nan'))
  188          df = grouped.drop('weighted_value', axis=1)
  189          if str_first is not None:
  190              df = df.merge(str_first, on=key_columns, how='left')
  191
  192      # Update original crop names to a common, standardized set of names.
  193      if call_modify_crop_names:
  194          df = modify_crop_names(df, key_columns, aggregate_function)
  195
  196      # Reorder columns so the last four are: value, units, area, area_units.
  197      trailing = [c for c in ['value', 'units', 'area', 'area_units'] if c in df.columns]
  198      other_cols = [c for c in df.columns if c not in trailing]
  199      df = df[other_cols + trailing]
  200
  201      write_dataframe_to_file(df, output_file)
  202      end_time = time.time()
  203      elapsed_time = end_time - start_time
  204      print(f"Elapsed time for producing {output_file}: {elapsed_time:.2f} seconds")
  205      return True
