  195  def plot_seasons(include_seasons, ax, df, x, output_label, plot_colors, linestyle_tuples, linewidth, columns,
  196                   reference_data=None, file_or_file_set_index=None, std_multiplier=None, error_bars_alpha=None):
  197      """
  198      Adds seasons (as specified in the include_seasons dictionary) to a given time series plot.
  199
  200      Parameters:
  201          include_seasons: Dictionary specifying the seasons to include in the plot.
  202          ax: Axis object for the plot.
  203          df: Pandas DataFrame containing the time series data.
  204          x: Years to put on the x-axis.
  205          output_label: Base output label.
  206          plot_colors: List of plot line colors.
  207          linestyle_tuples: Tuples for the linestyles.
  208          linewidth: Thickness of the lines.
  209          columns: String (if a single column) or list (multiple columns) for the datasets in the DataFrame that we want to plot.
  210                   If a list, plot the mean across all columns.
  211          reference_data: Dictionary to store the data from the first file or first file set, which will serve as the reference set for
  212                   percent difference calculations.
  213          file_or_file_set_index: Index for the particular file or file set, so that we can flag whether the current data pertain to the reference set.
  214          std_multiplier: Multiplier on the standard deviation for plotting error bars.
  215          error_bars_alpha: Opacity value for the error bars.
  216
  217      Returns:
  218          The reference_data, which will be updated with entries for the reference data for each season if this dictionary was not initially empty.
  219      """
  220      seasons = ['MAM', 'JJA', 'SON', 'DJF']
  221      months_geq = [3, 6, 9, 12]
  222      months_leq = [5, 8, 11, 2]
  223      linestyle_tuples = [linestyle_tuples[2][1], linestyle_tuples[2][1], linestyle_tuples[1][1], linestyle_tuples[1][1]]
  224      linewidths = [linewidth+2, linewidth, linewidth+2, linewidth]
  225      # Iterate over all seasons and include only the ones that are specified in the include_seasons dictionary.
  226      for season_index in range(len(seasons)):
  227          season = seasons[season_index]
  228          if include_seasons[season]:
  229              month_geq = months_geq[season_index]
  230              month_leq = months_leq[season_index]
  231              linestyle=linestyle_tuples[season_index]
  232              linewidth=linewidths[season_index]
  233              # December--January--February needs to be handled differently than the other seasons.
  234              if season == 'DJF':
  235                  condition = (df['Month'] >= month_geq) | (df['Month'] <= month_leq)
  236              else:
  237                  condition = (df['Month'] >= month_geq) & (df['Month'] <= month_leq)
  238              # If we have multiple datasets to plot, calculate the standard deviation across the datasets for each year for each of the seasons.
  239              if isinstance(columns, list):
  240                  y = df[condition].groupby('Year', as_index=False).mean()[columns].mean(axis=1)
  241                  y_std = df[condition].groupby('Year', as_index=False).mean()[columns].std(axis=1)
  242              else:
  243                  y = df[condition].groupby('Year', as_index=False).mean()[columns]
  244              # If the reference data is not None or not empty, we assume that the user wants to perform a percent difference calculation.
  245              if reference_data:
  246                  if file_or_file_set_index == 0:
  247                      reference_data[season] = y
  248                  y = (y - reference_data[season])/(reference_data[season] + EPSILON)*100
  249                  if isinstance(columns, list):
  250                      y_std = y_std/(reference_data[season] + EPSILON)*100
  251              # Add the season to the plot, including its error bar if the multiplier is not None.
  252              # Do not include the reference data if plotting a percent difference. Include the data in the plot otherwise.
  253              if not reference_data or file_or_file_set_index != 0:
  254                  ax.plot(x, y, label=output_label + f' ({season})', color=plot_colors, linestyle=linestyle, linewidth=linewidth)
  255                  if std_multiplier and isinstance(columns, list):
  256                      error = y_std*std_multiplier
  257                      ax.fill_between(x, y-error, y+error, color=plot_colors, alpha=error_bars_alpha)
  258      return reference_data

  260  def read_file_into_single_variable_dataframe(file, variable, start_year, end_year, multiplier, return_column=False):
  261      """
  262      Reads a file and puts the time series data for one specific variable from that file into a Pandas DataFrame.
  263
  264      Parameters:
  265          file: Directory and name for the file we want to read.
  266          variable: The variable of interest.
  267          start_year: First year in the extracted time series data.
  268          end_year: Last year in the extracted time series data.
  269          multiplier: Multiplier on the time series (in case the user wants to change the units).
  270          return_column: Boolean that specifies if we want to additionally return the column in the DataFrame corresponding to the variable.
  271
  272      Returns:
  273          DataFrame containing the data for the variable and also the column in the DataFrame for that variable if return_column is True.
  274      """
  275      df = read_file_into_dataframe(file)
  276      df = df[(df['Year'] >= start_year) & (df['Year'] <= end_year)]
  277
  278      # Check for missing year-month combinations in the filtered data. Warn the user about any gaps in the time series.
  279      if 'Month' in df.columns:
  280          expected = {(y, m) for y in range(start_year, end_year + 1) for m in range(1, 13)}
  281          actual = set(zip(df['Year'], df['Month']))
  282          missing = sorted(expected - actual)
  283          if missing:
  284              print(f"Warning: {len(missing)} missing year-month combinations in {file}: {missing[:5]}{'...' if len(missing) > 5 else ''}")
  285      else:
  286          expected_years = set(range(start_year, end_year + 1))
  287          actual_years = set(df['Year'])
  288          missing_years = sorted(expected_years - actual_years)
  289          if missing_years:
  290              print(f"Warning: {len(missing_years)} missing years in {file}: {missing_years}")
  291
  292      # Check that the year filter produced a non-empty DataFrame. If it is empty, the requested year range is not in the file.
  293      if df.empty:
  294          print(f"Warning: no data found in '{file}' for variable '{variable}' between years {start_year} and {end_year}. "
  295                f"Check that start_year and end_year are within the range of the file.")
  296          return (pd.DataFrame(), None) if return_column else pd.DataFrame()
  297
  298      # Find the column label for the variable. Warn clearly if it cannot be matched rather than passing None as a column name downstream.
  299      column = get_matching_column_in_dataframe(df, variable)
  300      if column is None:
  301          available = list(df.columns)
  302          print(f"Warning: variable '{variable}' not found in '{file}'. Available columns: {available}")
  303          return (pd.DataFrame(), None) if return_column else pd.DataFrame()
  304      if 'Month' in df.columns:
  305          df = df[['Year', 'Month', column]]
  306      else:
  307          df = df[['Year', column]]
  308      df[column] *= multiplier
  309      if return_column:
  310          return df, column
  311      else:
  312          return df

  314  def read_file_set_into_single_variable_dataframe(output_files, file_set_index, variable, start_year, end_year, multiplier):
  315      """
  316      Reads a list of output files and the time series data for one specific variable in those files into a Pandas DataFrame.
  317
  318      Parameters:
  319          output_files: List of lists (2D matrix) for the files. We want to read in one column of this matrix, which corresponds to a particular group
  320                        of files that we want to collect together into a set and later plot the various set means of a larger ensemble of sets.
  321          file_set_index: The column number for the output_files matrix. This indicates the index for the group/set of files that are of interest.
  322          variable: The variable of interest.
  323          start_year: First year in the extracted time series data.
  324          end_year: Last year in the extracted time series data.
  325          multiplier: Multiplier on the time series (in case the user wants to change the units).
  326
  327      Returns:
  328          DataFrame containing data from all files for the variable and also all DataFrame columns for the variable, one column from each of the files.
  329      """
  330      file = output_files[0][file_set_index]
  331      num_files_in_set = len(output_files)
  332      df, column = read_file_into_single_variable_dataframe(file, variable, start_year, end_year, multiplier, return_column=True)
  333      # Rename the columns before doing the join operation to avoid naming conflicts for the various columns that correspond to the variable.
  334      df = df.rename(columns={column: column+f'_0_{file_set_index}'})
  335      for file_index in range(1, num_files_in_set):
  336          file = output_files[file_index][file_set_index]
  337          df_for_this_file = read_file_into_single_variable_dataframe(file, variable, start_year, end_year, multiplier)
  338          df_for_this_file = df_for_this_file.rename(columns={column: column+f'_{file_index}_{file_set_index}'})
  339          if 'Month' in df.columns:
  340              df = pd.merge(df, df_for_this_file, on=['Year', 'Month'], how='inner')
  341          else:
  342              df = pd.merge(df, df_for_this_file, on=['Year'], how='inner')
  343      # After the join operation, there will be multiple columns for the variable. Return the list of all of these columns along with the DataFrame.
  344      columns = get_matching_column_in_dataframe(df, variable, get_all_matches=True)
  345
  346      # Warn if the inner merge dropped any rows, which would indicate inconsistent year coverage across files in this set.
  347      expected_rows = len(read_file_into_dataframe(output_files[0][file_set_index]))
  348      if len(df) < expected_rows:
  349          print(f"Warning: file set {file_set_index} has inconsistent year coverage across ensemble members — "
  350                f"{expected_rows - len(df)} rows were dropped during the merge. "
  351                f"Check that all files cover the same year range.")
  352      return df, columns

  354  def add_year_range_to_plot_name(plot_name, variable, start_year, end_year, suffix=None):
  355      """
  356      Return a plot name with year tags inserted after the variable name.
  357
  358      Examples:
  359          plots/time_series_ZCO2.png -> plots/time_series_ZCO2_2015-2100.png
  360          plots/time_series_ZCO2.png with suffix='monthly' -> plots/time_series_ZCO2_2015-2100_monthly.png
  361          plots/co2_plot.png -> plots/co2_plot_2015-2100.png when the variable name is not in the filename
  362      """
  363      year_tag = f"{start_year}-{end_year}"
  364
  365      # Split the path so the directory and file extension are preserved while only the filename stem is modified.
  366      directory, filename = os.path.split(plot_name)
  367      stem, extension = os.path.splitext(filename)
  368
  369      # If the variable name is already part of the filename, insert the year range immediately after it.
  370      # Otherwise, append the year range to the end of the stem.
  371      if variable in stem:
  372          insert_index = stem.find(variable) + len(variable)
  373          stem = f"{stem[:insert_index]}_{year_tag}{stem[insert_index:]}"
  374      else:
  375          stem = f"{stem}_{year_tag}"
  376
  377      # Add an optional plot-type suffix, such as monthly or seasons, before restoring the extension.
  378      if suffix:
  379          stem = f"{stem}_{suffix}"
  380
  381      updated_name = stem + extension
  382      return os.path.join(directory, updated_name) if directory else updated_name
