    4  def calculate_mean_and_std_of_da_list(da_list, calculate_std=False):
    5      """
    6      Calculates the mean of a list of xarray DataArrays and optionally also the standard deviation of this list.
    7
    8      Parameters:
    9          da_list: A list of DataArrays with the same dimensions.
   10
   11      Returns:
   12          A DataArray containing the mean of the input DataArrays and optionally also the standard deviation of this list.
   13      """
   14      if not all(isinstance(da, xr.DataArray) for da in da_list):
   15          raise TypeError('All elements in the list must be xarray DataArrays.')
   16
   17      if not all(da.dims == da_list[0].dims for da in da_list[1:]):
   18          raise ValueError('All DataArrays must have the same dimensions.')
   19
   20      # Stack the DataArrays along a new dimension.
   21      stacked_da = xr.concat(da_list, dim='new_dim')
   22
   23      # Calculate the mean along the new dimension and the standard deviation after reducing the dataArrays along all dimensions.
   24      mean_da = stacked_da.mean(dim='new_dim')
   25      if calculate_std:
   26          std = stacked_da.std().item()
   27          return mean_da, std
   28      else:
   29          return mean_da

   31  def calculate_statistics_of_xarray(data, variable=None, weights=None):
   32      """
   33      Calculates the min, mean, median, max, and the standard deviation of an xarray or uxarray object (DataArray, Dataset, or uxarray data structure).
   34
   35      Parameters:
   36          data: Object of type xarray or uxarray whose statistical properties we want to calculate.
   37          variable: Variable of interest, used in the case of an xarray Dataset or uxarray data structure (UxDataArray or UxDataset).
   38          weights: Optional xarray DataArray of weights to use when calculating the mean.
   39
   40      Returns:
   41          min, mean, median, max, and standard deviation of the xarray or uxarray object.
   42      """
   43      if isinstance(data, xr.DataArray):
   44          min = data.min().item()
   45          if weights is not None:
   46              aligned_data, aligned_weights = xr.align(data, weights, join='inner')
   47              valid_weights = aligned_weights.where(aligned_data.notnull())
   48              weight_sum = valid_weights.sum()
   49              if weight_sum.item() == 0:
   50                  mean = aligned_data.mean().item()
   51              else:
   52                  mean = ((aligned_data * valid_weights).sum()/weight_sum).item()
   53          else:
   54              mean = data.mean().item()
   55          median = data.median().item()
   56          max = data.max().item()
   57          std = data.std().item()
   58      else:
   59          min = float(data[variable].min())
   60          mean = float(data[variable].mean())
   61          median = float(data[variable].median())
   62          max = float(data[variable].max())
   63          std = float(data[variable].std())
   64      return min, mean, median, max, std

   66  def convert_xarray_to_uxarray(data, grid, variable=None, fillna=1):
   67      """
   68      Converts an xarray object (DataArray or Dataset) into an uxarray object (UxDataArray or UxDataset).
   69
   70      Parameters:
   71          data: Object of type xarray.
   72          grid: Grid object for the unstructured grid contained within the uxarray object.
   73          variable: Variable of interest.
   74          fillna: Value to fill NaNs.
   75
   76      Returns:
   77          uxarray (UxDataArray or UxDataset) version of the given xarray object.
   78      """
   79      ds = xr.Dataset()
   80      if variable:
   81          ds[variable] = data
   82      else:
   83          ds = data
   84      # Fill NaNs while the data are still a plain xarray Dataset. Calling fillna after constructing
   85      # the UxDataset can hit uxarray/xarray >= 2025.7.1 constructor incompatibilities.
   86      if fillna:
   87          ds = ds.fillna(fillna)
   88      # Rename the grid-cell dimension (for example, ncol) to uxarray's face dimension name.
   89      # Direct UxDataset construction does not do this mapping automatically, and uxarray expects
   90      # face-centered plot data to use the n_face dimension.
   91      if hasattr(grid, 'n_face'):
   92          for dim in list(ds.dims):
   93              if dim not in ('time', 'year', 'lev', 'ilev') and ds.sizes[dim] == grid.n_face:
   94                  ds = ds.rename({dim: 'n_face'})
   95                  break
   96      return ux.UxDataset(dict(ds), uxgrid=grid)
