   23  def abbreviate_label(label, max_length=30):
   24      """Return an abbreviated version of label if it exceeds max_length characters."""
   25      if len(label) <= max_length:
   26          return label
   27      # Do not abbreviate labels that contain LaTeX/math text; slicing those
   28      # strings can produce invalid markup such as an unmatched "$".
   29      if '$' in label:
   30          return label
   31      words = label.replace('_', ' ').split()
   32      abbreviated_words = [w[:4] + '.' if len(w) > 5 else w for w in words]
   33      result = ' '.join(abbreviated_words)
   34      if len(result) > max_length:
   35          result = result[:max_length].rstrip() + '...'
   36      return result

   38  def label_already_contains_units(label, units):
   39      """Return True if the user-supplied label already includes the units text."""
   40      if not label or not units:
   41          return False
   42      # Compare a simplified text form so labels such as "thousands km$^2$" are
   43      # recognized as containing CSV units such as "thous km2".
   44      def normalize(text):
   45          text = str(text).lower().replace('thousands', 'thous')
   46          return re.sub(r'[^a-z0-9%]+', '', text)
   47      return normalize(units) in normalize(label)

   69      """Return (handles, labels) for a legend section header with an optional blank spacer before it."""
   70      formatted = (r'$\mathbf{' + text.replace(' ', r'\ ').replace('_', r'\_') + r'}$'

   99      # one legend entry per line, build proxy handles that explain the visual

  102      # not visually distinguished.
  103      if isinstance(category_label, list) and col_unique_values:
  104          if len(col_unique_values) > 3:

  111              handles.extend(h); labels.extend(l)

  113                  if col_idx == 0:
  114                      handle = mlines.Line2D([], [], color=plot_colors[val_idx % len(plot_colors)],
  115                                             linestyle='-', linewidth=linewidth)

  123                  handles.append(handle); labels.append(str(val))

  126          h, l = _legend_section_header(col_name + ':', use_latex, bool(handles))
  127          handles.extend(h); labels.extend(l)
  128          for cat_idx, cat in enumerate(categories):
  129              color = plot_colors[cat_idx % len(plot_colors)] if color_by_category else 'gray'
  130              handle = mlines.Line2D([], [], color=color, linestyle='-',
  131                                      marker=markers[cat_idx % len(markers)],
  132                                      markersize=marker_size, linewidth=linewidth)
  133              handles.append(handle); labels.append(category_to_label_str(cat))
  134
  135      # Region legend entries are shown only for the single-column category case.
  136      # In the multi-column case, the category legend already uses the available
  137      # marker/line encodings, so adding regions would make the legend misleading.
  138      if not isinstance(category_label, list):
  139          all_regions = []
  140          for cat in categories:
  141              for region in regions_per_category.get(cat, ['Global']):
  142                  if region not in all_regions:

  152
  153      # Scenario entries describe line styles. For a two-scenario percent
  154      # difference plot, the reference scenario is not plotted, so there is no
  155      # separate scenario section to show.
  156      if multi_scen and not is_two_scen_pct_diff:

  158          handles.extend(h); labels.extend(l)
  159          start = 1 if plot_percent_difference else 0
  160          for scen_idx in range(start, num_scen):
  161              handle = mlines.Line2D([], [], color='gray',
  162                                      linestyle=linestyle_tuples[scen_idx % len(linestyle_tuples)][1],

  164              handles.append(handle); labels.append(str(scen_display[scen_idx]))

  166      if not handles:
  167          return

  169      if not legend_x_offset:
  170          legend_x_offset = 1.02
  171      kw = dict(prop={'size': legend_label_size}, frameon=False, ncol=legend_num_columns)
  172      if legend_place_outside:
  173          ax.legend(handles=handles, labels=labels, loc='center left',
  174                    bbox_to_anchor=(legend_x_offset, 0.5), **kw)
  175      else:
  176          ax.legend(handles=handles, labels=labels, loc='best', **kw)

  235  def process_inputs(inputs):
  236      """
  237      Processes a dictionary of inputs (keys are options, values are choices for those options) for creating time series plots.
  238
  239      Parameters:
  240          inputs: Dictionary containing the user plotting choice inputs for different options. This dictionary may be incomplete or have invalid values.
  241
  242      Returns:
  243          Dictionary that completely specifies all plotting options.
  244          If the user did not make a choice for a particular option, the default choice for that plotting option will be selected.
  245      """
  246      # Check that the output file exists and is non-empty before trying to read it.
  247      if not os.path.exists(inputs['output_file']):
  248          print(f"Warning: output file not found, skipping: '{inputs['output_file']}'")
  249          return None
  250      if os.path.getsize(inputs['output_file']) == 0:
  251          print(f"Warning: output file is empty, skipping: '{inputs['output_file']}'")
  252          return None
  253      df = read_file_into_dataframe(inputs['output_file'])
  254
  255      # If the category label (e.g., sector or landtype) has not been specified, use the default value.
  256      if 'category_label' not in inputs:
  257          category_label = default_inputs['category_label']
  258          inputs['category_label'] = category_label
  259      else:
  260          category_label = inputs['category_label']
  261
  262      # If the list of categories (e.g., sectors or landtypes) has not been
  263      # specified, populate it from the data. For a multi-column category_label,
  264      # each category is a tuple of values drawn from those columns.
  265      if 'categories' not in inputs:
  266          if isinstance(category_label, list) and all(c in df.columns for c in category_label):
  267              combos = df[category_label].drop_duplicates()
  268              inputs['categories'] = list(combos.itertuples(index=False, name=None))
  269              if 'categories_to_exclude' in inputs:
  270                  exclude = {tuple(e) if isinstance(e, list) else e for e in inputs['categories_to_exclude']}
  271                  inputs['categories'] = [c for c in inputs['categories'] if c not in exclude]
  272          elif not isinstance(category_label, list) and category_label in df.columns:
  273              inputs['categories'] = df[category_label].unique()
  274              if 'categories_to_exclude' in inputs:
  275                  inputs['categories'] = [column for column in inputs['categories'] if column not in inputs['categories_to_exclude']]
  276          else:
  277              inputs['categories'] = 'All'
  278      categories = inputs['categories']
  279      # A multi-column category can also be supplied as a dict, for example:
  280      # {"sector": ["crop"], "technology": ["Corn", "Wheat"]}. Expand that into
  281      # explicit tuples so the rest of the code can handle categories uniformly.
  282      if isinstance(categories, dict) and isinstance(category_label, list):
  283          per_column = [categories[col] for col in category_label]
  284          categories = list(itertools.product(*per_column))
  285          inputs['categories'] = categories
  286      # If the user entered a string indicating a single category, put that string in a list.
  287      elif isinstance(categories, str):
  288          categories = [categories]
  289          inputs['categories'] = categories
  290      elif isinstance(category_label, list):
  291          # Convert any list-type categories (from JSON) to tuples.
  292          categories = [tuple(c) if isinstance(c, list) else c for c in categories]
  293          inputs['categories'] = categories
  294
  295      # Create the plot directory if it does not already exist. By default, put
  296      # the p-value output file in the same directory as the plots.
  297      if 'plot_directory' not in inputs:
  298          inputs['plot_directory'] = default_inputs['plot_directory']
  299      if not os.path.exists(inputs['plot_directory']):
  300          os.makedirs(inputs['plot_directory'])
  301      if 'p_value_file' not in inputs:
  302          inputs['p_value_file'] = os.path.join(inputs['plot_directory'], default_inputs['p_value_file'])
  303
  304      # Use the output CSV name as the default y-axis label and plot name. Drop
  305      # "_processed" so generated plot labels read like user-facing quantities
  306      # rather than pipeline artifact names.
  307      index_of_last_backslash = inputs['output_file'].rfind('/')
  308      index_of_dot_csv = inputs['output_file'].find('.csv')
  309      if index_of_last_backslash == -1:
  310          output_file_name = inputs['output_file'][:index_of_dot_csv]
  311      else:
  312          output_file_name = inputs['output_file'][index_of_last_backslash+1:index_of_dot_csv]
  313      if 'y_label' not in inputs:
  314          inputs['y_label'] = output_file_name
  315      inputs['y_label'] = inputs['y_label'].replace('_processed', '')
  316      output_file_name_clean = output_file_name.replace('_processed', '')
  317      if 'plot_name' not in inputs:
  318          inputs['plot_name'] = os.path.join(inputs['plot_directory'], 'time_series_' + output_file_name_clean)
  319      elif 'plot_name' in inputs and '/' not in inputs['plot_name']:
  320          # If the user specified only a file name (no path) for the plot name, put the plot in the plot directory.
  321          inputs['plot_name'] = os.path.join(inputs['plot_directory'], inputs['plot_name'].replace('_processed', ''))
  322
  323      # Add keys for plotting options that have not been specified in the inputs dictionary and use default values for them.
  324      for key in default_inputs.keys():
  325          if key not in inputs:
  326              inputs[key] = default_inputs[key]
  327
  328      # If the scenarios have not been specified, use all the scenarios in the Pandas DataFrame.
  329      if 'scenarios' not in inputs:
  330          scenario_label = inputs['scenario_label']
  331          inputs['scenarios'] = df[scenario_label].unique()
  332
  333      # If the scenarios are specified as a list of lists (for ensemble plots), check if they need to be transposed.
  334      # Users can now specify scenarios in two formats:
  335      #   Format A (organized by ensemble member - original format):
  336      #       [["Control", "Full feedback"], ["Control_2", "Full feedback_2"], ...]
  337      #   Format B (organized by scenario set - new user-friendly format):
  338      #       [["Control", "Control_2", ...], ["Full feedback", "Full feedback_2", ...]]
  339      # The plotting functions expect Format A internally, so Format B will be automatically transposed.
  340      if check_is_list_of_lists(inputs['scenarios']):
  341          scenario_sets = inputs.get('scenario_sets', None)
  342          inputs['scenarios'], was_transposed = transpose_scenarios_if_needed(inputs['scenarios'], scenario_sets)
  343          if was_transposed and inputs['notify_scenarios_transposed']:
  344              print(f"Note: Scenarios were automatically transposed from 'organized by scenario set' format to 'organized by ensemble member' format.")
  345
  346      # Downstream loops expect regions to be indexed by category. If the user
  347      # supplied one region list for all categories, copy that list for every
  348      # category here.
  349      if 'regions' not in inputs or not isinstance(inputs['regions'], dict):
  350          inputs['regions'] = dict.fromkeys(categories, inputs['regions'])
  351
  352      # Check that the value_label column exists in the DataFrame.
  353      value_label = inputs['value_label']
  354      if value_label not in df.columns:
  355          raise KeyError(f"Error: value_label '{value_label}' not found in '{inputs['output_file']}'. "
  356                         f"Available columns: {list(df.columns)}")
  357
  358      # area_weighted_mean requires the processed CSV to include area metadata.
  359      # GCAM area columns are added during gcam_process_extracted_data.py using a
  360      # processed land-allocation file.
  361      if inputs['aggregation_type_in_each_year'] == 'area_weighted_mean' and 'area' not in df.columns:
  362          raise KeyError(f"Error: aggregation_type_in_each_year is 'area_weighted_mean' but no 'area' column "
  363                         f"found in '{inputs['output_file']}'. Consider using 'mean' instead, or add area data "
  364                         f"with gcam_process_extracted_data.py and a land_allocation_file entry.")
  365
  366      # Warn if any specified scenario is not found in the DataFrame.
  367      scenario_label = inputs['scenario_label']
  368      if scenario_label in df.columns:
  369          available_scenarios = set(df[scenario_label].unique())
  370          specified_scenarios = inputs['scenarios']
  371          if check_is_list_of_lists(specified_scenarios):
  372              flat_scenarios = [s for member in specified_scenarios for s in member]
  373          else:
  374              flat_scenarios = list(specified_scenarios)
  375          missing_scenarios = [s for s in flat_scenarios if s not in available_scenarios]
  376          if missing_scenarios:
  377              print(f"Warning: the following scenarios were not found in '{inputs['output_file']}' "
  378                    f"and will produce empty plots: {missing_scenarios}")
  379
  380      # Extract units from the CSV's units column (first non-null, non-NA value).
  381      # Users can override this with a literal "units" entry in the JSON.
  382      units_label = inputs['units_label']
  383      if 'units' not in inputs:
  384          if units_label in df.columns:
  385              units_values = df[units_label].dropna().unique()
  386              units_values = [u for u in units_values if str(u).strip().upper() not in ('NA', 'NONE', '')]
  387              inputs['units'] = str(units_values[0]) if units_values else None
  388          else:
  389              inputs['units'] = None
  390
  391      # Set the plot title to the full y_label if not specified by the user.
  392      if inputs['title'] is None:
  393          inputs['title'] = inputs['y_label']
  394
  395      # Keep the full quantity name in the title, but abbreviate long y-axis
  396      # labels so they fit better in dense time-series plots.
  397      if inputs['y_label_abbreviated'] is None:
  398          inputs['y_label_abbreviated'] = abbreviate_label(inputs['y_label'])
  399
  400      return inputs

  873                      if not plot_percent_difference or scenario_set_index != 0:
  874                          ax.plot(x, y, label=label, color=line_color, linestyle=linestyle, linewidth=plot_linewidth, marker=marker, markersize=marker_size)
  875                          if std_multiplier:
  876                              error = y_std*std_multiplier
  877                              ax.fill_between(x, y-error, y+error, color=line_color, alpha=error_bars_alpha)
  878
  879                      if num_scenario_sets > 1 and scenario_set_index > 0:
  880                          # Compare the whole control set-mean time series against
  881                          # the current set-mean time series. Require at least two
  882                          # numeric points in each series to avoid scipy warnings
  883                          # on empty or single-point inputs.
  884                          control_data = df_all_set_means[f'set=0_cat={category_index}_reg={region_index}']
  885                          test_data = df_all_set_means[f'set={scenario_set_index}_cat={category_index}_reg={region_index}']
  886                          control_num = pd.to_numeric(control_data, errors='coerce').dropna()
  887                          test_num = pd.to_numeric(test_data, errors='coerce').dropna()
  888                          if len(control_num) >= 2 and len(test_num) >= 2:
  889                              with warnings.catch_warnings(record=True) as w:
  890                                  warnings.simplefilter('always', RuntimeWarning)
  891                                  ttest = stats.ttest_ind(control_num, test_num)
  892                                  if w:
  893                                      print(f'Warning (t-test) in {plot_name}: {w[0].message}')
  894                              label = f'set={scenario_set_index}, category={category_to_label_str(category)}, region={region}'
  895                              print_p_values(ttest, label, p_value_threshold, p_value_file, plot_name, p_value_file_print_only_if_below_threshold)
  896                      elif num_scenario_sets == 1 and num_regions > 1:
  897                          # If there is only one scenario set, but multiple regions for that set, perform inter-regional t-tests for each category.
  898                          control_data = df_all_set_means[f'set=0_cat={category_index}_reg=0']
  899                          test_data = df_all_set_means[f'set=0_cat={category_index}_reg={region_index}']
  900                          control_num = pd.to_numeric(control_data, errors='coerce').dropna()
  901                          test_num = pd.to_numeric(test_data, errors='coerce').dropna()
  902                          if len(control_num) >= 2 and len(test_num) >= 2:
  903                              with warnings.catch_warnings(record=True) as w:
  904                                  warnings.simplefilter('always', RuntimeWarning)
  905                                  ttest = stats.ttest_ind(control_num, test_num)
  906                                  if w:
  907                                      print(f'Warning (t-test) in {plot_name}: {w[0].message}')
  908                              label = f'category={category_to_label_str(category)}, region={region}'
  909                              print_p_values(ttest, label, p_value_threshold, p_value_file, plot_name, p_value_file_print_only_if_below_threshold)
  910
  911                      # Also compare ensemble members year by year. Significant
  912                      # years are marked directly on the plotted curve.
  913                      if num_scenario_sets > 1 and scenario_set_index > 0:
  914                          columns_control_set = [column for column in df_all_time_series.columns if
  915                                          (f'set=0' in column and f'cat={category_index}_reg={region_index}' in column)]
  916                          columns_this_set = [column for column in df_all_time_series.columns if
  917                                          (f'set={scenario_set_index}' in column and f'cat={category_index}_reg={region_index}' in column)]
  918                          p_values = df_all_time_series.apply(perform_ttest, columns_set_1=columns_control_set, \
  919                                                                 columns_set_2=columns_this_set, axis=1).fillna(1)
  920                          # If the p-value is below the threshold, add a marker to the plot at that time period (year).
  921                          mask = p_values <= p_value_threshold
  922                          ax.plot(df_all_time_series['Year'][mask], y_series[mask], color=line_color, linestyle='None', linewidth=linewidth, \
  923                                    marker=marker, markersize=p_value_marker_size)
  924
  925                  # Optionally include the overall mean across all scenario sets and error bars indicating the standard deviation for group/set means.
  926                  if include_mean_across_all_data:
  927                      if plot_percent_difference:
  928                      # If plotting a percent difference, do not include the first scenario set in calculating the overall mean.
  929                          columns = [column for column in df_all_set_means.columns if column != 'Year' \
  930                                     and ('set=0' not in column and f'cat={category_index}_reg={region_index}' in column)]
  931                      else:
  932                          columns = [column for column in df_all_set_means.columns if column != 'Year' \
  933                                     and f'cat={category_index}_reg={region_index}' in column]
  934                      x = df_all_set_means['Year']
  935                      y = df_all_set_means[columns].mean(axis=1)
  936                      label = 'Mean'
  937                      if num_categories > 1 and num_regions > 1:
  938                          label += f' ({category_to_label_str(category)}_{region})'
  939                      elif num_categories > 1:
  940                          label += f' ({category_to_label_str(category)})'
  941                      elif num_regions > 1:
  942                          label += f' ({region})'
  943                      ax.plot(x, y, label=label, color='k', linestyle=linestyle, linewidth=linewidth, marker=marker, markersize=marker_size)
  944                      if std_mean_across_all_data_multiplier:
  945                          error = df_all_set_means[columns].std(axis=1)*std_mean_across_all_data_multiplier
  946                          ax.fill_between(x, y-error, y+error, color='k', alpha=error_bars_alpha)
  947
  948      # Finalize the time series plot now that all curves have been processed.
  949      # The custom grouped legend has already been attached, so disable the
  950      # generic legend behavior inside set_figure_options.
  951      plot_options['name'] = plot_name
  952      if legend_on:
  953          build_grouped_legend(ax, categories, scenarios, scenario_sets, inputs['regions'],
  954                                plot_colors, markers, linestyle_tuples, marker_size, linewidth,
  955                                category_label, col_unique_values, plot_percent_difference, plot_options)
  956      set_figure_options(fig, ax, {**plot_options, 'legend_on': False})
  957
  958      # Close the plot now that we are done with it. Record the elapsed time.
  959      plt.close(fig)
  960      end_time = time.time()
  961      elapsed_time = end_time - start_time
