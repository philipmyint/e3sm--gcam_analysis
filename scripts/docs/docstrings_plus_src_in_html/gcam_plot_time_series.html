   22  def abbreviate_label(label, max_length=30):
   23      """Return an abbreviated version of label if it exceeds max_length characters."""
   24      if len(label) <= max_length:
   25          return label
   26      words = label.replace('_', ' ').split()
   27      abbreviated_words = [w[:4] + '.' if len(w) > 5 else w for w in words]
   28      result = ' '.join(abbreviated_words)
   29      if len(result) > max_length:
   30          result = result[:max_length].rstrip() + '...'
   31      return result

   40      """Filter df to rows matching category, supporting a single-column string or multi-column list."""
   41      if isinstance(category_label, list):
   42          mask = pd.Series(True, index=df.index)

   49      """Return (handles, labels) for a legend section header with an optional blank spacer before it."""
   50      formatted = (r'$\mathbf{' + text.replace(' ', r'\ ').replace('_', r'\_') + r'}$'

   75      multi_scen = num_scen > 1
   76      color_by_category = True
   77

   79      # one legend entry per line, build proxy handles that explain the visual
   80      # encoding: first category column -> color, second -> marker, third ->
   81      # linewidth. Additional category columns are still filtered, but they are
   82      # not visually distinguished.
   83      if isinstance(category_label, list) and col_unique_values:
   84          if len(col_unique_values) > 3:
   85              print(f"Warning: only 3 category columns are visually encoded (color, marker, linewidth). "
   86                    f"Columns beyond the third will not be visually distinguished: {category_label[3:]}")
   87          for col_idx, col_name in enumerate(category_label):
   88              if col_idx >= 3:  # color (0), marker (1), linewidth (2) exhaust available encodings
   89                  continue

   91              handles.extend(h); labels.extend(l)
   92              for val_idx, val in enumerate(col_unique_values[col_idx]):
   93                  if col_idx == 0:
   94                      handle = mlines.Line2D([], [], color=plot_colors[val_idx % len(plot_colors)],
   95                                             linestyle='-', linewidth=linewidth)
   96                  elif col_idx == 1:

   98                                             marker=markers[val_idx % len(markers)],
   99                                             markersize=marker_size, linewidth=linewidth)

  101                      handle = mlines.Line2D([], [], color='gray', linestyle='-',
  102                                             linewidth=linewidth * (val_idx + 1))
  103                  handles.append(handle); labels.append(str(val))
  104      elif len(categories) > 1 or color_by_category:

  106          h, l = _legend_section_header(col_name + ':', use_latex, bool(handles))
  107          handles.extend(h); labels.extend(l)
  108          for cat_idx, cat in enumerate(categories):
  109              color = plot_colors[cat_idx % len(plot_colors)] if color_by_category else 'gray'
  110              handle = mlines.Line2D([], [], color=color, linestyle='-',
  111                                      marker=markers[cat_idx % len(markers)],
  112                                      markersize=marker_size, linewidth=linewidth)
  113              handles.append(handle); labels.append(category_to_label_str(cat))
  114
  115      # Region legend entries are shown only for the single-column category case.
  116      # In the multi-column case, the category legend already uses the available
  117      # marker/line encodings, so adding regions would make the legend misleading.
  118      if not isinstance(category_label, list):
  119          all_regions = []
  120          for cat in categories:
  121              for region in regions_per_category.get(cat, ['Global']):
  122                  if region not in all_regions:
  123                      all_regions.append(region)

  128                  handle = mlines.Line2D([], [], color='gray',
  129                                          linestyle=linestyle_tuples[reg_idx % len(linestyle_tuples)][1],
  130                                          linewidth=linewidth)
  131                  handles.append(handle); labels.append(region)
  132
  133      # Scenario entries describe line styles. For a two-scenario percent
  134      # difference plot, the reference scenario is not plotted, so there is no
  135      # separate scenario section to show.
  136      if multi_scen and not is_two_scen_pct_diff:
  137          h, l = _legend_section_header('Scenarios:', use_latex, bool(handles))
  138          handles.extend(h); labels.extend(l)
  139          start = 1 if plot_percent_difference else 0
  140          for scen_idx in range(start, num_scen):
  141              handle = mlines.Line2D([], [], color='gray',
  142                                      linestyle=linestyle_tuples[scen_idx % len(linestyle_tuples)][1],
  143                                      linewidth=linewidth)
  144              handles.append(handle); labels.append(str(scen_display[scen_idx]))

  146      if not handles:

  148
  149      if not legend_x_offset:
  150          legend_x_offset = 1.02
  151      kw = dict(prop={'size': legend_label_size}, frameon=False, ncol=legend_num_columns)
  152      if legend_place_outside:
  153          ax.legend(handles=handles, labels=labels, loc='center left',
  154                    bbox_to_anchor=(legend_x_offset, 0.5), **kw)
  155      else:
  156          ax.legend(handles=handles, labels=labels, loc='best', **kw)

  215  def process_inputs(inputs):
  216      """
  217      Processes a dictionary of inputs (keys are options, values are choices for those options) for creating time series plots.
  218
  219      Parameters:
  220          inputs: Dictionary containing the user plotting choice inputs for different options. This dictionary may be incomplete or have invalid values.
  221
  222      Returns:
  223          Dictionary that completely specifies all plotting options.
  224          If the user did not make a choice for a particular option, the default choice for that plotting option will be selected.
  225      """
  226      # Check that the output file exists and is non-empty before trying to read it.
  227      if not os.path.exists(inputs['output_file']):
  228          print(f"Warning: output file not found, skipping: '{inputs['output_file']}'")
  229          return None
  230      if os.path.getsize(inputs['output_file']) == 0:
  231          print(f"Warning: output file is empty, skipping: '{inputs['output_file']}'")
  232          return None
  233      df = read_file_into_dataframe(inputs['output_file'])
  234
  235      # If the category label (e.g., sector or landtype) has not been specified, use the default value.
  236      if 'category_label' not in inputs:
  237          category_label = default_inputs['category_label']
  238          inputs['category_label'] = category_label
  239      else:
  240          category_label = inputs['category_label']
  241
  242      # If the list of categories (e.g., sectors or landtypes) has not been
  243      # specified, populate it from the data. For a multi-column category_label,
  244      # each category is a tuple of values drawn from those columns.
  245      if 'categories' not in inputs:
  246          if isinstance(category_label, list) and all(c in df.columns for c in category_label):
  247              combos = df[category_label].drop_duplicates()
  248              inputs['categories'] = list(combos.itertuples(index=False, name=None))
  249              if 'categories_to_exclude' in inputs:
  250                  exclude = {tuple(e) if isinstance(e, list) else e for e in inputs['categories_to_exclude']}
  251                  inputs['categories'] = [c for c in inputs['categories'] if c not in exclude]
  252          elif not isinstance(category_label, list) and category_label in df.columns:
  253              inputs['categories'] = df[category_label].unique()
  254              if 'categories_to_exclude' in inputs:
  255                  inputs['categories'] = [column for column in inputs['categories'] if column not in inputs['categories_to_exclude']]
  256          else:
  257              inputs['categories'] = 'All'
  258      categories = inputs['categories']
  259      # A multi-column category can also be supplied as a dict, for example:
  260      # {"sector": ["crop"], "technology": ["Corn", "Wheat"]}. Expand that into
  261      # explicit tuples so the rest of the code can handle categories uniformly.
  262      if isinstance(categories, dict) and isinstance(category_label, list):
  263          per_column = [categories[col] for col in category_label]
  264          categories = list(itertools.product(*per_column))
  265          inputs['categories'] = categories
  266      # If the user entered a string indicating a single category, put that string in a list.
  267      elif isinstance(categories, str):
  268          categories = [categories]
  269          inputs['categories'] = categories
  270      elif isinstance(category_label, list):
  271          # Convert any list-type categories (from JSON) to tuples.
  272          categories = [tuple(c) if isinstance(c, list) else c for c in categories]
  273          inputs['categories'] = categories
  274
  275      # Create the plot directory if it does not already exist. By default, put
  276      # the p-value output file in the same directory as the plots.
  277      if 'plot_directory' not in inputs:
  278          inputs['plot_directory'] = default_inputs['plot_directory']
  279      if not os.path.exists(inputs['plot_directory']):
  280          os.makedirs(inputs['plot_directory'])
  281      if 'p_value_file' not in inputs:
  282          inputs['p_value_file'] = os.path.join(inputs['plot_directory'], default_inputs['p_value_file'])
  283
  284      # Use the output CSV name as the default y-axis label and plot name. Drop
  285      # "_processed" so generated plot labels read like user-facing quantities
  286      # rather than pipeline artifact names.
  287      index_of_last_backslash = inputs['output_file'].rfind('/')
  288      index_of_dot_csv = inputs['output_file'].find('.csv')
  289      if index_of_last_backslash == -1:
  290          output_file_name = inputs['output_file'][:index_of_dot_csv]
  291      else:
  292          output_file_name = inputs['output_file'][index_of_last_backslash+1:index_of_dot_csv]
  293      if 'y_label' not in inputs:
  294          inputs['y_label'] = output_file_name
  295      inputs['y_label'] = inputs['y_label'].replace('_processed', '')
  296      output_file_name_clean = output_file_name.replace('_processed', '')
  297      if 'plot_name' not in inputs:
  298          inputs['plot_name'] = os.path.join(inputs['plot_directory'], 'time_series_' + output_file_name_clean)
  299      elif 'plot_name' in inputs and '/' not in inputs['plot_name']:
  300          # If the user specified only a file name (no path) for the plot name, put the plot in the plot directory.
  301          inputs['plot_name'] = os.path.join(inputs['plot_directory'], inputs['plot_name'].replace('_processed', ''))
  302
  303      # Add keys for plotting options that have not been specified in the inputs dictionary and use default values for them.
  304      for key in default_inputs.keys():
  305          if key not in inputs:
  306              inputs[key] = default_inputs[key]
  307
  308      # If the scenarios have not been specified, use all the scenarios in the Pandas DataFrame.
  309      if 'scenarios' not in inputs:
  310          scenario_label = inputs['scenario_label']
  311          inputs['scenarios'] = df[scenario_label].unique()
  312
  313      # If the scenarios are specified as a list of lists (for ensemble plots), check if they need to be transposed.
  314      # Users can now specify scenarios in two formats:
  315      #   Format A (organized by ensemble member - original format):
  316      #       [["Control", "Full feedback"], ["Control_2", "Full feedback_2"], ...]
  317      #   Format B (organized by scenario set - new user-friendly format):
  318      #       [["Control", "Control_2", ...], ["Full feedback", "Full feedback_2", ...]]
  319      # The plotting functions expect Format A internally, so Format B will be automatically transposed.
  320      if check_is_list_of_lists(inputs['scenarios']):
  321          scenario_sets = inputs.get('scenario_sets', None)
  322          inputs['scenarios'], was_transposed = transpose_scenarios_if_needed(inputs['scenarios'], scenario_sets)
  323          if was_transposed and inputs['notify_scenarios_transposed']:
  324              print(f"Note: Scenarios were automatically transposed from 'organized by scenario set' format to 'organized by ensemble member' format.")
  325
  326      # Downstream loops expect regions to be indexed by category. If the user
  327      # supplied one region list for all categories, copy that list for every
  328      # category here.
  329      if 'regions' not in inputs or not isinstance(inputs['regions'], dict):
  330          inputs['regions'] = dict.fromkeys(categories, inputs['regions'])
  331
  332      # Check that the value_label column exists in the DataFrame.
  333      value_label = inputs['value_label']
  334      if value_label not in df.columns:
  335          raise KeyError(f"Error: value_label '{value_label}' not found in '{inputs['output_file']}'. "
  336                         f"Available columns: {list(df.columns)}")
  337
  338      # area_weighted_mean requires the processed CSV to include area metadata.
  339      # GCAM area columns are added during gcam_process_extracted_data.py using a
  340      # processed land-allocation file.
  341      if inputs['aggregation_type_in_each_year'] == 'area_weighted_mean' and 'area' not in df.columns:
  342          raise KeyError(f"Error: aggregation_type_in_each_year is 'area_weighted_mean' but no 'area' column "
  343                         f"found in '{inputs['output_file']}'. Consider using 'mean' instead, or add area data "
  344                         f"with gcam_process_extracted_data.py and a land_allocation_file entry.")
  345
  346      # Warn if any specified scenario is not found in the DataFrame.
  347      scenario_label = inputs['scenario_label']
  348      if scenario_label in df.columns:
  349          available_scenarios = set(df[scenario_label].unique())
  350          specified_scenarios = inputs['scenarios']
  351          if check_is_list_of_lists(specified_scenarios):
  352              flat_scenarios = [s for member in specified_scenarios for s in member]
  353          else:
  354              flat_scenarios = list(specified_scenarios)
  355          missing_scenarios = [s for s in flat_scenarios if s not in available_scenarios]
  356          if missing_scenarios:
  357              print(f"Warning: the following scenarios were not found in '{inputs['output_file']}' "
  358                    f"and will produce empty plots: {missing_scenarios}")
  359
  360      # Extract units from the CSV's units column (first non-null, non-NA value).
  361      # Users can override this with a literal "units" entry in the JSON.
  362      units_label = inputs['units_label']
  363      if 'units' not in inputs:
  364          if units_label in df.columns:
  365              units_values = df[units_label].dropna().unique()
  366              units_values = [u for u in units_values if str(u).strip().upper() not in ('NA', 'NONE', '')]
  367              inputs['units'] = str(units_values[0]) if units_values else None
  368          else:
  369              inputs['units'] = None
  370
  371      # Set the plot title to the full y_label if not specified by the user.
  372      if inputs['title'] is None:
  373          inputs['title'] = inputs['y_label']
  374
  375      # Keep the full quantity name in the title, but abbreviate long y-axis
  376      # labels so they fit better in dense time-series plots.
  377      if inputs['y_label_abbreviated'] is None:
  378          inputs['y_label_abbreviated'] = abbreviate_label(inputs['y_label'])
  379
  380      return inputs
