   23  def abbreviate_label(label, max_length=30):
   24      """Return an abbreviated version of label if it exceeds max_length characters."""
   25      if len(label) <= max_length:
   26          return label
   27      # Do not abbreviate labels that contain LaTeX/math text; slicing those
   28      # strings can produce invalid markup such as an unmatched "$".
   29      if '$' in label:
   30          return label
   31      words = label.replace('_', ' ').split()
   32      abbreviated_words = [w[:4] + '.' if len(w) > 5 else w for w in words]
   33      result = ' '.join(abbreviated_words)
   34      if len(result) > max_length:
   35          result = result[:max_length].rstrip() + '...'
   36      return result

   38  def label_already_contains_units(label, units):
   39      """Return True if the user-supplied label already includes the units text."""
   40      if not label or not units:
   41          return False
   42      # Compare a simplified text form so labels such as "thousands km$^2$" are
   43      # recognized as containing CSV units such as "thous km2".
   44      def normalize(text):
   45          text = str(text).lower().replace('thousands', 'thous')
   46          return re.sub(r'[^a-z0-9%]+', '', text)
   47      return normalize(units) in normalize(label)

  231  def process_inputs(inputs):
  232      """
  233      Processes a dictionary of inputs (keys are options, values are choices for those options) for creating time series plots.
  234
  235      Parameters:
  236          inputs: Dictionary containing the user plotting choice inputs for different options. This dictionary may be incomplete or have invalid values.
  237
  238      Returns:
  239          Dictionary that completely specifies all plotting options.
  240          If the user did not make a choice for a particular option, the default choice for that plotting option will be selected.
  241      """
  242      # Check that the output file exists and is non-empty before trying to read it.
  243      if not os.path.exists(inputs['output_file']):
  244          print(f"Warning: output file not found, skipping: '{inputs['output_file']}'")
  245          return None
  246      if os.path.getsize(inputs['output_file']) == 0:
  247          print(f"Warning: output file is empty, skipping: '{inputs['output_file']}'")
  248          return None
  249      df = read_file_into_dataframe(inputs['output_file'])
  250
  251      # If the category label (e.g., sector or landtype) has not been specified, use the default value.
  252      if 'category_label' not in inputs:
  253          category_label = default_inputs['category_label']
  254          inputs['category_label'] = category_label
  255      else:
  256          category_label = inputs['category_label']
  257
  258      # If the list of categories (e.g., sectors or landtypes) has not been
  259      # specified, populate it from the data. For a multi-column category_label,
  260      # each category is a tuple of values drawn from those columns.
  261      if 'categories' not in inputs:
  262          if isinstance(category_label, list) and all(c in df.columns for c in category_label):
  263              combos = df[category_label].drop_duplicates()
  264              inputs['categories'] = list(combos.itertuples(index=False, name=None))
  265              if 'categories_to_exclude' in inputs:
  266                  exclude = {tuple(e) if isinstance(e, list) else e for e in inputs['categories_to_exclude']}
  267                  inputs['categories'] = [c for c in inputs['categories'] if c not in exclude]
  268          elif not isinstance(category_label, list) and category_label in df.columns:
  269              inputs['categories'] = df[category_label].unique()
  270              if 'categories_to_exclude' in inputs:
  271                  inputs['categories'] = [column for column in inputs['categories'] if column not in inputs['categories_to_exclude']]
  272          else:
  273              inputs['categories'] = 'All'
  274      categories = inputs['categories']
  275      # A multi-column category can also be supplied as a dict, for example:
  276      # {"sector": ["crop"], "technology": ["Corn", "Wheat"]}. Expand that into
  277      # explicit tuples so the rest of the code can handle categories uniformly.
  278      if isinstance(categories, dict) and isinstance(category_label, list):
  279          per_column = [categories[col] for col in category_label]
  280          categories = list(itertools.product(*per_column))
  281          inputs['categories'] = categories
  282      # If the user entered a string indicating a single category, put that string in a list.
  283      elif isinstance(categories, str):
  284          categories = [categories]
  285          inputs['categories'] = categories
  286      elif isinstance(category_label, list):
  287          # Convert any list-type categories (from JSON) to tuples.
  288          categories = [tuple(c) if isinstance(c, list) else c for c in categories]
  289          inputs['categories'] = categories
  290
  291      # Create the plot directory if it does not already exist. By default, put
  292      # the p-value output file in the same directory as the plots.
  293      if 'plot_directory' not in inputs:
  294          inputs['plot_directory'] = default_inputs['plot_directory']
  295      if not os.path.exists(inputs['plot_directory']):
  296          os.makedirs(inputs['plot_directory'])
  297      if 'p_value_file' not in inputs:
  298          inputs['p_value_file'] = os.path.join(inputs['plot_directory'], default_inputs['p_value_file'])
  299
  300      # Use the output CSV name as the default y-axis label and plot name. Drop
  301      # "_processed" so generated plot labels read like user-facing quantities
  302      # rather than pipeline artifact names.
  303      index_of_last_backslash = inputs['output_file'].rfind('/')
  304      index_of_dot_csv = inputs['output_file'].find('.csv')
  305      if index_of_last_backslash == -1:
  306          output_file_name = inputs['output_file'][:index_of_dot_csv]
  307      else:
  308          output_file_name = inputs['output_file'][index_of_last_backslash+1:index_of_dot_csv]
  309      if 'y_label' not in inputs:
  310          inputs['y_label'] = output_file_name
  311      inputs['y_label'] = inputs['y_label'].replace('_processed', '')
  312      output_file_name_clean = output_file_name.replace('_processed', '')
  313      if 'plot_name' not in inputs:
  314          inputs['plot_name'] = os.path.join(inputs['plot_directory'], 'time_series_' + output_file_name_clean)
  315      elif 'plot_name' in inputs and '/' not in inputs['plot_name']:
  316          # If the user specified only a file name (no path) for the plot name, put the plot in the plot directory.
  317          inputs['plot_name'] = os.path.join(inputs['plot_directory'], inputs['plot_name'].replace('_processed', ''))
  318
  319      # Add keys for plotting options that have not been specified in the inputs dictionary and use default values for them.
  320      for key in default_inputs.keys():
  321          if key not in inputs:
  322              inputs[key] = default_inputs[key]
  323
  324      # If the scenarios have not been specified, use all the scenarios in the Pandas DataFrame.
  325      if 'scenarios' not in inputs:
  326          scenario_label = inputs['scenario_label']
  327          inputs['scenarios'] = df[scenario_label].unique()
  328
  329      # If the scenarios are specified as a list of lists (for ensemble plots), check if they need to be transposed.
  330      # Users can now specify scenarios in two formats:
  331      #   Format A (organized by ensemble member - original format):
  332      #       [["Control", "Full feedback"], ["Control_2", "Full feedback_2"], ...]
  333      #   Format B (organized by scenario set - new user-friendly format):
  334      #       [["Control", "Control_2", ...], ["Full feedback", "Full feedback_2", ...]]
  335      # The plotting functions expect Format A internally, so Format B will be automatically transposed.
  336      if check_is_list_of_lists(inputs['scenarios']):
  337          scenario_sets = inputs.get('scenario_sets', None)
  338          inputs['scenarios'], was_transposed = transpose_scenarios_if_needed(inputs['scenarios'], scenario_sets)
  339          if was_transposed and inputs['notify_scenarios_transposed']:
  340              print(f"Note: Scenarios were automatically transposed from 'organized by scenario set' format to 'organized by ensemble member' format.")
  341
  342      # Downstream loops expect regions to be indexed by category. If the user
  343      # supplied one region list for all categories, copy that list for every
  344      # category here.
  345      if 'regions' not in inputs or not isinstance(inputs['regions'], dict):
  346          inputs['regions'] = dict.fromkeys(categories, inputs['regions'])
  347
  348      # Check that the value_label column exists in the DataFrame.
  349      value_label = inputs['value_label']
  350      if value_label not in df.columns:
  351          raise KeyError(f"Error: value_label '{value_label}' not found in '{inputs['output_file']}'. "
  352                         f"Available columns: {list(df.columns)}")
  353
  354      # area_weighted_mean requires the processed CSV to include area metadata.
  355      # GCAM area columns are added during gcam_process_extracted_data.py using a
  356      # processed land-allocation file.
  357      if inputs['aggregation_type_in_each_year'] == 'area_weighted_mean' and 'area' not in df.columns:
  358          raise KeyError(f"Error: aggregation_type_in_each_year is 'area_weighted_mean' but no 'area' column "
  359                         f"found in '{inputs['output_file']}'. Consider using 'mean' instead, or add area data "
  360                         f"with gcam_process_extracted_data.py and a land_allocation_file entry.")
  361
  362      # Warn if any specified scenario is not found in the DataFrame.
  363      scenario_label = inputs['scenario_label']
  364      if scenario_label in df.columns:
  365          available_scenarios = set(df[scenario_label].unique())
  366          specified_scenarios = inputs['scenarios']
  367          if check_is_list_of_lists(specified_scenarios):
  368              flat_scenarios = [s for member in specified_scenarios for s in member]
  369          else:
  370              flat_scenarios = list(specified_scenarios)
  371          missing_scenarios = [s for s in flat_scenarios if s not in available_scenarios]
  372          if missing_scenarios:
  373              print(f"Warning: the following scenarios were not found in '{inputs['output_file']}' "
  374                    f"and will produce empty plots: {missing_scenarios}")
  375
  376      # Extract units from the CSV's units column (first non-null, non-NA value).
  377      # Users can override this with a literal "units" entry in the JSON.
  378      units_label = inputs['units_label']
  379      if 'units' not in inputs:
  380          if units_label in df.columns:
  381              units_values = df[units_label].dropna().unique()
  382              units_values = [u for u in units_values if str(u).strip().upper() not in ('NA', 'NONE', '')]
  383              inputs['units'] = str(units_values[0]) if units_values else None
  384          else:
  385              inputs['units'] = None
  386
  387      # Set the plot title to the full y_label if not specified by the user.
  388      if inputs['title'] is None:
  389          inputs['title'] = inputs['y_label']
  390
  391      # Keep the full quantity name in the title, but abbreviate long y-axis
  392      # labels so they fit better in dense time-series plots.
  393      if inputs['y_label_abbreviated'] is None:
  394          inputs['y_label_abbreviated'] = abbreviate_label(inputs['y_label'])
  395
  396      return inputs

  873                              ax.fill_between(x, y-error, y+error, color=line_color, alpha=error_bars_alpha)
  874
  875                      if num_scenario_sets > 1 and scenario_set_index > 0:
  876                          # Compare the whole control set-mean time series against
  877                          # the current set-mean time series. Require at least two
  878                          # numeric points in each series to avoid scipy warnings
  879                          # on empty or single-point inputs.
  880                          control_data = df_all_set_means[f'set=0_cat={category_index}_reg={region_index}']
  881                          test_data = df_all_set_means[f'set={scenario_set_index}_cat={category_index}_reg={region_index}']
  882                          control_num = pd.to_numeric(control_data, errors='coerce').dropna()
  883                          test_num = pd.to_numeric(test_data, errors='coerce').dropna()
  884                          if len(control_num) >= 2 and len(test_num) >= 2:
  885                              with warnings.catch_warnings(record=True) as w:
  886                                  warnings.simplefilter('always', RuntimeWarning)
  887                                  ttest = stats.ttest_ind(control_num, test_num)
  888                                  if w:
  889                                      print(f'Warning (t-test) in {plot_name}: {w[0].message}')
  890                              label = f'set={scenario_set_index}, category={category_to_label_str(category)}, region={region}'
  891                              print_p_values(ttest, label, p_value_threshold, p_value_file, plot_name, p_value_file_print_only_if_below_threshold)
  892                      elif num_scenario_sets == 1 and num_regions > 1:
  893                          # If there is only one scenario set, but multiple regions for that set, perform inter-regional t-tests for each category.
  894                          control_data = df_all_set_means[f'set=0_cat={category_index}_reg=0']
  895                          test_data = df_all_set_means[f'set=0_cat={category_index}_reg={region_index}']
  896                          control_num = pd.to_numeric(control_data, errors='coerce').dropna()
  897                          test_num = pd.to_numeric(test_data, errors='coerce').dropna()

  899                              with warnings.catch_warnings(record=True) as w:
  900                                  warnings.simplefilter('always', RuntimeWarning)
  901                                  ttest = stats.ttest_ind(control_num, test_num)
  902                                  if w:
  903                                      print(f'Warning (t-test) in {plot_name}: {w[0].message}')
  904                              label = f'category={category_to_label_str(category)}, region={region}'
  905                              print_p_values(ttest, label, p_value_threshold, p_value_file, plot_name, p_value_file_print_only_if_below_threshold)
  906
  907                      # Also compare ensemble members year by year. Significant
  908                      # years are marked directly on the plotted curve.
  909                      if num_scenario_sets > 1 and scenario_set_index > 0:
  910                          columns_control_set = [column for column in df_all_time_series.columns if
  911                                          (f'set=0' in column and f'cat={category_index}_reg={region_index}' in column)]
  912                          columns_this_set = [column for column in df_all_time_series.columns if
  913                                          (f'set={scenario_set_index}' in column and f'cat={category_index}_reg={region_index}' in column)]
  914                          p_values = df_all_time_series.apply(perform_ttest, columns_set_1=columns_control_set, \
  915                                                                 columns_set_2=columns_this_set, axis=1).fillna(1)
  916                          # If the p-value is below the threshold, add a marker to the plot at that time period (year).
  917                          mask = p_values <= p_value_threshold
  918                          ax.plot(df_all_time_series['Year'][mask], y_series[mask], color=line_color, linestyle='None', linewidth=linewidth, \
  919                                    marker=marker, markersize=p_value_marker_size)
  920
  921                  # Optionally include the overall mean across all scenario sets and error bars indicating the standard deviation for group/set means.
  922                  if include_mean_across_all_data:
  923                      if plot_percent_difference:
  924                      # If plotting a percent difference, do not include the first scenario set in calculating the overall mean.
  925                          columns = [column for column in df_all_set_means.columns if column != 'Year' \
  926                                     and ('set=0' not in column and f'cat={category_index}_reg={region_index}' in column)]
  927                      else:
  928                          columns = [column for column in df_all_set_means.columns if column != 'Year' \
  929                                     and f'cat={category_index}_reg={region_index}' in column]
  930                      x = df_all_set_means['Year']
  931                      y = df_all_set_means[columns].mean(axis=1)
  932                      label = 'Mean'
  933                      if num_categories > 1 and num_regions > 1:
  934                          label += f' ({category_to_label_str(category)}_{region})'
  935                      elif num_categories > 1:
  936                          label += f' ({category_to_label_str(category)})'
  937                      elif num_regions > 1:
  938                          label += f' ({region})'
  939                      ax.plot(x, y, label=label, color='k', linestyle=linestyle, linewidth=linewidth, marker=marker, markersize=marker_size)
  940                      if std_mean_across_all_data_multiplier:
  941                          error = df_all_set_means[columns].std(axis=1)*std_mean_across_all_data_multiplier
  942                          ax.fill_between(x, y-error, y+error, color='k', alpha=error_bars_alpha)
  943
  944      # Finalize the time series plot now that all curves have been processed.
  945      # The custom grouped legend has already been attached, so disable the
  946      # generic legend behavior inside set_figure_options.

  950                                plot_colors, markers, linestyle_tuples, marker_size, linewidth,
  951                                category_label, col_unique_values, plot_percent_difference, plot_options)
  952      set_figure_options(fig, ax, {**plot_options, 'legend_on': False})
  953
