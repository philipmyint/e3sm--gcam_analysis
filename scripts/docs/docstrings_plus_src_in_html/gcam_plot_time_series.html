   23  def abbreviate_label(label, max_length=30):
   24      """Return an abbreviated version of label if it exceeds max_length characters."""
   25      if len(label) <= max_length:
   26          return label
   27      # Do not abbreviate labels that contain LaTeX/math text; slicing those
   28      # strings can produce invalid markup such as an unmatched "$".
   29      if '$' in label:
   30          return label
   31      words = label.replace('_', ' ').split()
   32      abbreviated_words = [w[:4] + '.' if len(w) > 5 else w for w in words]
   33      result = ' '.join(abbreviated_words)
   34      if len(result) > max_length:
   35          result = result[:max_length].rstrip() + '...'
   36      return result

   38  def label_already_contains_units(label, units):
   39      """Return True if the user-supplied label already includes the units text."""
   40      if not label or not units:
   41          return False
   42      # Compare a simplified text form so labels such as "thousands km$^2$" are
   43      # recognized as containing CSV units such as "thous km2".
   44      def normalize(text):
   45          text = str(text).lower().replace('thousands', 'thous')
   46          return re.sub(r'[^a-z0-9%]+', '', text)
   47      return normalize(units) in normalize(label)

  247  def process_inputs(inputs):
  248      """
  249      Processes a dictionary of inputs (keys are options, values are choices for those options) for creating time series plots.
  250
  251      Parameters:
  252          inputs: Dictionary containing the user plotting choice inputs for different options. This dictionary may be incomplete or have invalid values.
  253
  254      Returns:
  255          Dictionary that completely specifies all plotting options.
  256          If the user did not make a choice for a particular option, the default choice for that plotting option will be selected.
  257      """
  258      # Check that the output file exists and is non-empty before trying to read it.
  259      if not os.path.exists(inputs['output_file']):
  260          print(f"Warning: output file not found, skipping: '{inputs['output_file']}'")
  261          return None
  262      if os.path.getsize(inputs['output_file']) == 0:
  263          print(f"Warning: output file is empty, skipping: '{inputs['output_file']}'")
  264          return None
  265      df = read_file_into_dataframe(inputs['output_file'])
  266
  267      # If the category label (e.g., sector or landtype) has not been specified, use the default value.
  268      if 'category_label' not in inputs:
  269          category_label = default_inputs['category_label']
  270          inputs['category_label'] = category_label
  271      else:
  272          category_label = inputs['category_label']
  273
  274      # If the list of categories (e.g., sectors or landtypes) has not been
  275      # specified, populate it from the data. For a multi-column category_label,
  276      # each category is a tuple of values drawn from those columns.
  277      if 'categories' not in inputs:
  278          if isinstance(category_label, list) and all(c in df.columns for c in category_label):
  279              combos = df[category_label].drop_duplicates()
  280              inputs['categories'] = list(combos.itertuples(index=False, name=None))
  281              if 'categories_to_exclude' in inputs:
  282                  exclude = {tuple(e) if isinstance(e, list) else e for e in inputs['categories_to_exclude']}
  283                  inputs['categories'] = [c for c in inputs['categories'] if c not in exclude]
  284          elif not isinstance(category_label, list) and category_label in df.columns:
  285              inputs['categories'] = df[category_label].unique()
  286              if 'categories_to_exclude' in inputs:
  287                  inputs['categories'] = [column for column in inputs['categories'] if column not in inputs['categories_to_exclude']]
  288          else:
  289              inputs['categories'] = 'All'
  290      categories = inputs['categories']
  291      # A multi-column category can also be supplied as a dict, for example:
  292      # {"sector": ["crop"], "technology": ["Corn", "Wheat"]}. Expand that into
  293      # explicit tuples so the rest of the code can handle categories uniformly.
  294      if isinstance(categories, dict) and isinstance(category_label, list):
  295          per_column = [categories[col] for col in category_label]
  296          categories = list(itertools.product(*per_column))
  297          inputs['categories'] = categories
  298      # If the user entered a string indicating a single category, put that string in a list.
  299      elif isinstance(categories, str):
  300          categories = [categories]
  301          inputs['categories'] = categories
  302      elif isinstance(category_label, list):
  303          # Convert any list-type categories (from JSON) to tuples.
  304          categories = [tuple(c) if isinstance(c, list) else c for c in categories]
  305          inputs['categories'] = categories
  306
  307      # Create the plot directory if it does not already exist. By default, put
  308      # the p-value output file in the same directory as the plots.
  309      if 'plot_directory' not in inputs:
  310          inputs['plot_directory'] = default_inputs['plot_directory']
  311      if not os.path.exists(inputs['plot_directory']):
  312          os.makedirs(inputs['plot_directory'])
  313      if 'p_value_file' not in inputs:
  314          inputs['p_value_file'] = os.path.join(inputs['plot_directory'], default_inputs['p_value_file'])
  315
  316      # Use the output CSV name as the default y-axis label and plot name. Drop
  317      # "_processed" so generated plot labels read like user-facing quantities
  318      # rather than pipeline artifact names.
  319      index_of_last_backslash = inputs['output_file'].rfind('/')
  320      index_of_dot_csv = inputs['output_file'].find('.csv')
  321      if index_of_last_backslash == -1:
  322          output_file_name = inputs['output_file'][:index_of_dot_csv]
  323      else:
  324          output_file_name = inputs['output_file'][index_of_last_backslash+1:index_of_dot_csv]
  325      if 'y_label' not in inputs:
  326          inputs['y_label'] = output_file_name
  327      inputs['y_label'] = inputs['y_label'].replace('_processed', '')
  328      output_file_name_clean = output_file_name.replace('_processed', '')
  329      if 'plot_name' not in inputs:
  330          inputs['plot_name'] = os.path.join(inputs['plot_directory'], 'time_series_' + output_file_name_clean)
  331      elif 'plot_name' in inputs and '/' not in inputs['plot_name']:
  332          # If the user specified only a file name (no path) for the plot name, put the plot in the plot directory.
  333          inputs['plot_name'] = os.path.join(inputs['plot_directory'], inputs['plot_name'].replace('_processed', ''))
  334
  335      # Add keys for plotting options that have not been specified in the inputs dictionary and use default values for them.
  336      for key in default_inputs.keys():
  337          if key not in inputs:
  338              inputs[key] = default_inputs[key]
  339
  340      # If the scenarios have not been specified, use all the scenarios in the Pandas DataFrame.
  341      if 'scenarios' not in inputs:
  342          scenario_label = inputs['scenario_label']
  343          inputs['scenarios'] = df[scenario_label].unique()
  344
  345      # If the scenarios are specified as a list of lists (for ensemble plots), check if they need to be transposed.
  346      # Users can now specify scenarios in two formats:
  347      #   Format A (organized by ensemble member - original format):
  348      #       [["Control", "Full feedback"], ["Control_2", "Full feedback_2"], ...]
  349      #   Format B (organized by scenario set - new user-friendly format):
  350      #       [["Control", "Control_2", ...], ["Full feedback", "Full feedback_2", ...]]
  351      # The plotting functions expect Format A internally, so Format B will be automatically transposed.
  352      if check_is_list_of_lists(inputs['scenarios']):
  353          scenario_sets = inputs.get('scenario_sets', None)
  354          inputs['scenarios'], was_transposed = transpose_scenarios_if_needed(inputs['scenarios'], scenario_sets)
  355          if was_transposed and inputs['notify_scenarios_transposed']:
  356              print(f"Note: Scenarios were automatically transposed from 'organized by scenario set' format to 'organized by ensemble member' format.")
  357
  358      # Downstream loops expect regions to be indexed by category. If the user
  359      # supplied one region list for all categories, copy that list for every
  360      # category here.
  361      if 'regions' not in inputs or not isinstance(inputs['regions'], dict):
  362          inputs['regions'] = dict.fromkeys(categories, inputs['regions'])
  363
  364      # Check that the value_label column exists in the DataFrame.
  365      value_label = inputs['value_label']
  366      if value_label not in df.columns:
  367          raise KeyError(f"Error: value_label '{value_label}' not found in '{inputs['output_file']}'. "
  368                         f"Available columns: {list(df.columns)}")
  369
  370      # area_weighted_mean requires the processed CSV to include area metadata.
  371      # GCAM area columns are added during gcam_process_extracted_data.py using a
  372      # processed land-allocation file.
  373      if inputs['aggregation_type_in_each_year'] == 'area_weighted_mean' and 'area' not in df.columns:
  374          raise KeyError(f"Error: aggregation_type_in_each_year is 'area_weighted_mean' but no 'area' column "
  375                         f"found in '{inputs['output_file']}'. Consider using 'mean' instead, or add area data "
  376                         f"with gcam_process_extracted_data.py and a land_allocation_file entry.")
  377
  378      # Warn if any specified scenario is not found in the DataFrame.
  379      scenario_label = inputs['scenario_label']
  380      if scenario_label in df.columns:
  381          available_scenarios = set(df[scenario_label].unique())
  382          specified_scenarios = inputs['scenarios']
  383          if check_is_list_of_lists(specified_scenarios):
  384              flat_scenarios = [s for member in specified_scenarios for s in member]
  385          else:
  386              flat_scenarios = list(specified_scenarios)
  387          missing_scenarios = [s for s in flat_scenarios if s not in available_scenarios]
  388          if missing_scenarios:
  389              print(f"Warning: the following scenarios were not found in '{inputs['output_file']}' "
  390                    f"and will produce empty plots: {missing_scenarios}")
  391
  392      # Extract units from the CSV's units column (first non-null, non-NA value).
  393      # Users can override this with a literal "units" entry in the JSON.
  394      units_label = inputs['units_label']
  395      if 'units' not in inputs:
  396          if units_label in df.columns:
  397              units_values = df[units_label].dropna().unique()
  398              units_values = [u for u in units_values if str(u).strip().upper() not in ('NA', 'NONE', '')]
  399              inputs['units'] = str(units_values[0]) if units_values else None
  400          else:
  401              inputs['units'] = None
  402
  403      # Set the plot title to the full y_label if not specified by the user.
  404      if inputs['title'] is None:
  405          inputs['title'] = inputs['y_label']
  406
  407      # Keep the full quantity name in the title, but abbreviate long y-axis
  408      # labels so they fit better in dense time-series plots.
  409      if inputs['y_label_abbreviated'] is None:
  410          inputs['y_label_abbreviated'] = abbreviate_label(inputs['y_label'])
  411
  412      return inputs
