   89          category_label = inputs['category_label']
   90

   92      # category column is absent, use "All" so the plot aggregates over the whole file.

  101      # If the user entered a string indicating a single category, put that string in a list.
  102      if isinstance(categories, str):

  105
  106      # Create the plot directory if it does not already exist.
  107      if 'plot_directory' not in inputs:
  108          inputs['plot_directory'] = default_inputs['plot_directory']

  130              if num_scenario_sets == 1:
  131                  scenario_part = str(scenario_sets[0] if scenario_sets else 'set0').replace(' ', '_')
  132              elif inputs.get('plot_percent_difference', False):
  133                  scenario_part = 'pct_diff'
  134              else:
  135                  scenario_part = 'comparison'
  136          elif len(scenarios) == 1:
  137              scenario_part = str(scenarios[0]).replace(' ', '_')
  138          elif inputs.get('plot_percent_difference', False):
  139              scenario_part = 'pct_diff'
  140          else:
  141              scenario_part = 'comparison'

  143          plot_years = inputs.get('plot_years', default_inputs['plot_years'])
  144          if plot_years is None:

  146          elif len(plot_years) == 1:

  151          name = f'box_plot_{output_file_name}_{scenario_part}_{year_part}'

  153      elif 'plot_name' in inputs and '/' not in inputs['plot_name']:
  154          # If the user specified only a file name (no path) for the plot name, put the plot in the plot directory.

  158      for key in default_inputs.keys():

  168      # Users can now specify scenarios in two formats:
  169      #   Format A (organized by ensemble member - original format):
  170      #       [["Control", "Full feedback"], ["Control_2", "Full feedback_2"], ...]
  171      #   Format B (organized by scenario set - new user-friendly format):

  174      if check_is_list_of_lists(inputs['scenarios']):

  176          inputs['scenarios'], was_transposed = transpose_scenarios_if_needed(inputs['scenarios'], scenario_sets)

  187      if value_label not in df.columns:
  188          raise KeyError(f"Error: value_label '{value_label}' not found in '{inputs['output_file']}'. "
  189                         f"Available columns: {list(df.columns)}")
  190
  191      # area_weighted_mean requires area metadata from gcam_process_extracted_data.py.
  192      if inputs['aggregation_function'] == 'area_weighted_mean' and 'area' not in df.columns:
  193          raise KeyError(f"Error: aggregation_function is 'area_weighted_mean' "
  194                         f"but no 'area' column found in '{inputs['output_file']}'. "
  195                         f"Consider using 'mean' or 'sum' instead, or add area data "
  196                         f"with gcam_process_extracted_data.py and a land_allocation_file entry.")

  198      # Check that hue and x_variable columns exist in the DataFrame if specified.
  199      hue = inputs.get('hue', None)
  200      if hue and hue not in df.columns:
  201          raise KeyError(f"Error: hue '{hue}' not found in '{inputs['output_file']}'. "
  202                         f"Available columns: {list(df.columns)}")
  203      x_variable = inputs.get('x_variable', inputs['category_label'])
  204      if x_variable not in df.columns:
  205          raise KeyError(f"Error: x_variable '{x_variable}' not found in '{inputs['output_file']}'. "
  206                         f"Available columns: {list(df.columns)}")
  207
  208      # Warn if any specified scenario is not found in the DataFrame.
  209      scenario_label = inputs['scenario_label']
  210      if scenario_label in df.columns:
  211          available_scenarios = set(df[scenario_label].unique())
  212          specified_scenarios = inputs['scenarios']
  213          if check_is_list_of_lists(specified_scenarios):
  214              flat_scenarios = [s for member in specified_scenarios for s in member]
  215          else:
  216              flat_scenarios = list(specified_scenarios)
  217          missing_scenarios = [s for s in flat_scenarios if s not in available_scenarios]
  218          if missing_scenarios:
  219              print(f"Warning: the following scenarios were not found in '{inputs['output_file']}' "
  220                    f"and will produce empty plots: {missing_scenarios}")
  221
  222      # Verify the plot directory is writable before doing the plotting work.
  223      plot_name = inputs['plot_name']
  224      plot_dir = os.path.dirname(plot_name) or '.'
  225      if not os.access(plot_dir, os.W_OK):
  226          raise OSError(f"Error: plot directory is not writable: '{plot_dir}'")
  227
  228      return inputs

  230  def plot_box_and_whiskers(inputs):
  231      """
  232      Creates a box plot (box-and-whisker plot) and perform basic statistical analysis for a single output file. The data in the file are organized
  233      into scenarios or scenario sets, categories, and regions.
  234      Types of plots: 1) Plots in which each scenario is treated as an individual data set (not grouped into an ensemble) in the collection.
  235      2) Ensemble plots in which the scenarios are grouped into sets (referred to as an ensemble) and results from of each set are plotted.
  236      The scenarios must be specified as a list of lists (nested list) for ensemble plots.
  237
  238      Parameters:
  239          input: Dictionary containing user inputs for different plotting options, where the keys are options and values are choices for those options.
  240                 This dictionary is assumed to be complete (pre-processed).
  241
  242      Returns:
  243          N/A.
  244      """
  245      # This function creates one multi-panel box-and-whisker figure from a single
  246      # processed GCAM CSV. The data may be split by scenario, scenario set,
  247      # category, region, basin, hue, and year depending on the JSON options.
  248      start_time = time.time()
  249      output_file = inputs['output_file']
  250
  251      # Extract all other plotting options.
  252      basin_label = inputs['basin_label']
  253      basins = inputs['basins']
  254      categories = inputs['categories']
  255      category_label = inputs['category_label']
  256      fill_boxes = inputs['fill_boxes']
  257      height = inputs['height']
  258      hue = inputs['hue']
  259      key_columns = inputs['key_columns']
  260      landtype_groups = inputs['landtype_groups']
  261      legend_label_size = inputs['legend_label_size']
  262      legend_num_columns = inputs['legend_num_columns']
  263      legend_on = inputs['legend_on']
  264      legend_place_outside = inputs['legend_place_outside']
  265      legend_x_offset = inputs['legend_x_offset']
  266      linewidth = inputs['linewidth']
  267      marker_size = inputs['marker_size']
  268      multiplier = inputs['multiplier']
  269      aggregation_function = inputs['aggregation_function']
  270      plot_colors = inputs['plot_colors']
  271      plot_name = inputs['plot_name']
  272      plot_percent_difference = inputs['plot_percent_difference']
  273      plot_years = inputs['plot_years']
  274      plot_type = inputs['plot_type']
  275      produce_png = inputs['produce_png']
  276      region_label = inputs['region_label']
  277      regions = inputs['regions']
  278      scenario_label = inputs['scenario_label']
  279      scenario_sets = inputs['scenario_sets']
  280      scenarios = inputs['scenarios']
  281      use_latex = inputs['use_latex']
  282      value_label = inputs['value_label']
  283      width = inputs['width']
  284      x_label = inputs['x_label']
  285      x_label_size = inputs['x_label_size']
  286      x_scale = inputs['x_scale']
  287      x_tick_labels = inputs['x_tick_labels'] or {}
  288      x_tick_label_size = inputs['x_tick_label_size']
  289      x_variable = inputs.get('x_variable', category_label)
  290      y_label = inputs['y_label']
  291      y_label_size = inputs['y_label_size']
  292      y_limits = inputs['y_limits']
  293      y_scale = inputs['y_scale']
  294      y_tick_label_size = inputs['y_tick_label_size']
  295      year_label = inputs['year_label']
  296
  297      # Collect the plotting options that are shared across all year panels.
  298      plot_options = dict(width=width, height=height, name=plot_name, x_label=x_label, y_label=fr'{y_label}')
  299      plot_options.update(zip(['x_scale', 'y_scale', 'y_limits', 'use_latex'], [x_scale, y_scale, y_limits, use_latex]))
  300      plot_options.update(zip(['x_tick_label_size', 'y_tick_label_size', 'legend_on'], [x_tick_label_size, y_tick_label_size, legend_on]))
  301      plot_options.update(zip(['x_label_size', 'y_label_size', 'legend_label_size'], [x_label_size, y_label_size, legend_label_size]))
  302      plot_options.update(zip(['produce_png', 'legend_x_offset', 'legend_place_outside'], [produce_png, legend_x_offset, legend_place_outside]))
  303      plot_options.update(zip(['legend_num_columns'], [legend_num_columns]))
  304
  305      # Use LaTeX fonts for figures and set font size of tick labels.
  306      setup_plot_params(plot_options)
  307
  308      # Read the data file, filter to requested years, and apply the user-specified multiplier.
  309      df = read_file_into_dataframe(output_file)
  310      if plot_years is not None:
  311          df = df[df[year_label].isin(plot_years)]
  312      df[value_label] *= multiplier
  313
  314      # Create regional subsets for plotting. "Global" is represented by a copy
  315      # of the whole DataFrame with the region label overwritten, so it pools all regions.
  316      dataframes = []
  317      if 'Global' in regions:
  318          df_global = df.copy()
  319          df_global[region_label] = 'Global'
  320          dataframes.append(df_global)
  321          regions.remove('Global')
  322      if regions:
  323          df_these_regions = df[df[region_label].isin(regions)]
  324          dataframes.append(df_these_regions)
  325      if basins:
  326          df_these_basins = df[df[basin_label].isin(basins)]
  327          dataframes.append(df_these_basins)
  328      df = pd.concat(dataframes)
  329
  330      # Option 1: individual plots, where each scenario remains its own plotted data set.
  331      if not check_is_list_of_lists(scenarios) or plot_type == 'individual':
  332
  333          # If scenarios is a list of lists, but we do not actually want to group each inner list into an ensemble,
  334          # unpack scenarios into a list of strings, with one string for each output file.
  335          if check_is_list_of_lists(scenarios) and plot_type == 'individual':
  336              scenarios = list(itertools.chain.from_iterable(scenarios))
  337              # Turn the corresponding plot_colors to list of lists with enough rows to match, then unpack them.
  338              plot_colors = list(itertools.chain.from_iterable(plot_colors))
  339
  340          # Select the scenarios of interest.
  341          df = df[df[scenario_label].isin(scenarios)]
  342          scenario_list = scenarios
  343
  344      # Option 2: ensemble plots, where scenarios are grouped into named scenario sets.
  345      elif check_is_list_of_lists(scenarios) and plot_type == 'ensemble':
  346          # For each scenario set, collect all scenarios that belong to that set and put in a DataFrame, then concatenate the DataFrames for all sets.
  347          dataframes = []
  348          for index, scenario_set_name in enumerate(scenario_sets):
  349              scenarios_in_set = [scenarios[i][index] for i in range(len(scenarios))]
  350              df_this_set = df[df[scenario_label].isin(scenarios_in_set)].copy()
  351              # All scenarios in this set will be collected together and have the same label.
  352              df_this_set[scenario_label] = scenario_set_name
  353              dataframes.append(df_this_set)
  354          # Concatenate all DataFrames for all scenario sets by row.
  355          # Averages over each scenario set will be automatically computed later when creating the box plot.
  356          df = pd.concat(dataframes)
  357          scenario_list = scenario_sets
  358
  359      # From this point on, individual and ensemble plots share the same category
  360      # preparation and plotting logic.
  361
  362      # Set the appropriate dictionary for the landtype group.
  363      if landtype_groups == 'modified':
  364          landtype_groups = gcam_landtype_groups
  365      elif landtype_groups == 'original':
  366          landtype_groups = gcam_landtype_groups_original
  367
  368      # Create separate DataFrames for three category cases: all rows, named GCAM
  369      # landtype groups (e.g., forest/crop/grass), and exact category values.
  370      dataframes = []
  371      if 'All' in categories:
  372          df_all = df.copy()
  373          df_all[category_label] = 'All'
  374          dataframes.append(df_all)
  375          categories.remove('All')
  376      if any(item in landtype_groups for item in categories):
  377          for landtype_group in landtype_groups:
  378              if landtype_group in categories:
  379                  if not df[category_label].isin(landtype_groups[landtype_group]).any():
  380                      continue
  381                  df_this_landtype_group = produce_dataframe_for_landtype_group(df, landtype_group, category_label,
  382                      value_label, landtype_groups, aggregation_function, key_columns)
  383                  dataframes.append(df_this_landtype_group)
  384                  categories.remove(landtype_group)
  385      if categories:
  386          df = df[df[category_label].isin(categories)]
  387          dataframes.append(df)
  388      df = pd.concat(dataframes).reset_index()
  389
  390      # If different plotted categories use related units such as kg/yr and kg/m2/yr,
  391      # keep the shared numerator on the y-axis and put each denominator on the x tick.
  392      units_by_category = {}
  393      if isinstance(category_label, str) and x_variable == category_label and 'units' in df.columns:
  394          for category, group in df.groupby(category_label):
  395              units = group['units'].dropna().unique()
  396              if len(units) == 1 and '/' in units[0]:
  397                  units_by_category[str(category)] = units[0]
  398          numerators = {unit.split('/', 1)[0] for unit in units_by_category.values()}
  399          if units_by_category and len(numerators) == 1:
  400              numerator = numerators.pop()
  401              if numerator not in y_label:
  402                  y_label += f' (% diff in {numerator})' if plot_percent_difference else f' ({numerator})'
  403                  plot_options['y_label'] = fr'{y_label}'
  404      units = df['units'].dropna().unique() if 'units' in df.columns else []
  405      if not units_by_category and len(units) == 1 and units[0] not in y_label:
  406          unit_label = f'% diff in {units[0]}' if plot_percent_difference else units[0]
  407          y_label += f' ({unit_label})'
  408          plot_options['y_label'] = fr'{y_label}'
  409
  410      # Each requested year gets its own panel in the same output figure.
  411      years_to_plot = plot_years if plot_years is not None else sorted(df[year_label].unique())
  412      ncols = inputs['plot_years_ncols'] or len(years_to_plot)
  413      nrows = math.ceil(len(years_to_plot) / ncols)
  414      fig, axes = plt.subplots(nrows=nrows, ncols=ncols, figsize=(width * ncols, height * nrows))
  415      axes_flat = list(axes.flat) if hasattr(axes, 'flat') else [axes]
  416
  417      for panel_idx, year in enumerate(years_to_plot):
  418          ax = axes_flat[panel_idx]
  419          df_year = df[df[year_label] == year].copy()
  420          if df_year.empty:
  421              print(f"Warning: year {year} not found in data. Skipping panel.")
  422              ax.set_visible(False)
  423              continue
  424          if plot_percent_difference:
  425              # Percent-difference box plots use the first scenario/scenario set as
  426              # the reference and then drop that reference from the plotted data.
  427              reference = df_year[df_year[scenario_label] == scenario_list[0]][value_label].to_numpy()
  428              for scenario in scenario_list:
  429                  scenario_filter = df_year[scenario_label] == scenario
  430                  df_year.loc[scenario_filter, value_label] -= reference
  431                  df_year.loc[scenario_filter, value_label] *= 100 / (reference + EPSILON)
  432              df_year = df_year[df_year[scenario_label] != scenario_list[0]]
  433
  434          if df_year.empty or not df_year[value_label].notna().any():
  435              print(f"Warning: no plottable data for year {year}. Skipping panel.")
  436              ax.set_visible(False)
  437              continue
  438
  439          num_x_variables = len(df_year[hue].unique()) if hue else len(df_year[x_variable].unique())
  440          palette = plot_colors[:num_x_variables] if not plot_percent_difference else plot_colors[1:num_x_variables + 1]
  441          box_properties = {'edgecolor': 'black'} if fill_boxes else {'color': 'black'}
  442          line_properties = {'color': 'black'}
  443          flier_properties = {'markeredgecolor': 'black', 'markerfacecolor': 'none'}
  444          if hue:
  445              sns.boxplot(df_year, x=x_variable, y=value_label, hue=hue, ax=ax, linewidth=linewidth,
  446                          palette=palette, fliersize=marker_size, fill=fill_boxes, boxprops=box_properties,
  447                          whiskerprops=line_properties, capprops=line_properties, medianprops=line_properties,
  448                          flierprops=flier_properties)
  449          else:
  450              sns.boxplot(df_year, x=x_variable, y=value_label, hue=x_variable, ax=ax, legend=False, linewidth=linewidth,
  451                          palette=palette, fliersize=marker_size, fill=fill_boxes, boxprops=box_properties,
  452                          whiskerprops=line_properties, capprops=line_properties, medianprops=line_properties,
  453                          flierprops=flier_properties)
  454          if units_by_category:
  455              ticks = ax.get_xticks()
