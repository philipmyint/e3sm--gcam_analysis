   65  def _safe_plot_name_part(value):
   66      """Return a filesystem-friendly name fragment for a year or category."""
   67      if isinstance(value, (list, tuple)):
   68          value = '_'.join(str(item) for item in value)
   69      value = str(value).strip().replace(' ', '_')
   70      return re.sub(r'[^A-Za-z0-9_.=-]+', '_', value).strip('_')

   73  def _add_suffix_to_plot_name(plot_name, *suffix_parts):
   74      """Insert non-empty suffix parts before the plot file extension."""
   75      clean_parts = [_safe_plot_name_part(part) for part in suffix_parts if part is not None]
   76      clean_parts = [part for part in clean_parts if part]
   77      if not clean_parts:
   78          return plot_name
   79      directory, filename = os.path.split(plot_name)
   80      stem, extension = os.path.splitext(filename)
   81      updated_name = f"{stem}_{'_'.join(clean_parts)}{extension}"
   82      return os.path.join(directory, updated_name) if directory else updated_name

   85  def _format_units_for_label(units, use_latex=False):
   86      """Return a readable units label, or None if the units value is empty/noisy."""
   87      if units is None:
   88          return None
   89      units_str = str(units).strip()
   90      if units_str.upper() in ('', 'NA', 'NONE', 'NAN', 'NULL') or units_str in ('(', ')'):
   91          return None
   92      units_str = units_str.replace('thous km2', 'thousands km2')
   93      if use_latex:
   94          units_str = units_str.replace('^2', r'$^2$')
   95          units_str = units_str.replace('km2', r'km$^2$')
   96          units_str = units_str.replace('m2', r'm$^2$')
   97      return units_str

  251              flat_scenarios = [s for member in specified_scenarios for s in member]

  253              flat_scenarios = list(specified_scenarios)

  260      plot_name = inputs['plot_name']
  261      plot_dir = os.path.dirname(plot_name) or '.'
  262      if not os.access(plot_dir, os.W_OK):
  263          raise OSError(f"Error: plot directory is not writable: '{plot_dir}'")

  321      width = inputs['width']

  416
  417      # Create separate DataFrames for three category cases: all rows, named GCAM
  418      # landtype groups (e.g., forest/crop/grass), and exact category values.
  419      dataframes = []
  420      if 'All' in categories:
  421          df_all = df.copy()
  422          df_all[category_label] = 'All'
  423          dataframes.append(df_all)

  425      if any(item in landtype_groups for item in categories):

  433                  categories.remove(landtype_group)
  434      if categories:
  435          df = df[df[category_label].isin(categories)]
  436          dataframes.append(df)
  437      df = pd.concat(dataframes).reset_index()
  438
  439      # If different plotted categories use related units such as kg/yr and kg/m2/yr,
  440      # keep the shared numerator on the y-axis and put each denominator on the x tick.
  441      units_by_category = {}
  442      if isinstance(category_label, str) and x_variable == category_label and 'units' in df.columns:
  443          for category, group in df.groupby(category_label):
  444              units = [_format_units_for_label(unit, use_latex) for unit in group['units'].dropna().unique()]
  445              units = [unit for unit in units if unit]
  446              if len(units) == 1 and '/' in units[0]:
  447                  units_by_category[str(category)] = units[0]
  448          numerators = {unit.split('/', 1)[0] for unit in units_by_category.values()}
  449          if units_by_category and len(numerators) == 1:
  450              numerator = numerators.pop()
  451              if numerator not in y_label:
  452                  y_label += f' (% diff in {numerator})' if plot_percent_difference else f' ({numerator})'
  453                  plot_options['y_label'] = fr'{y_label}'
  454      units = [_format_units_for_label(unit, use_latex) for unit in df['units'].dropna().unique()] if 'units' in df.columns else []
  455      units = [unit for unit in units if unit]
  456      if not units_by_category and len(units) == 1 and units[0] not in y_label:
  457          unit_label = f'% diff in {units[0]}' if plot_percent_difference else units[0]
  458          y_label += f' ({unit_label})'
  459          plot_options['y_label'] = fr'{y_label}'

  461      # Each requested year gets its own panel in the same output figure.
  462      years_to_plot = plot_years if plot_years is not None else sorted(df[year_label].unique())

  465      fig, axes = plt.subplots(nrows=nrows, ncols=ncols, figsize=(width * ncols, height * nrows))
  466      axes_flat = list(axes.flat) if hasattr(axes, 'flat') else [axes]
  467
  468      for panel_idx, year in enumerate(years_to_plot):
  469          ax = axes_flat[panel_idx]
  470          df_year = df[df[year_label] == year].copy()
  471          if df_year.empty:
  472              print(f"Warning: year {year} not found in data. Skipping panel.")
  473              ax.set_visible(False)
  474              continue
  475          if plot_percent_difference:
  476              # Percent-difference box plots use the first scenario/scenario set as
  477              # the reference and then drop that reference from the plotted data.
  478              reference = df_year[df_year[scenario_label] == scenario_list[0]][value_label].to_numpy()
  479              for scenario in scenario_list:
  480                  scenario_filter = df_year[scenario_label] == scenario
  481                  df_year.loc[scenario_filter, value_label] -= reference
  482                  df_year.loc[scenario_filter, value_label] *= 100 / (reference + EPSILON)
  483              df_year = df_year[df_year[scenario_label] != scenario_list[0]]
  484
  485          if df_year.empty or not df_year[value_label].notna().any():
  486              print(f"Warning: no plottable data for year {year}. Skipping panel.")
  487              ax.set_visible(False)
  488              continue
  489
  490          num_x_variables = len(df_year[hue].unique()) if hue else len(df_year[x_variable].unique())

  492          box_properties = {'edgecolor': 'black'} if fill_boxes else {'color': 'black'}
  493          line_properties = {'color': 'black'}
  494          flier_properties = {'markeredgecolor': 'black', 'markerfacecolor': 'none'}
  495          if hue:
  496              sns.boxplot(df_year, x=x_variable, y=value_label, hue=hue, ax=ax, linewidth=linewidth,
  497                          palette=palette, fliersize=marker_size, fill=fill_boxes, boxprops=box_properties,
  498                          whiskerprops=line_properties, capprops=line_properties, medianprops=line_properties,
  499                          flierprops=flier_properties)
  500          else:
  501              sns.boxplot(df_year, x=x_variable, y=value_label, hue=x_variable, ax=ax, legend=False, linewidth=linewidth,
  502                          palette=palette, fliersize=marker_size, fill=fill_boxes, boxprops=box_properties,
  503                          whiskerprops=line_properties, capprops=line_properties, medianprops=line_properties,
  504                          flierprops=flier_properties)
  505          if units_by_category:
  506              ticks = ax.get_xticks()
  507              labels = [tick.get_text() for tick in ax.get_xticklabels()]
  508              ax.set_xticks(ticks, labels=[
  509                  f"{x_tick_labels.get(label, label)} ({units_by_category[label].split('/', 1)[1]})"
  510                  if label in units_by_category else x_tick_labels.get(label, label)
  511                  for label in labels
  512              ])
  513          elif x_tick_labels:
  514              ticks = ax.get_xticks()
  515              labels = [tick.get_text() for tick in ax.get_xticklabels()]
  516              ax.set_xticks(ticks, labels=[x_tick_labels.get(label, label) for label in labels])
  517          plt.setp(ax.get_xticklabels(), rotation=45, ha='right', rotation_mode='anchor')
  518          ax.set_title(f'{chr(ord("a") + panel_idx)}) {year}')
  519          panel_options = plot_options.copy()
  520          panel_options['legend_on'] = legend_on if hue else False
  521          panel_options['y_label'] = ''
  522          set_figure_options(fig, ax, panel_options)
  523
  524      for panel_idx in range(len(years_to_plot), nrows * ncols):
  525          axes_flat[panel_idx].set_visible(False)
  526
  527      fig.set_size_inches(width * ncols, height * nrows)
  528      fig.supylabel(plot_options['y_label'], fontsize=y_label_size)
  529      fig.tight_layout(rect=(0.04, 0, 1, 1))
  530      plot_options['name'] = plot_name
  531      save_figure(plot_name, fig, plot_options)

  534      plt.close(fig)
