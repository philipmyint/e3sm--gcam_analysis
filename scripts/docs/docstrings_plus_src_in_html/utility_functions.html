    6      """
    7      Performs elementwise addition of two lists.
    8
    9      Parameters:
   10          list1: The first list.
   11          list2: The second list.
   12          list2_are_units: Boolean that specifies whether list2 represents the units corresponding to list1. This would be useful when forming
   13                           column headers, where list1 specifies the quantities and list2 specifies the corresponding units for those quantities.
   14
   15      Returns:
   16          A new list containing the element-wise sums of list1 and list2. If list2_are_units is True, then the elements of list2 will be in parentheses
   17          so that each element of the combined list will be of the form 'a (b)', where a is from list1 and b is from list2.
   18          Returns None if the lists are not of the same length.
   19      """
   20      if len(list1) != len(list2):
   21          return None
   22      if list2_are_units:
   23          return [a + ' (' + b + ')' for a, b in zip(list1, list2)]
   24      else:
   25          return [a + b for a, b in zip(list1, list2)]

   87      np_arrays = []

   91          np_arrays.append(array)
   92      if len(np_arrays) == 1:
   93          return array
   94      else:
   95          return np_arrays

   98      """
   99      Finds and returns the substring located between the first occurrence of start_char and the first occurrence of end_char after start_char.
  100
  101      Parameters:
  102          text: The string to search within.
  103          start_char: The character marking the beginning of the desired substring.
  104          end_char: The character marking the end of the desired substring.
  105
  106      Returns:
  107          str or None: The extracted substring if both characters are found in order, otherwise None.
  108      """
  109      start_index = text.find(start_char)
  110      if start_index == -1:
  111          return None  # Start character not found
  112      end_index = text.find(end_char, start_index + 1)
  113      if end_index == -1:
  114          return None  # End character not found after start character
  115      return text[start_index + 1:end_index]

  118      """
  119      Gets a list of complete paths for all files that are in a particular directory.
  120
  121      Parameters:
  122          path: Path of directory where we want to search for files.
  123          file_name_substrings: A list of all substrings that must be in the file names.
  124          file_extension: File extension that should be in all files.
  125
  126      Returns:
  127          A list with complete paths to all files in the directory.
  128          If both file_name_substrings and file_extension are None, then all files in the directory will be included in the list.
  129      """
  130      file_paths = []
  131      seen_names = {}  # basename -> first full path seen, for duplicate detection
  132      for root, _, files in os.walk(path):
  133          for file in files:
  134              if not file_name_substrings or all([substring in file for substring in file_name_substrings]):
  135                  if not file_extension or file.endswith(file_extension):
  136                      file_path = os.path.join(root, file)
  137                      if file in seen_names:
  138                          print(f"Warning: duplicate filename '{file}' found at '{file_path}' and '{seen_names[file]}'; keeping the first.")
  139                          continue
  140                      seen_names[file] = file_path
  141                      file_paths.append(file_path)
  142      return file_paths

  236  def transpose_scenarios_if_needed(scenarios, scenario_sets=None):
  237      """
  238      Detects the format of the scenarios list of lists representing collections of ensembles of scenarios,
  239      and transposes the list of lists if necessary to match the internal format expected by the plotting functions.
  240
  241      The plotting functions expect scenarios organized by ensemble member (rows = members, columns = scenario sets):
  242          [["Control", "Full feedback"],           # Member 1
  243           ["Control_2", "Full feedback_2"],       # Member 2
  244           ["Control_3", "Full feedback_3"]]       # Member 3
  245
  246      This function also supports an alternative user-friendly format organized by scenario set (rows = sets, columns = members):
  247          [["Control", "Control_2", "Control_3"],           # Control set
  248           ["Full feedback", "Full feedback_2", "Full feedback_3"]]  # Full feedback set
  249
  250      The function uses the following heuristics to detect the format:
  251      1. If scenario_sets is provided, its length should match the number of scenario sets (columns in the original format,
  252         rows in the alternative format). This is the primary detection method and works regardless of scenario naming conventions.
  253      2. If scenario_sets is not provided and the matrix is not square, the function assumes the format where rows > columns
  254         is the original format (more ensemble members than scenario sets is typical).
  255      3. For square matrices without scenario_sets, the function cannot determine the format and returns as-is.
  256
  257      Parameters:
  258          scenarios: A list of lists containing scenario names. Can be in either format. Scenario names can follow any
  259                     naming convention - they do not need to share common base names or follow patterns like "Name_1", "Name_2", etc.
  260          scenario_sets: Optional list of scenario set names (e.g., ["Control", "Full feedback"]). This is the recommended
  261                         way to specify the format, as it allows unambiguous detection regardless of scenario naming conventions.
  262                         When provided, the function compares len(scenario_sets) to the dimensions of the scenarios matrix
  263                         to determine which format is being used.
  264
  265      Returns:
  266          A tuple of (transposed_scenarios, was_transposed):
  267          - transposed_scenarios: The scenarios in the internal format (organized by ensemble member).
  268          - was_transposed: Boolean indicating whether the input was transposed (True if the input was in the
  269            alternative format and was converted to the internal format).
  270
  271      Examples:
  272          # Example 1: Using scenario_sets for unambiguous detection (recommended)
  273          scenarios = [["run_A", "run_B", "run_C"], ["exp_X", "exp_Y", "exp_Z"]]
  274          scenario_sets = ["Control", "Treatment"]
  275          # len(scenario_sets) = 2 matches num_rows = 2, so this is the alternative format
  276          # Result: [["run_A", "exp_X"], ["run_B", "exp_Y"], ["run_C", "exp_Z"]], True
  277
  278          # Example 2: Without scenario_sets, using row/column count heuristic
  279          scenarios = [["A1", "B1"], ["A2", "B2"], ["A3", "B3"]]  # 3 rows, 2 cols
  280          # More rows than columns suggests original format (3 members, 2 sets)
  281          # Result: scenarios (unchanged), False
  282
  283          # Example 3: Square matrix without scenario_sets (ambiguous)
  284          scenarios = [["A", "B"], ["C", "D"]]  # 2x2 - could be either format
  285          # Result: scenarios (unchanged), False
  286      """
  287      if not scenarios or not isinstance(scenarios[0], list):
  288          # Not a list of lists; return as-is.
  289          return scenarios, False
  290
  291      num_rows = len(scenarios)
  292      num_cols = len(scenarios[0]) if scenarios else 0
  293
  294      # If all inner lists don't have the same length, we cannot reliably detect format
  295      # abort with message
  296      if not all(len(row) == num_cols for row in scenarios):
  297          print("In json input file: netdcf_files requires equal length lists of file names")
  298          os._exit(1)
  299
  300      # Heuristic 1 (Primary): If scenario_sets is provided, use its length to determine the format.
  301      # This is the most reliable method and works regardless of scenario naming conventions.
  302      if scenario_sets is not None:
  303          num_sets = len(scenario_sets)
  304          if num_sets == num_rows and num_sets != num_cols:
  305              # scenario_sets length matches rows, so input is in alternative format (organized by scenario set).
  306              # Transpose to convert to internal format (organized by ensemble member).
  307              transposed = [[scenarios[row][col] for row in range(num_rows)] for col in range(num_cols)]
  308              return transposed, True
  309          elif num_sets == num_cols and num_sets != num_rows:
  310              # scenario_sets length matches columns, so input is already in internal format.
  311              return scenarios, False
  312          elif num_sets == num_rows == num_cols:
  313              # Square matrix and scenario_sets matches both dimensions - ambiguous.
  314              # In this case, we assume the user is using the new format (organized by scenario set) since that's
  315              # the more intuitive format for users. Each row in scenarios corresponds to a scenario set.
  316              transposed = [[scenarios[row][col] for row in range(num_rows)] for col in range(num_cols)]
  317              return transposed, True
  318          # If num_sets matches neither, fall through to heuristic 2.
  319
  320      # Heuristic 2: Without scenario_sets, use the shape of the matrix.
  321      # Typically, there are more ensemble members than scenario sets (e.g., 5 members x 2 sets).
  322      # So if rows > cols, assume original format; if cols > rows, assume alternative format and transpose.
  323      if num_cols > num_rows:
  324          # More columns than rows suggests alternative format (few sets, many members per set).
  325          # Transpose to convert to internal format.
  326          transposed = [[scenarios[row][col] for row in range(num_rows)] for col in range(num_cols)]
  327          return transposed, True
  328      else:
  329          # rows >= cols: assume original format or ambiguous square matrix
  330          # abort with error message
  331          print("In json input file: netdcf_files has ambiguous format. Specify netcdf_file_sets for two separate ensemble inputs and set netcdf_files accordingly (one ensemble in each sublist)")
  332          os._exit(1)
