   98  def modify_crop_names(df, columns, mean_or_sum_if_more_than_one_row_for_crop_name='area_weighted_mean'):
   99      """
  100      Applies the mappings in gcam_crop_mappings dictionary to produce a modified set of crop names in the given Pandas DataFrame.
  101      An aggregation followed by a mean or sum is performed if there happens to be more than one row that matches a value for the given set of columns.
  102
  103      Parameters:
  104          df: DataFrame to modify.
  105          columns: Columns over which to perform the aggregation (group-by) operation.
  106          mean_or_sum_if_more_than_one_row_for_crop_name: Specifies whether to calculate an area-weighted mean, a mean, or a sum after performing the aggregation.
  107
  108      Returns:
  109          DataFrame with the crop names modified so that they belong to the modified common set.
  110      """
  111      df = df.replace(gcam_crop_mappings)
  112      # Identify string columns that are not groupby keys; these cannot be aggregated numerically
  113      # and must be saved and rejoined after groupby operations.
  114      str_non_key = [c for c in df.columns if c not in columns and df[c].dtype == object]
  115      str_first = df.groupby(columns)[str_non_key].first().reset_index() if str_non_key else None
  116      df_numeric = df.drop(columns=str_non_key)
  117      if mean_or_sum_if_more_than_one_row_for_crop_name == 'mean':
  118          mean_df = df_numeric.groupby(columns).mean().reset_index()
  119          # Areas should be summed, not averaged, even in the 'mean' case.
  120          if 'area' in df.columns:
  121              area_sum = df_numeric.groupby(columns)['area'].sum().reset_index()
  122              # Replace/merge the area column in mean_df.
  123              mean_df = mean_df.drop(columns='area', errors='ignore').merge(area_sum, on=columns)
  124          if str_first is not None:
  125              mean_df = mean_df.merge(str_first, on=columns, how='left')
  126          return mean_df
  127      elif mean_or_sum_if_more_than_one_row_for_crop_name == 'sum':
  128          result = df_numeric.groupby(columns).sum().reset_index()
  129          if str_first is not None:
  130              result = result.merge(str_first, on=columns, how='left')
  131          return result
  132      elif mean_or_sum_if_more_than_one_row_for_crop_name == 'area_weighted_mean':
  133          # Identify columns to be averaged (these are numeric, non-area columns that are not part of the aggregation operation).
  134          numeric_cols = df_numeric.select_dtypes('number').columns.tolist()
  135          numeric_cols = [col for col in numeric_cols if col not in columns + ['area']]
  136
  137          # Define a function to compute area-weighted mean for each group.
  138          def weighted_mean(g):
  139              result = {}
  140              total_area = g['area'].sum()
  141              for col in numeric_cols:
  142                  if total_area != 0:
  143                      result[col] = (g[col] * g['area']).sum() / total_area
  144                  else:
  145                      result[col] = 0
  146                      # result[col] = float('nan')  # Avoid division by zero.
  147              result['area'] = total_area
  148              return pd.Series(result)
  149
  150          # Apply the weighted mean function to each group.
  151          result = df_numeric.groupby(columns).apply(weighted_mean, include_groups=False).reset_index()
  152          if str_first is not None:
  153              result = result.merge(str_first, on=columns, how='left')
  154          return result

  156  def produce_dataframe_for_landtype_group(df, category, category_label, value_label,
  157                  landtype_groups, mean_or_sum_if_more_than_one_row_in_same_landtype_group, key_columns):
  158      """
  159      Aggregates the rows of a given Pandas DataFrame that match the specified landtype_group (e.g., crop, forest, pasture, shrub, grass).
  160      Performs one of four user-specified operations on the rows in the group: mean, sum, area-weighted mean, or area-weighted sum.
  161
  162      Parameters:
  163          df: DataFrame containing the data of interest.
  164          category: String specifying the name of the landtype category of interest (e.g., 'forest', 'shrub', 'pasture').
  165          category_label: String specifying the label for the landtype column in the DataFrame (most likely this will just be 'landtype').
  166          value_label: String specifying the label for the column containing the value of interest.
  167          landtype_groups: Dictionary where the keys are landtype group names and the values are all the landtypes that belong to each group.
  168          mean_or_sum_if_more_than_one_row_in_same_landtype_group: String that indicates the operation that should be performed on each group.
  169          key_columns: Columns on which the aggregation (group-by) operation should be performed.
  170
  171      Returns:
  172          DataFrame with aggregated rows and the value_label column modified to reflect a mean, sum, area-weighted mean, or area-weighted sum.
  173      """
  174      # Get all landtypes for the group and filter the DataFrame to keep only the rows that correspond to one of the landtypes in the group.
  175      landtypes = landtype_groups[category]
  176      df = df[df[category_label].isin(landtypes)]
  177
  178      # Units are metadata, not a quantity to sum; save a single value (if consistent) and exclude them from the aggregation
  179      # so they aren't concatenated together by sum().
  180      saved_metadata = {}
  181      for metadata_col in ('units', 'area_units'):
  182          if metadata_col in df.columns:
  183              unique_values = df[metadata_col].dropna().unique()
  184              saved_metadata[metadata_col] = unique_values[0] if len(unique_values) == 1 else None
  185      df = df.drop(columns=[c for c in saved_metadata if c in df.columns])
  186
  187      # Infer groupby columns from the DataFrame when none are specified; exclude category_label so landtypes collapse into one group row.
  188      if key_columns is None:
  189          non_key = {value_label, 'area', category_label}
  190          key_columns = [c for c in df.columns if c not in non_key]
  191
  192      # dropna=False so landtypes with no meaningful value in a key column (e.g., 'water'/'fertilizer' for non-crop landtypes) aren't dropped from the group.
  193      if mean_or_sum_if_more_than_one_row_in_same_landtype_group == 'mean':
  194          landtypes_in_df = [x for x in landtypes if x in df[category_label].unique()]
  195          num_landtypes_in_df = len(landtypes_in_df)
  196          df = df.groupby(key_columns, dropna=False).sum()
  197          df.loc[:, value_label] /= num_landtypes_in_df
  198          df = df.reset_index()
  199      elif mean_or_sum_if_more_than_one_row_in_same_landtype_group == 'sum':
  200          df = df.groupby(key_columns, dropna=False).sum().reset_index()
  201      elif mean_or_sum_if_more_than_one_row_in_same_landtype_group == 'area_weighted_mean':
  202          df.loc[:, value_label] = df['area']*df[value_label]
  203          df = df.groupby(key_columns, dropna=False).sum()
  204          total_area = df['area']
  205          df.loc[:, value_label] = df.loc[:, value_label].div(total_area, axis=0)
  206          df = df.reset_index()
  207      elif mean_or_sum_if_more_than_one_row_in_same_landtype_group == 'area_weighted_sum':
  208          df.loc[:, value_label] = df['area']*df[value_label]
  209          df = df.groupby(key_columns, dropna=False).sum().reset_index()
  210      for metadata_col, metadata_value in saved_metadata.items():
  211          df[metadata_col] = metadata_value
  212      df[category_label] = category
  213      return df

  215  def add_areas_to_dataframe(df, df_land):
  216      """
  217      Adds areas from a processed detailed land allocation DataFrame as an extra 'area' column to df.
  218      Merges on all columns common to df and df_land, excluding non-key columns ('value', 'area').
  219      'sector' in df is treated as a placeholder and excluded from merge keys even if present in both.
  220      If df has a 'market' column and no merge matches are found, falls back to matching region names
  221      within the market text string.
  222
  223      Parameters:
  224          df: DataFrame to add areas to (loaded from a CSV produced by
  225              gcam_extract_csv_from_xml_or_project_files.R).
  226          df_land: Processed detailed land allocation DataFrame. Must contain at minimum the columns
  227                   'scenario', 'region', 'year', and 'value'. Additional key columns (e.g. 'basin',
  228                   'landtype', and any management columns) are determined by the land allocation file
  229                   and are used automatically as merge keys where present in both DataFrames.
  230
  231      Returns:
  232          DataFrame with added 'area' and 'area_units' columns.
  233      """
  234      # Validate that df_land has the minimum columns needed for any merge path.
  235      required_land_cols = ['scenario', 'region', 'year', 'value']
  236      missing = [c for c in required_land_cols if c not in df_land.columns]
  237      if missing:
  238          print(f"Error: df_land is missing required columns: {missing}. Available: {list(df_land.columns)}")
  239          return df
  240
  241      # Rename 'value' -> 'area' and 'units' -> 'area_units' in the land allocation DataFrame so
  242      # that df's own 'units' column (for the value data) is not used as a merge key and is not
  243      # overwritten. The land area units are carried through as a separate 'area_units' column.
  244      rename_map = {}
  245      if 'value' in df_land.columns:
  246          rename_map['value'] = 'area'
  247      if 'units' in df_land.columns:
  248          rename_map['units'] = 'area_units'
  249      area_df = df_land.rename(columns=rename_map) if rename_map else df_land.copy()
  250
  251      # Drop any existing 'area' or 'area_units' columns to avoid duplicates after merge.
  252      for col in ['area', 'area_units']:
  253          if col in df.columns:
  254              df = df.drop(col, axis=1)
  255
  256      # Determine merge columns: all columns common to df and area_df, excluding non-key columns.
  257      # 'units' is excluded so df's own units are preserved; area units come through as 'area_units'.
  258      exclude_from_merge = {'area', 'area_units', 'value', 'units'}
  259      merge_cols = [c for c in df.columns if c in area_df.columns and c not in exclude_from_merge]
  260
  261      has_area_units = 'area_units' in area_df.columns
  262
  263      # If the only merge columns are 'scenario' and 'year' and df has a 'market' column, fall back
  264      # to matching region names within the market text string for a more specific area lookup.
  265      # Otherwise merge on all common columns, summing areas for any many-to-one matches.
  266      if set(merge_cols) == {'scenario', 'year'} and 'market' in df.columns:
  267          regions = area_df['region'].unique()
  268          agg_dict = {'area': 'sum'}
  269          if has_area_units:
  270              agg_dict['area_units'] = 'first'
  271          area_by_region = area_df.groupby(['scenario', 'region', 'year']).agg(agg_dict).reset_index()
  272
  273          def get_area_from_market(row):
  274              matching = [r for r in regions if r in str(row['market'])]
  275              if not matching:
  276                  return 0
  277              matched_region = max(matching, key=len)  # prefer the most specific match
  278              mask = (
  279                  (area_by_region['scenario'] == row['scenario']) &
  280                  (area_by_region['region'] == matched_region) &
  281                  (area_by_region['year'] == row['year'])
  282              )
  283              result = area_by_region.loc[mask, 'area']
  284              return result.sum() if not result.empty else 0
  285
  286          df['area'] = df.apply(get_area_from_market, axis=1)
  287          if has_area_units:
  288              df['area_units'] = area_df['area_units'].iloc[0] if not area_df.empty else ''
  289      elif 'landtype' not in df.columns and 'sector' in df.columns and df['sector'].isin(ag_sectors).any():
  290          # Apply crop name mappings first so that original crop names (e.g. 'CornC4') are standardized
  291          # (e.g. 'Corn') to match the ag_sector values in df before grouping.
  292          # Then aggregate df_land areas by (scenario, region, ag_sector, year),
  293          # rename landtype -> sector, then include sector in the merge keys.
  294          area_df_mapped = area_df.copy()
  295          area_df_mapped['landtype'] = area_df_mapped['landtype'].replace(ag_sector_mappings)
  296          agg_dict = {'area': 'sum'}
  297          if has_area_units:
  298              agg_dict['area_units'] = 'first'
  299          area_by_sector = (area_df_mapped
  300                            .groupby(['scenario', 'region', 'landtype', 'year'], as_index=False)
  301                            .agg(agg_dict)
  302                            .rename(columns={'landtype': 'sector'}))
  303          sector_merge_cols = merge_cols + ['sector']
  304          df = df.merge(area_by_sector, on=sector_merge_cols, how='left')
  305      elif merge_cols:
  306          agg_dict = {'area': 'sum'}
  307          if has_area_units:
  308              agg_dict['area_units'] = 'first'
  309          area_summed = (area_df[merge_cols + list(agg_dict.keys())]
  310                         .groupby(merge_cols, as_index=False)
  311                         .agg(agg_dict))
  312          df = df.merge(area_summed, on=merge_cols, how='left')
  313
  314          # Fallback: for rows that still have no area match (NaN), retry with a reduced set
  315          # of merge keys that excludes any merge_col where the unmatched row has an empty
  316          # string. This handles cases where df has less detail (e.g. water='') than df_land.
  317          unmatched = df['area'].isna()
  318          if unmatched.any():
  319              empty_col_patterns = (df.loc[unmatched, merge_cols]
  320                                    .apply(lambda row: tuple(c for c in merge_cols if row[c] == ''), axis=1)
  321                                    .unique())
  322              for empty_cols in empty_col_patterns:
  323                  if not empty_cols:
  324                      continue
  325                  fallback_merge_cols = [c for c in merge_cols if c not in empty_cols]
  326                  if not fallback_merge_cols:
  327                      continue
  328                  fallback_area = (area_df[fallback_merge_cols + list(agg_dict.keys())]
  329                                   .groupby(fallback_merge_cols, as_index=False)
  330                                   .agg(agg_dict))
  331                  pattern_mask = unmatched.copy()
  332                  for c in empty_cols:
  333                      pattern_mask &= (df[c] == '')
  334                  if not pattern_mask.any():
  335                      continue
  336                  fill_df = df.loc[pattern_mask, fallback_merge_cols].merge(
  337                      fallback_area, on=fallback_merge_cols, how='left')
  338                  df.loc[pattern_mask, 'area'] = fill_df['area'].values
  339                  if has_area_units:
  340                      if 'area_units' not in df.columns:
  341                          df['area_units'] = None
  342                      df.loc[pattern_mask, 'area_units'] = fill_df['area_units'].values
  343                  unmatched = df['area'].isna()
  344
  345      df['area'] = df['area'].fillna(0)
  346      return df
